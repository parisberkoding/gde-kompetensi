<script>
/* ==========================================================================
   KONTROLER DASHBOARD
   ========================================================================== */
var DashState = { divisi: '', data: null };

function initDashboard(param) {
  'use strict';
  DashState.divisi = DashState.divisi || '';
  renderDashKerangka();
  muatDashboard();
  if (param) {
    var q = String(param);
    if (q.indexOf('divisi=') === 0) {
      DashState.divisi = decodeURIComponent(q.substring(7));
    }
  }
}

/* -------------------------------------------------------------- kerangka -- */
function renderDashKerangka() {
  var isHc = MyCertif.isHc();
  MyCertif.isiHtml('dashRoot', ''
    + '<div class="gde-page-head">'
    + '  <div>'
    + '    <div class="gde-u-eyebrow">Ringkasan</div>'
    + '    <h1>Dashboard Sertifikasi</h1>'
    + '    <p>Pantau masa berlaku sertifikasi seluruh karyawan dan lihat mana yang perlu ditindaklanjuti.</p>'
    + '  </div>'
    + '  <div class="gde-page-actions">'
    + '    <select class="gde-select" id="dashDivisi" style="min-width:210px">'
    + '      <option value="">Semua divisi</option>'
    + '    </select>'
    + '    <button class="gde-btn gde-btn--outline" type="button" id="dashRefresh">'
    + '      <i class="bi bi-arrow-clockwise"></i> Segarkan</button>'
    + (isHc ? '    <button class="gde-btn gde-btn--outline-primary" type="button" id="dashExportPdf">'
      + '      <i class="bi bi-file-earmark-pdf"></i> Ekspor PDF</button>'
      + '    <button class="gde-btn gde-btn--primary" type="button" id="dashReminder">'
      + '      <i class="bi bi-send"></i> Kirim reminder</button>' : '')
    + '  </div>'
    + '</div>'
    + '<div id="dashIsi"></div>');

  var sel = MyCertif.el('dashDivisi');
  if (sel) {
    sel.value = DashState.divisi;
    sel.addEventListener('change', function () {
      DashState.divisi = sel.value;
      muatDashboard();
    });
  }
  var ref = MyCertif.el('dashRefresh');
  if (ref) ref.addEventListener('click', function () { muatDashboard(true); });

  var ex = MyCertif.el('dashExportPdf');
  if (ex) {
    ex.addEventListener('click', function () {
      MyCertif.tombolSibuk(ex, true, 'Menyiapkan...');
      MyCertif.api('apiExportPdf', [{
        divisi: DashState.divisi,
        judul: 'Rekap Sertifikasi & Masa Berlaku'
      }]).then(function (r) {
        MyCertif.tombolSibuk(ex, false);
        MyCertif.sukses('PDF dibuat: ' + r.name + ' (' + r.jumlah + ' baris).', 'Ekspor selesai');
        window.open(r.url, '_blank');
      }).catch(function (e) {
        MyCertif.tombolSibuk(ex, false);
        MyCertif.gagal(e.message);
      });
    });
  }

  var rem = MyCertif.el('dashReminder');
  if (rem) {
    rem.addEventListener('click', function () {
      MyCertif.konfirmasi({
        judul: 'Kirim reminder sekarang',
        pesan: 'Sistem akan memeriksa seluruh sertifikasi dan mengirim email pengingat '
          + 'ke karyawan, HC, dan atasan sesuai tingkat statusnya. Email yang sudah pernah '
          + 'dikirim untuk tingkat yang sama tidak akan dikirim ulang.',
        labelOk: 'Kirim sekarang'
      }).then(function (ya) {
        if (!ya) return;
        MyCertif.tombolSibuk(rem, true, 'Mengirim...');
        MyCertif.api('apiRunReminderNow', []).then(function (r) {
          MyCertif.tombolSibuk(rem, false);
          MyCertif.sukses('Target ' + r.target + ' · terkirim ' + r.terkirim + ' · gagal ' + r.gagal + '.',
            'Reminder selesai');
          muatDashboard(true);
        }).catch(function (e) {
          MyCertif.tombolSibuk(rem, false);
          MyCertif.gagal(e.message);
        });
      });
    });
  }
}

/* ------------------------------------------------------------- pemuatan -- */
function muatDashboard(paksa) {
  MyCertif.isiHtml('dashIsi', MyCertif.pemuatHtml('Menghitung ringkasan...'));
  MyCertif.api('apiGetDashboard', [{ divisi: DashState.divisi }]).then(function (d) {
    DashState.data = d;
    isiPilihanDivisi(d.divisi_options || []);
    renderDashIsi(d);
  }).catch(function (e) {
    MyCertif.isiHtml('dashIsi', MyCertif.galatHtml(e.message));
  });
  if (paksa) MyCertif.perbaruiBadgeNotifikasi();
}

function isiPilihanDivisi(opsi) {
  var sel = MyCertif.el('dashDivisi');
  if (!sel) return;
  var nilai = DashState.divisi;
  sel.innerHTML = '<option value="">Semua divisi</option>'
    + opsi.map(function (o) {
      return '<option value="' + MyCertif.attr(o) + '"' + (o === nilai ? ' selected' : '') + '>'
        + MyCertif.esc(o) + '</option>';
    }).join('');
}

/* --------------------------------------------------------------- render -- */
function renderDashIsi(d) {
  var k = d.kpi || {};
  var isHc = MyCertif.isHc();

  var html = '';

  /* ---------- 1. KPI ---------- */
  html += '<div class="gde-grid gde-grid--4 mycertif-mb-4">'
    + kpiCard('total_pekerja', 'Pekerja terdaftar', k.total_pekerja, 'bi-people', 'blue')
    + kpiCard('total_sertifikasi', 'Total sertifikasi', k.total_sertifikasi, 'bi-award', 'blue')
    + kpiCard('kadaluarsa', 'Sudah kadaluarsa', k.kadaluarsa, 'bi-x-octagon', 'dead')
    + kpiCard('mendesak', 'Perlu ditindaklanjuti', k.mendesak, 'bi-exclamation-triangle', 'red')
    + '</div>';

  /* ---------- 1b. antrean pemeriksaan HC ---------- */
  /* Dipisah dari KPI masa berlaku: menunggu verifikasi BUKAN berarti dokumennya
     hampir habis, jadi tidak boleh dicampur dalam satu angka. */
  if (MyCertif.isHc()) {
    html += '<div class="gde-grid gde-grid--2 mycertif-mb-4">'
      + kpiCard('perlu_verifikasi', 'Menunggu verifikasi HC', k.perlu_verifikasi,
        'bi-hourglass-split', 'yellow')
      + kpiCard('perlu_revisi', 'Perlu revisi pemilik data', k.perlu_revisi,
        'bi-arrow-counterclockwise', 'red')
      + '</div>';
  }

  /* ---------- 2. baris kedua: donut + tren ---------- */
  html += '<div class="mycertif-2col mycertif-mb-4">';
  html += '<div class="gde-card"><div class="mycertif-panel-head">'
    + '<i class="bi bi-pie-chart-fill" style="color:var(--gde-blue)"></i>'
    + '<div><h3>Sebaran Status</h3><div class="mycertif-panel-head__sub">'
    + MyCertif.angka(k.total_sertifikasi) + ' sertifikasi terpantau</div></div></div>'
    + '<div class="gde-card__body">'

    + donutHtml(d.sebaran, k.total_sertifikasi)

    + '<div class="gde-u-between" style="margin-bottom:6px">'
    + '<span class="mycertif-inline-note">Kelengkapan lampiran</span>'
    + '<b style="font-size:.85rem">' + MyCertif.angka(k.ada_lampiran) + ' / ' + MyCertif.angka(k.total_sertifikasi) + '</b></div>'
    + '<div class="gde-progress"><span class="gde-progress__bar" style="width:'
    + (k.total_sertifikasi ? Math.round((k.ada_lampiran / k.total_sertifikasi) * 100) : 0) + '%"></span></div>'
    + '</div>'
    + '</div></div>';

  html += '<div class="gde-card"><div class="mycertif-panel-head">'
    + '<i class="bi bi-bar-chart-fill" style="color:var(--gde-blue)"></i>'
    + '<div><h3>Jadwal Kedaluwarsa 12 Bulan</h3><div class="mycertif-panel-head__sub">'
    + 'Jumlah sertifikasi yang berakhir tiap bulan</div></div></div>'
    + '<div class="gde-card__body">' + barsHtml(d.tren) + '</div></div>';

  html += '</div>';

  /* ---------- 3. prioritas + pekerja kritis ---------- */
  html += '<div class="mycertif-2col mycertif-mb-4">';

  html += '<div class="gde-card"><div class="mycertif-panel-head">'
    + '<i class="bi bi-exclamation-triangle-fill" style="color:var(--gde-red)"></i>'
    + '<div><h3>Perlu Segera Ditindaklanjuti</h3><div class="mycertif-panel-head__sub">'
    + MyCertif.angka(d.prioritas_total) + ' sertifikasi berstatus merah/hitam</div></div>'
    + '<div class="mycertif-panel-head__actions">'
    + '<button class="gde-btn gde-btn--outline gde-btn--sm" type="button" data-view="data">Lihat semua</button>'
    + '</div></div>';
  if (!d.prioritas || !d.prioritas.length) {
    html += '<div class="gde-card__body">' + MyCertif.kosongHtml('bi-check2-circle', 'Semua aman',

      + '<thead><tr><th>Pemegang</th><th>Sertifikasi</th><th class="gde-ctr">Berakhir</th>'

      + d.prioritas.map(function (r) {
        return '<tr style="cursor:pointer" data-dash-pekerja="' + MyCertif.attr(r.employee_id) + '">'
          + '<td class="mycertif-td-nama"><strong>' + MyCertif.esc(r.nama_pemegang || '-') + '</strong>'
          + '<span>' + MyCertif.esc(r.employee_id) + ' · ' + MyCertif.esc(r.divisi || '-') + '</span></td>'
          + '<td class="mycertif-td-sert"><strong>' + MyCertif.esc(r.nama_sertifikasi) + '</strong>'
          + '<span>' + MyCertif.esc(r.penyelenggara || '-') + '</span></td>'
          + '<td class="gde-ctr">' + MyCertif.esc(MyCertif.tglSingkat(r.expired_at)) + '</td>'

    + '<i class="bi bi-diagram-3-fill" style="color:var(--gde-blue)"></i>'
    + '<div><h3>Rekap per Divisi</h3><div class="mycertif-panel-head__sub">'
    + 'Sebaran sertifikasi dan statusnya</div></div></div>'

    html += MyCertif.kosongHtml('bi-award', 'Belum ada data', 'Data sertifikasi masih kosong.');

      return '<div style="margin-bottom:.85rem">'
        + '<div class="gde-u-between" style="margin-bottom:4px">'
        + '<span style="font-size:.78rem;font-weight:600;color:var(--gde-ink-2)">' + MyCertif.esc(j.nama) + '</span>'
        + '<span style="font-size:.72rem;color:var(--gde-muted)">' + MyCertif.angka(j.total)
        + (j.kadaluarsa ? (' · <span style="color:#C4142A">' + MyCertif.angka(j.kadaluarsa) + ' kadaluarsa</span>') : '')
        + '</span></div>'
        + '<div class="gde-progress"><span class="gde-progress__bar" style="width:' + persen + '%"></span></div>'
        + '</div>';
    }).join('');
  }
  html += '</div></div>';
  html += '</div>';

  /* ---------- 5. catatan kaki ---------- */
  html += '<p style="margin-top:1.5rem;font-size:.7rem;color:var(--gde-muted-2)">'
    + 'Data dihitung pada ' + MyCertif.esc(d.pada) + ' (' + MyCertif.esc(d.zona_waktu || '') + '). '
    + 'Ambang status: kadaluarsa &lt; 0 hari · mendekati kadaluarsa ≤ 2 bulan · '
    + 'segera perpanjang ≤ 3 bulan · perlu perhatian ≤ 6 bulan · peringatan ≤ 9 bulan.'
    + '</p>';

  MyCertif.isiHtml('dashIsi', html);

  /* klik baris -> profil pekerja */
  MyCertif.on(MyCertif.el('dashIsi'), 'click', '[data-dash-pekerja]', function (e, t) {
    var kode = t.getAttribute('data-dash-pekerja');
    MyCertif.state.paramPekerja = kode;
    MyCertif.navigasi('pekerja');
  });
}

/* ------------------------------------------------------------------ KPI -- */
function kpiCard(key, label, nilai, ikon, tone) {
  return '<div class="mycertif-mini mycertif-mini--' + tone + '">'
    + '<div class="mycertif-mini__icon"><i class="bi ' + ikon + '"></i></div>'
    + '<div style="min-width:0">'
    + '<div class="mycertif-mini__label">' + MyCertif.esc(label) + '</div>'
    + '<div class="mycertif-mini__value">' + MyCertif.angka(nilai) + '</div>'
    + '</div></div>';
}

function warnaSisa(hari) {
  if (hari === null || hari === undefined || hari === '') return 'var(--gde-muted)';
  var h = MyCertif.num(hari);
  if (h < 0) return '#7F1D1D';
  if (h <= 60) return '#1B2430';
  if (h <= 90) return '#C4142A';
  if (h <= 180) return '#8A4B00';
  if (h <= 270) return '#6B5F00';
  return '#3F6B18';
}

/* ---------------------------------------------------------------- donut -- */
function donutHtml(sebaran, total) {
  var warna = {
    green: '#8DC63F', yellow: '#FFF56B', orange: '#E07B14',
    red: '#ED1B2F', black: '#1B2430', dead: '#7F1D1D', gray: '#94A3B8'
  };
  var t = MyCertif.num(total);
  var stops = [];
  var jalan = 0;
  (sebaran || []).forEach(function (s) {
    if (!s.jumlah) return;
    var porsi = t ? (s.jumlah / t) * 100 : 0;
    var dari = jalan;
    jalan += porsi;
    stops.push((warna[s.tone] || '#94A3B8') + ' ' + dari.toFixed(2) + '% ' + jalan.toFixed(2) + '%');
  });
  var latar = stops.length
    ? 'conic-gradient(' + stops.join(',') + ')'
    : 'conic-gradient(#E4E9F0 0% 100%)';

  return '<div class="mycertif-donut" style="background:' + latar + '">'
    + '<div class="mycertif-donut__center"><b>' + MyCertif.angka(t) + '</b><span>sertifikasi</span></div>'
    + '</div>';
}

function legendaHtml(sebaran) {
  return '<div style="display:grid;gap:.55rem">'
    + (sebaran || []).map(function (s) {
      return '<div class="gde-u-between" style="gap:.5rem">'
        + '<span style="display:flex;align-items:center;gap:8px;font-size:.76rem;color:var(--gde-muted)">'
        + '<span class="gde-dot gde-dot--' + s.tone + '"></span>' + MyCertif.esc(s.label) + '</span>'
        + '<b style="font-size:.78rem;color:var(--gde-ink)">' + MyCertif.angka(s.jumlah) + '</b>'
        + '</div>';
    }).join('')
    + '</div>';
}

/* ----------------------------------------------------------------- bars -- */
function barsHtml(tren) {
  var data = tren || [];
  var max = 1;
  data.forEach(function (t) { if (t.jumlah > max) max = t.jumlah; });

  if (!data.length) return MyCertif.kosongHtml('bi-bar-chart', 'Belum ada data', '');

  return '<div class="mycertif-bars">'
    + data.map(function (t) {
      var tinggi = Math.max(Math.round((t.jumlah / max) * 128), 3);
      var warna = t.jumlah === 0 ? '#E4E9F0'
        : (t.jumlah >= max * 0.66 ? '#ED1B2F' : (t.jumlah >= max * 0.33 ? '#0B50A1' : '#8DC63F'));
      return '<div class="mycertif-bars__col" title="' + MyCertif.attr(t.label + ': ' + t.jumlah + ' sertifikasi') + '">'
        + '<span class="mycertif-bars__val">' + (t.jumlah || '') + '</span>'
        + '<div class="mycertif-bars__bar" style="height:' + tinggi + 'px;background:' + warna + '"></div>'
        + '<span class="mycertif-bars__label">' + MyCertif.esc(t.label) + '</span>'
        + '</div>';
    }).join('')
    + '</div>'
    + '<div class="mycertif-legend">'
    + '<span class="mycertif-legend__item"><span class="gde-dot gde-dot--red"></span> bulan terpadat</span>'
    + '<span class="mycertif-legend__item"><span class="gde-dot gde-dot--blue"></span> sedang</span>'
    + '<span class="mycertif-legend__item"><span class="gde-dot gde-dot--green"></span> sedikit</span>'
    + '</div>';
}
