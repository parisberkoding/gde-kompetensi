var DsState = {
  filter: {
    q: '', status: '', divisi: '', verifikasi: '',
    dari: '', sampai: '', sort: 'prioritas', dir: 'asc',
    page: 1, per_page: 25
  },
  hasil: null,
  detail: null,
  daftarPemegang: []
};

function initDataSertifikasi(param) {
  'use strict';

  if (param) {
    var q = String(param);
    if (q.indexOf('q=') === 0) DsState.filter.q = decodeURIComponent(q.substring(2));
    if (q.indexOf('status=') === 0) DsState.filter.status = decodeURIComponent(q.substring(7)).toUpperCase();
    if (q.indexOf('pekerja=') === 0) {
      DsState.filter.q = decodeURIComponent(q.substring(8));
    }
  }
  if (MyCertif.state.paramDoc) {
    var doc = MyCertif.state.paramDoc;
    MyCertif.state.paramDoc = '';
    bukaDetail(doc);
  }

  renderDsKerangka();
  muatDs(true);
}

function isiOpsiDivisiDs(opsi) {
  var sel = MyCertif.el('dsDivisi');
  if (!sel) return;
  sel.innerHTML = '<option value="">Semua divisi</option>'
    + opsi.map(function (o) {
      return '<option value="' + MyCertif.attr(o) + '">' + MyCertif.esc(o) + '</option>';
    }).join('');
  sel.value = DsState.filter.divisi;
}

/* ------------------------------------------------------------- pemuatan -- */
function muatDs(sekaligus) {
  if (sekaligus) {
    var meta = MyCertif.state.meta || {};
    isiOpsiDivisiDs(meta.divisi_options || []);
  }
  MyCertif.isiHtml('dsTabel', MyCertif.pemuatHtml('Memuat data...'));
  MyCertif.api('apiListFiles', [DsState.filter]).then(function (d) {
    DsState.hasil = d;
    renderDsRingkas(d);
    renderDsTabel(d);
  }).catch(function (e) {
    MyCertif.isiHtml('dsTabel', MyCertif.galatHtml(e.message));
  });
}

function renderDsRingkas(d) {
  var r = d.ringkasan || {};
  var defs = d.status_defs || [];
  var html = defs.filter(function (s) {
    return ['KADALUARSA', 'MENDEKATI_KADALUARSA', 'SEGERA_PERPANJANG', 'MASIH_BERLAKU'].indexOf(s.key) !== -1;
  }).map(function (s) {
    return '<div class="mycertif-mini mycertif-mini--' + s.tone + '">'
      + '<div class="mycertif-mini__icon"><i class="bi ' + s.icon + '"></i></div>'
      + '<div style="min-width:0">'
      + '<div class="mycertif-mini__label">' + MyCertif.esc(s.label) + '</div>'
      + '<div class="mycertif-mini__value">' + MyCertif.angka(r[s.key] || 0) + '</div>'
      + '</div></div>';
  }).join('');
  MyCertif.isiHtml('dsRingkas', html);
}

function renderPager(d) {
  if (d.pages <= 1) return '';
  var out = [];
  var p = d.page;
  var dari = Math.max(1, p - 2);
  var sampai = Math.min(d.pages, dari + 4);
  dari = Math.max(1, sampai - 4);

  out.push('<a href="#" data-hal="' + (p - 1) + '"' + (p <= 1 ? ' class="is-disabled"' : '') + '>'
    + '<i class="bi bi-chevron-left"></i></a>');
  if (dari > 1) {
    out.push('<a href="#" data-hal="1">1</a>');
    if (dari > 2) out.push('<span class="is-disabled">…</span>');
  }
  for (var i = dari; i <= sampai; i++) {
    out.push('<a href="#" data-hal="' + i + '"' + (i === p ? ' class="is-active"' : '') + '>' + i + '</a>');
  }
  if (sampai < d.pages) {
    if (sampai < d.pages - 1) out.push('<span class="is-disabled">…</span>');
    out.push('<a href="#" data-hal="' + d.pages + '">' + d.pages + '</a>');
  }
  out.push('<a href="#" data-hal="' + (p + 1) + '"' + (p >= d.pages ? ' class="is-disabled"' : '') + '>'
    + '<i class="bi bi-chevron-right"></i></a>');
  return out.join('');
}

/** Badge verifikasi — memakai definisi terpusat MyCertif.verifikasiBadge(). */
function badgeVerifikasi(v) { return MyCertif.verifikasiBadge(v); }

function warnaSisaDs(hari) {
  if (hari === null || hari === undefined || hari === '') return 'var(--gde-muted)';
  var h = MyCertif.num(hari);
  if (h < 0) return '#7F1D1D';
  if (h <= 60) return '#1B2430';
  if (h <= 90) return '#C4142A';
  if (h <= 180) return '#8A4B00';
  if (h <= 270) return '#6B5F00';
  return '#3F6B18';
}

function pasangEventTabel() {
  var wadah = MyCertif.el('dsTabel');

  MyCertif.on(wadah, 'click', '[data-hal]', function (e, t) {
    e.preventDefault();
    if (t.classList.contains('is-disabled')) return;
    DsState.filter.page = MyCertif.num(t.getAttribute('data-hal'), 1);
    muatDs(false);
    try { window.scrollTo({ top: 0, behavior: 'smooth' }); } catch (x) { }
  });

  MyCertif.on(wadah, 'click', '[data-aksi]', function (e, t) {
    e.stopPropagation();
    var aksi = t.getAttribute('data-aksi');
    var doc = t.getAttribute('data-doc');
    if (aksi === 'detail') bukaDetail(doc);
    if (aksi === 'verifikasi') bukaVerifikasi(doc);
    if (aksi === 'notify') bukaKirimNotifikasi(doc);
  });

  MyCertif.on(wadah, 'click', 'tbody tr[data-doc]', function (e, t) {
    if (e.target.closest('[data-aksi]') || e.target.closest('a')) return;
    bukaDetail(t.getAttribute('data-doc'));
  });
}

/* ---------------------------------------------------------- modal detail -- */
function bukaDetail(docId) {
  MyCertif.isiHtml('dsDetailIsi', MyCertif.pemuatHtml('Memuat detail...'));
  MyCertif.isiHtml('dsDetailAksi', '');
  MyCertif.modalBuka('dsModalDetail');

  MyCertif.api('apiGetFile', [docId]).then(function (d) {
    DsState.detail = d.file;
    renderDetail(d.file);
  }).catch(function (e) {
    MyCertif.isiHtml('dsDetailIsi', MyCertif.galatHtml(e.message));
  });
}

function renderDetail(f) {

    ['Divisi / Jabatan', (f.divisi || '-') + ' / ' + (f.jabatan || '-')],
    ['Nama sertifikasi', f.nama_sertifikasi],
    ['Penyelenggara', f.penyelenggara || '-'],
    ['No. sertifikat', f.no_sertifikat || '-'],
    ['Mulai pelaksanaan', MyCertif.tglIndo(f.start_pelaksanaan)],
    ['Selesai pelaksanaan', MyCertif.tglIndo(f.end_pelaksanaan)],
    ['Tanggal sertifikasi', f.tanggal_sertifikasi ? MyCertif.tglIndo(f.tanggal_sertifikasi) : '—'],
    ['Tanggal perpanjangan', f.tanggal_waktu_perpanjangan ? MyCertif.tglIndo(f.tanggal_waktu_perpanjangan) : 'Belum diperpanjang'],
    ['Masa berlaku', f.masa_berlaku_bulan ? (f.masa_berlaku_bulan + ' bulan') : '-'],
    ['Berlaku Hingga', MyCertif.tglIndo(f.expired_at)],
    ['Sisa masa berlaku', MyCertif.masaBerlakuTeks(f) + (f.sisa_hari !== null && f.sisa_hari !== undefined ? (' (' + MyCertif.sisaTeks(f.sisa_hari) + ')') : '')],
    ['Recom. Update', f.recom_update ? MyCertif.tglIndo(f.recom_update) : '-'],
    ['Status', null],
    ['Verifikasi', null],
    ['Terakhir diperbarui', f.updated_at ? MyCertif.waktuIndo(f.updated_at) : '—'],
    ['Pertama diinput', f.created_at
      ? (MyCertif.waktuIndo(f.created_at) + (f.created_by ? (' oleh ' + f.created_by) : ''))
      : '—'],
    ['Diverifikasi oleh', f.verified_by
      ? (f.verified_by + (f.verified_at ? (' · ' + MyCertif.waktuIndo(f.verified_at)) : ''))
      : 'Belum diverifikasi']
  ];

  var html = '<div class="mycertif-modal-grid">'
    + '<div>'
    + '<dl class="mycertif-dl">'
    + baris.map(function (b) {
      if (b[1] === null && b[0] === 'Status') {
        return '<dt>Status</dt><dd>' + MyCertif.statusBadge(f.status, f.status_label, f.status_tone, f.status_icon) + '</dd>';
      }
      if (b[1] === null && b[0] === 'Verifikasi') {
        return '<dt>Verifikasi</dt><dd>' + badgeVerifikasi(f.status_verifikasi)
          + (f.verified_by ? (' <span class="mycertif-inline-note">oleh ' + MyCertif.esc(f.verified_by) + '</span>') : '') + '</dd>';
      }
      return '<dt>' + MyCertif.esc(b[0]) + '</dt><dd>' + MyCertif.esc(b[1]) + '</dd>';
    }).join('')
    + '</dl>'
    + (f.catatan ? ('<div class="gde-alert gde-alert--info mycertif-mt-4">'
      + '<i class="bi bi-chat-left-text gde-alert__icon"></i>'
      + '<div class="gde-alert__body"><strong>Catatan</strong>' + MyCertif.esc(f.catatan) + '</div></div>') : '')
    + '</div>'
    + '<div>'
    + '<div class="gde-u-eyebrow" style="margin-bottom:.5rem">Lampiran</div>'
    + pratinjauLampiranDs(f)
    + '</div>'
    + '</div>';

  MyCertif.isiHtml('dsDetailIsi', html);

  /* ---- aksi modal ---- */
  var isHc = MyCertif.isHc();
  var aksi = '<button class="gde-btn gde-btn--outline" type="button" data-tutup="dsModalDetail">Tutup</button>';
  if (f.file_id) {
    aksi += '<a class="gde-btn gde-btn--outline-primary" href="' + MyCertif.attr(f.view_url) + '" target="_blank">'
      + '<i class="bi bi-box-arrow-up-right"></i> Buka di Drive</a>';
    aksi += '<a class="gde-btn gde-btn--outline-primary" href="' + MyCertif.attr(f.download_url) + '" target="_blank">'
      + '<i class="bi bi-download"></i> Unduh</a>';
  }
  if (isHc && f.status_verifikasi !== 'VERIFIED') {
    aksi += '<button class="gde-btn gde-btn--green" type="button" id="dsVerifikasiBtn">'
      + '<i class="bi bi-patch-check"></i> Verifikasi</button>';
  }
  if (isHc) {
    aksi += '<button class="gde-btn gde-btn--outline-danger" type="button" id="dsHapusBtn">'
      + '<i class="bi bi-trash"></i> Hapus</button>';

    + '<i class="bi bi-pencil"></i> Ubah</button>';
  MyCertif.isiHtml('dsDetailAksi', aksi);

  var vb = MyCertif.el('dsVerifikasiBtn');
  if (vb) vb.addEventListener('click', function () { MyCertif.modalTutup('dsModalDetail'); bukaVerifikasi(f.doc_id); });
  var hb = MyCertif.el('dsHapusBtn');
  if (hb) hb.addEventListener('click', function () { hapusDs(f); });
  var ub = MyCertif.el('dsUbahBtn');
  if (ub) ub.addEventListener('click', function () {
    MyCertif.state.paramDoc = f.doc_id;
    MyCertif.modalTutup('dsModalDetail');
    MyCertif.navigasi('register');
  });

  MyCertif.on(MyCertif.el('dsModalDetail'), 'click', '[data-tutup]', function () {
    MyCertif.modalTutup('dsModalDetail');
  });
}

function pratinjauLampiranDs(f) {
  if (!f.file_id) {
    return '<div class="gde-alert gde-alert--warning">'
      + '<i class="bi bi-paperclip gde-alert__icon"></i>'
      + '<div class="gde-alert__body"><strong>Belum ada lampiran</strong>'
      + 'Unggah berkas sertifikat agar mudah diverifikasi dan diaudit.</div></div>';
  }
  var mime = String(f.mime_type || '');
  var isGambar = mime.indexOf('image/') === 0;
  var isPdf = mime.indexOf('pdf') !== -1 || MyCertif.extDari(f.file_name) === 'pdf';

  var isi;
  if (isGambar) {
    isi = '<img src="' + MyCertif.attr(MyCertif.pratinjauUrl(f.file_id, f.mime_type, f.file_name)) + '" alt="Lampiran">';
  } else if (isPdf) {
    isi = '<iframe src="' + MyCertif.attr(f.preview_url) + '" title="Pratinjau PDF" loading="lazy"></iframe>';
  } else {
    isi = '<div class="mycertif-lampiran__empty">'
      + '<i class="bi ' + MyCertif.attr(f.file_icon) + '"></i>'
      + '<strong>' + MyCertif.esc(f.file_name || 'Berkas') + '</strong>'
      + '<p class="mycertif-inline-note">Pratinjau tidak tersedia untuk jenis berkas ini. '
      + 'Klik "Buka di Drive" untuk melihat isinya.</p></div>';
  }

  return '<div style="border:1px solid var(--gde-line);border-radius:var(--gde-r-md);overflow:hidden">'
    + '<div class="mycertif-lampiran__toolbar">'
    + '<i class="bi ' + MyCertif.attr(f.file_icon) + '" style="color:var(--gde-blue);font-size:1.1rem"></i>'
    + '<span style="font-size:.78rem;font-weight:600;flex:1;overflow:hidden;text-overflow:ellipsis;white-space:nowrap">'
    + MyCertif.esc(f.file_name || '-') + '</span>'
    + '<span class="gde-chip gde-chip--soft" style="font-size:.65rem">' + MyCertif.esc(f.ukuran_label) + '</span>'
    + '</div>'
    + '<div class="mycertif-lampiran__viewer">' + isi + '</div>'
    + '</div>';
}

/* ------------------------------------------------------------ verifikasi -- */
/**
 * HC memeriksa kebenaran data lalu memilih: Terverifikasi atau Perlu Revisi.
 *
 * Istilahnya "Perlu Revisi", bukan "Ditolak": datanya tidak dibuang, pemiliknya
 * hanya diminta memperbaiki. Catatan alasan WAJIB diisi saat meminta revisi supaya
 * pemiliknya tahu bagian mana yang salah — dan catatan itu ikut terkirim sebagai
 * notifikasi in-app ke pemilik data.
 */
/**
 * Tombol "Kirim notifikasi" pada tabel: HC mengirim pengingat untuk SATU data
 * sertifikasi, di luar jadwal otomatis.
 *
 * Alamat tujuan diambil dari Master_User (Email_Pribadi lebih diutamakan) —
 * tidak pernah dari alamat bawaan aplikasi. Penerima ditampilkan lebih dulu
 * supaya HC tahu ke mana pesannya pergi sebelum menekan kirim.
 */
function bukaKirimNotifikasi(docId) {
  MyCertif.api('apiPratinjauNotifikasi', [docId]).then(function (p) {
    if (!p.ada_tujuan) {
      return MyCertif.konfirmasi({
        judul: 'Belum ada alamat email tujuan',
        pesanHtml: '<b>' + MyCertif.esc(p.nama_sertifikasi) + '</b><br>'
          + MyCertif.esc(p.nama_pemegang || p.employee_code) + ' (' + MyCertif.esc(p.employee_code) + ')'
          + '<br><br><b>Email pribadi</b> dan <b>email perusahaan</b> pemilik ini belum diisi di '
          + 'Master User, jadi pengingat tidak bisa dikirim ke mana pun.'
          + '<br><br>Lengkapi dulu di halaman <b>Master User</b>, lalu coba lagi.',
        labelOk: 'Buka Master User',
        labelBatal: 'Tutup'
      }).then(function (buka) {
        if (buka) MyCertif.navigasi('masteruser');
        return null;
      });
    }

    var tujuan = p.ke.map(function (e) { return MyCertif.esc(e); }).join('<br>');
    var cc = p.cc.length
      ? ('<br><br><b>Ditembuskan ke:</b><br>' + p.cc.map(function (e) { return MyCertif.esc(e); }).join('<br>'))
      : '';
    var cadangan = p.pakai_cadangan
      ? ('<br><span style="color:var(--gde-warning)"><i class="bi bi-info-circle"></i> '
        + 'Email pribadi belum diisi, memakai email perusahaan sebagai cadangan.</span>')
      : '';

    return MyCertif.dialogForm({
      judul: 'Kirim notifikasi pengingat',
      pesanHtml: '<b>' + MyCertif.esc(p.nama_sertifikasi) + '</b><br>'
        + MyCertif.esc(p.nama_pemegang || p.employee_code) + ' (' + MyCertif.esc(p.employee_code) + ')<br>'
        + 'Berlaku sampai ' + MyCertif.esc(MyCertif.tglIndo(p.expired_at))
        + (p.sisa_teks ? (' &#183; ' + MyCertif.esc(p.sisa_teks)) : '')
        + '<br>Tingkat pengingat: <b>' + MyCertif.esc(p.level_label) + '</b>'
        + '<div class="mycertif-kode mycertif-mt-3"><b>Kepada:</b><br>' + tujuan + cc + cadangan + '</div>',
      fields: [
        {
          name: 'pesan', label: 'Catatan tambahan (opsional)', tipe: 'textarea', rows: 3,
          placeholder: 'mis. Mohon unggah ulang sertifikat yang sudah diperpanjang sebelum 30 Juni.',
          hint: 'Ditampilkan di bagian atas email dan ikut terkirim ke penerima.'
        }
      ],
      labelOk: 'Kirim sekarang'
    });
  }).then(function (v) {
    if (!v) return null;
    return MyCertif.api('apiKirimNotifikasiFile', [docId, v.pesan]);
  }).then(function (r) {
    if (!r) return;
    if (!r.terkirim) {
      MyCertif.gagal(r.alasan || 'Notifikasi tidak terkirim.', 'Gagal mengirim');
      return;
    }
    MyCertif.sukses('Pengingat terkirim ke <b>' + MyCertif.esc((r.ke || []).join(', ')) + '</b>'
      + ((r.cc || []).length ? ('<br>Ditembuskan ke: ' + MyCertif.esc(r.cc.join(', '))) : ''),
    'Notifikasi terkirim');
    MyCertif.perbaruiBadgeMendesak();
  }).catch(function (e) { MyCertif.gagal(e.message); });
}

function bukaVerifikasi(docId) {
  MyCertif.api('apiGetFile', [docId]).then(function (d) {
    var f = d.file;
    var defs = (MyCertif.state.meta && MyCertif.state.meta.verifikasi_defs) || [];
    var opsi = defs.filter(function (v) { return v.key !== 'PENDING'; }).map(function (v) {
      return { value: v.key, label: v.label + ' — ' + (v.ket || '') };
    });

    return MyCertif.dialogForm({
      judul: 'Pemeriksaan data sertifikasi',
      pesanHtml: '<b>' + MyCertif.esc(f.nama_sertifikasi) + '</b><br>'
        + MyCertif.esc(f.nama_pemegang) + ' (' + MyCertif.esc(f.employee_id) + ')<br>'
        + 'Berlaku sampai ' + MyCertif.esc(MyCertif.tglIndo(f.expired_at)) + ' &#183; '
        + MyCertif.statusBadge(f.status, f.status_label, f.status_tone, f.status_icon)
        + ' &#183; status saat ini: ' + MyCertif.verifikasiBadge(f.status_verifikasi),
      fields: [
        {
          name: 'keputusan', label: 'Hasil pemeriksaan', tipe: 'select',
          nilai: 'VERIFIED', wajib: true, opsi: opsi
        },
        {
          name: 'catatan', label: 'Catatan untuk pemilik data', tipe: 'textarea',
          nilai: f.catatan || '', rows: 3,
          placeholder: 'mis. Lampiran tidak terbaca, mohon unggah ulang halaman 2.',
          hint: 'Wajib diisi bila memilih Perlu Revisi. Catatan ini dikirim ke pemilik data.'
        }
      ],
      labelOk: 'Simpan hasil pemeriksaan',
      validasi: function (v) {
        if (v.keputusan === 'REVISI' && !v.catatan) {
          return 'Catatan wajib diisi bila memilih Perlu Revisi, supaya pemilik data tahu apa yang harus diperbaiki.';
        }
        return '';
      }
    });
  }).then(function (v) {
    if (!v) return null;
    return MyCertif.api('apiVerifyFile', [docId, v.keputusan, v.catatan]);
  }).then(function (r) {
    if (!r) return;
    var pesan = (r.status_verifikasi === 'REVISI')
      ? 'Data ditandai Perlu Revisi. Pemiliknya sudah menerima notifikasi berisi catatan Anda.'
      : 'Data ditandai Terverifikasi. Pemiliknya sudah menerima notifikasi.';
    MyCertif.sukses(pesan, 'Berhasil');
    muatDs(false);
    MyCertif.perbaruiBadgeMendesak();
    var modal = MyCertif.el('dsModalDetail');
    if (modal && modal.classList.contains('is-open')) bukaDetail(r.doc_id);
  }).catch(function (e) { MyCertif.gagal(e.message); });
}

/* ---------------------------------------------------------------- hapus -- */
function hapusDs(f) {
  MyCertif.konfirmasi({
    judul: 'Hapus data sertifikasi',
    pesanHtml: 'Data <b>' + MyCertif.esc(f.nama_sertifikasi) + '</b> milik '
      + MyCertif.esc(f.nama_pemegang || f.employee_id) + ' akan dihapus permanen'
      + (f.file_id ? ' beserta lampirannya' : '') + '. Tindakan ini tidak dapat dibatalkan.',
    labelOk: 'Hapus permanen',
    bahaya: true
  }).then(function (ya) {
    if (!ya) return;
    return MyCertif.api('apiDeleteFile', [f.doc_id]).then(function () {
      MyCertif.sukses('Data sertifikasi dihapus.', 'Berhasil');
      MyCertif.modalTutup('dsModalDetail');
      muatDs(false);
    });
  }).catch(function (e) { MyCertif.gagal(e.message); });
}

/* ---------------------------------------------------------------- ekspor -- */
function jalankanEkspor(btn, jenis) {
  MyCertif.tombolSibuk(btn, true, 'Menyiapkan...');
  var filter = {
    q: DsState.filter.q, status: DsState.filter.status, divisi: DsState.filter.divisi,
    verifikasi: DsState.filter.verifikasi,
    dari: DsState.filter.dari, sampai: DsState.filter.sampai,
    judul: 'Rekap Sertifikasi & Masa Berlaku'
  };
  var panggil = (jenis === 'excel')
    ? MyCertif.api('apiExportExcel', [filter])
    : MyCertif.api('apiExportPdf', [filter]);

  panggil.then(function (r) {
    MyCertif.tombolSibuk(btn, false);
    MyCertif.sukses(r.name + ' berisi ' + MyCertif.angka(r.jumlah) + ' baris berhasil dibuat.', 'Ekspor selesai');
    window.open(r.url, '_blank');
  }).catch(function (e) {
    MyCertif.tombolSibuk(btn, false);
    MyCertif.gagal(e.message);
  });
}
