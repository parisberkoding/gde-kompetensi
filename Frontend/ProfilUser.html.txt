/* ==========================================================================
   KONTROLER PROFIL SAYA

var PuState = { data: null };

function initProfilUser(param) {
  'use strict';
  puRenderKerangka();
  puMuat();
}

/* --------------------------------------------------------------- kerangka -- */
function puRenderKerangka() {
  MyCertif.isiHtml('puRoot', ''
    + '<div class="gde-page-head">'
    + '  <div>'
    + '    <div class="gde-u-eyebrow">Akun</div>'
    + '    <h1>Profil Saya</h1>'
    + '    <p>Identitas, kontak notifikasi, dan ringkasan sertifikasi Anda. '
    + 'Pastikan nomor HP dan email selalu aktif agar pengingat masa berlaku tidak terlewat.</p>'
    + '  </div>'
    + '  <div class="gde-page-actions">'
    + '    <button class="gde-btn gde-btn--outline" type="button" id="puMuatUlang">'
    + '      <i class="bi bi-arrow-clockwise"></i> Segarkan</button>'
    + '  </div>'
    + '</div>'
    + '<div id="puIsi">' + MyCertif.pemuatHtml('Memuat profil...') + '</div>');

  var ulang = MyCertif.el('puMuatUlang');
  if (ulang) ulang.addEventListener('click', function () { puMuat(); });
}

/* ---------------------------------------------------------------- pemuatan -- */
function puMuat() {
  var isi = MyCertif.el('puIsi');
  if (!isi) return;
  isi.innerHTML = MyCertif.pemuatHtml('Memuat profil...');

  var kode = String(MyCertif.pengguna().employee_code || '');
  MyCertif.api('apiGetEmployeeProfile', [kode]).then(function (d) {
    PuState.data = d;
    puRenderIsi(d);
  }).catch(function (e) {
    isi.innerHTML = MyCertif.galatHtml(e.message);
  });
}

function puRenderIsi(d) {
  var user = (d && d.user) || MyCertif.pengguna();
  var ring = (d && d.ringkasan) || {};

  if (!user || !user.employee_code) {
    MyCertif.isiHtml('puIsi', MyCertif.kosongHtml('bi-person-x', 'Profil tidak ditemukan',
      'Data karyawan tidak dapat dimuat. Hubungi Human Capital untuk memastikan akun Anda terdaftar.'));
    return;
  }

  MyCertif.isiHtml('puIsi', '<div class="mycertif-2col">'
    + puKolomKiri(user, ring)
    + puKolomKanan(user)
    + '</div>');

  puPasangPerilaku(user);
  puMuatDivisi();
}

  var ada = MyCertif.num(ring.ada_lampiran, 0);
  var persen = MyCertif.num(ring.persen_lengkap, total ? Math.round((ada / total) * 100) : 0);

  return ''
    + '<div class="gde-card">'
    + '  <div class="mycertif-panel-head">'
    + '    <i class="bi bi-clipboard-data-fill" style="color:var(--gde-blue)"></i>'
    + '    <div><h3>Ringkasan sertifikasi</h3>'
    + '    <div class="mycertif-panel-head__sub">Data sertifikasi atas nama Anda</div></div>'
    + '  </div>'
    + '  <div class="gde-card__body">'
    + '    <div class="gde-grid gde-grid--2" style="gap:.75rem">'
    + puMini('Total sertifikasi', total, 'bi-award', 'blue')
    + puMini('Kadaluarsa', ring.kadaluarsa, 'bi-x-octagon', 'dead')
    + puMini('Mendesak', ring.mendesak, 'bi-exclamation-triangle', 'red')
    + puMini('Aman', ring.aman, 'bi-check-circle', 'green')
    + '    </div>'
    + '    <div class="mycertif-mt-4" style="border-top:1px dashed var(--gde-line);padding-top:1rem">'
    + '      <div class="gde-u-between" style="margin-bottom:6px">'
    + '        <span class="mycertif-inline-note">Kelengkapan lampiran</span>'
    + '        <b style="font-size:.85rem">' + MyCertif.angka(ada) + ' / ' + MyCertif.angka(total) + '</b>'
    + '      </div>'
    + '      <div class="gde-progress"><span class="gde-progress__bar" style="width:' + persen + '%"></span></div>'
    + '      <div class="gde-hint">' + MyCertif.angka(persen) + '% data sertifikasi sudah memiliki lampiran.</div>'
    + '    </div>'
    + '  </div>'
    + '</div>';
}

function puMini(label, nilai, ikon, tone) {
  return '<div class="mycertif-mini mycertif-mini--' + MyCertif.attr(tone) + '">'
    + '<div class="mycertif-mini__icon"><i class="bi ' + MyCertif.attr(ikon) + '"></i></div>'
    + '<div style="min-width:0">'
    + '<div class="mycertif-mini__label">' + MyCertif.esc(label) + '</div>'
    + '<div class="mycertif-mini__value">' + MyCertif.angka(nilai) + '</div>'
    + '</div></div>';
}

function puBaris(k, v) {
  var nilai = (v === null || v === undefined || String(v).trim() === '') ? '—' : String(v);
  return '<dt>' + MyCertif.esc(k) + '</dt><dd>' + MyCertif.esc(nilai) + '</dd>';
}

function puBarisHtml(k, html) {
  return '<dt>' + MyCertif.esc(k) + '</dt><dd>' + html + '</dd>';
}

function puStatusAkun(status) {
  var s = String(status || 'ACTIVE').toUpperCase();
  if (s === 'ACTIVE') {
    return '<span class="gde-status gde-status--green"><i class="bi bi-check-circle-fill"></i>Aktif</span>';
  }
  if (s === 'INACTIVE') {
    return '<span class="gde-status gde-status--gray"><i class="bi bi-dash-circle"></i>Nonaktif</span>';
  }
  return '<span class="gde-status gde-status--gray">' + MyCertif.esc(s) + '</span>';
}

/* ------------------------------------------------------------ perilaku UI -- */
function puPasangPerilaku(user) {
  var form = MyCertif.el('puForm');
  if (form) form.addEventListener('submit', function (e) { puSimpan(e, user); });

  var reset = MyCertif.el('puReset');
  if (reset) {
    reset.addEventListener('click', function () {
      puIsiForm(user);
      MyCertif.bersihkanInvalid(form);
      MyCertif.info('Perubahan yang belum disimpan dibatalkan.');
    });
  }

  var kePekerja = MyCertif.el('puKePekerja');
  if (kePekerja) {
    kePekerja.addEventListener('click', function () {
      MyCertif.state.paramPekerja = String(MyCertif.pengguna().employee_code || '');
      MyCertif.navigasi('pekerja');
    });
  }

  var expor = MyCertif.el('puExport');
  if (expor) expor.addEventListener('click', puEksporPdf);

  MyCertif.qsa('[data-pu-nav]').forEach(function (btn) {
    btn.addEventListener('click', function () {
      MyCertif.navigasi(btn.getAttribute('data-pu-nav'));
    });
  });
}

function puIsiForm(user) {
  var form = MyCertif.el('puForm');
  if (!form) return;
  MyCertif.isiForm(form, user);
}

function puMuatDivisi() {
  if (!MyCertif.isHc()) return;
  MyCertif.muatMeta().then(function (m) {
    var list = MyCertif.el('puDivisiList');
    if (!list) return;
    list.innerHTML = ((m && m.divisi_options) || []).map(function (d) {
      return '<option value="' + MyCertif.attr(d) + '"></option>';
    }).join('');
  }).catch(function () { });
}

/* --------------------------------------------------------------- validasi -- */
function puEmailValid(nilai) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(String(nilai || '').trim());
}

function puCekEmail(input, nilai) {
  if (!nilai) return true;                       // opsional
  if (puEmailValid(nilai)) return true;
  MyCertif.tandaiInvalid(input, 'Format email tidak valid.');
  return false;
}

function puSimpan(e, user) {
  e.preventDefault();
  var form = MyCertif.el('puForm');
  MyCertif.bersihkanInvalid(form);

  var v = MyCertif.nilaiForm(form);
  var data = {
    no_hp: String(v.no_hp || '').trim(),
    email_pribadi: String(v.email_pribadi || '').trim(),
    email_atasan: String(v.email_atasan || '').trim()
  };

  if (data.no_hp && data.no_hp.replace(/[^0-9]/g, '').length < 8) {
    MyCertif.tandaiInvalid(MyCertif.el('puNoHp'), 'Nomor HP minimal 8 digit angka.');
    return;
  }
  if (!puCekEmail(MyCertif.el('puEmailPribadi'), data.email_pribadi)) return;
  if (!puCekEmail(MyCertif.el('puEmailAtasan'), data.email_atasan)) return;

  if (MyCertif.isHc()) {
    data.nama_lengkap = String(v.nama_lengkap || '').trim();
    data.email_perusahaan = String(v.email_perusahaan || '').trim();
    data.divisi = String(v.divisi || '').trim();
    data.jabatan = String(v.jabatan || '').trim();
    data.lokasi_kerja = String(v.lokasi_kerja || '').trim();

    if (!data.nama_lengkap) {
      MyCertif.tandaiInvalid(MyCertif.el('puNama'), 'Nama lengkap wajib diisi.');
      return;
    }
    if (!puCekEmail(MyCertif.el('puEmailPerusahaan'), data.email_perusahaan)) return;
  }

  var tombol = MyCertif.el('puSubmit');
  MyCertif.tombolSibuk(tombol, true, 'Menyimpan...');

  MyCertif.api('apiUpdateProfilSaya', [data]).then(function (r) {
    MyCertif.tombolSibuk(tombol, false);
    var baru = (r && r.user) || MyCertif.pengguna();

    /* Selaraskan sesi + shell agar topbar/sidebar ikut berubah. */
    MyCertif.simpanSesi(MyCertif.state.token, baru);
    MyCertif.sukses('Profil Anda berhasil diperbarui.', 'Tersimpan');
    MyCertif.renderShell();
    MyCertif.muatView('profil');
  }).catch(function (err) {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.gagal(err.message);
  });
}

function puEksporPdf() {
  var tombol = MyCertif.el('puExport');
  MyCertif.tombolSibuk(tombol, true, 'Menyiapkan...');
  MyCertif.api('apiExportProfilPdf', [MyCertif.pengguna().employee_code]).then(function (r) {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.sukses('Berkas PDF dibuat: ' + ((r && r.name) || 'profil') + '.', 'Ekspor selesai');
    if (r && r.url) window.open(r.url, '_blank');
  }).catch(function (err) {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.gagal(err.message);
  });
}
