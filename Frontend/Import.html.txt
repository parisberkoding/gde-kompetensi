/* --------------------------------------------------------------- muat -- */
function muatDaftarSheet(paksa) {
  MyCertif.api('apiDaftarSheet', []).then(function (d) {
    ImpState.daftarSheet = d.sheets || [];
    var sel = MyCertif.el('impSheet');
    if (!sel) return;
    var disarankan = d.sheet_disarankan || 'IMPORT_DATA';
    var sistem = ['Config', 'Master_User', 'T_Sertifikasi',
      'Session_Token', 'T_Notifikasi', 'T_Email_Log', 'T_Audit_Trail'];
    var opsi = ImpState.daftarSheet.filter(function (n) { return sistem.indexOf(n) === -1; });
    if (opsi.indexOf(disarankan) === -1) opsi.unshift(disarankan);

    sel.innerHTML = opsi.map(function (n) {
      var ada = ImpState.daftarSheet.indexOf(n) !== -1;
      return '<option value="' + MyCertif.attr(n) + '">' + MyCertif.esc(n)
        + (ada ? '' : '  (belum ada — akan dibuat saat menempel data)') + '</option>';
    }).join('');
    sel.value = ImpState.sheet || disarankan;
    ImpState.sheet = sel.value;

    if (paksa) MyCertif.info('Daftar sheet dimuat ulang.', 'Segar');
  }).catch(function (e) { MyCertif.gagal(e.message); });
}

/* ----------------------------------------------------------- pratinjau -- */
function jalankanPratinjau() {
  var tombol = MyCertif.el('impPratinjau');
  var sheet = ImpState.sheet;
  if (!sheet) { MyCertif.gagal('Pilih sheet sumber lebih dulu.'); return; }

  MyCertif.tombolSibuk(tombol, true, 'Membaca...');
  MyCertif.isiHtml('impHasil', MyCertif.pemuatHtml('Membaca sheet ' + sheet + '...'));

  MyCertif.api('apiPratinjauImpor', [sheet]).then(function (p) {
    MyCertif.tombolSibuk(tombol, false);
    ImpState.pratinjau = p;
    renderPratinjau(p);
  }).catch(function (e) {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.el('impHasil').classList.remove('gde-u-hide');
    MyCertif.isiHtml('impHasil', MyCertif.galatHtml(e.message));
  });
}

/* ------------------------------------------------------------- jalankan -- */
function konfirmasiJalankanImpor() {
  var p = ImpState.pratinjau || {};
  var hanyaUser = !!(MyCertif.el('impHanyaUser') && MyCertif.el('impHanyaUser').checked);
  var perbaruiNama = !(MyCertif.el('impPerbaruiNama') && !MyCertif.el('impPerbaruiNama').checked);

  MyCertif.konfirmasi({
    judul: 'Jalankan impor data',
    pesanHtml: 'Akan dibuat/diperbarui:<br>'
      + '• <b>' + MyCertif.angka(p.pekerja_baru) + '</b> akun baru (username = nama depan, '
      + 'password = Employee Code)<br>'
      + '• <b>' + MyCertif.angka(p.pekerja_update) + '</b> akun sudah ada<br>'
      + (hanyaUser ? '' : ('• <b>' + MyCertif.angka(p.sertifikasi_baru) + '</b> data sertifikasi baru<br>'))
      + '<br>Proses berjalan di server dan tidak dapat dibatalkan. Lanjutkan?',
    labelOk: 'Ya, jalankan impor'
  }).then(function (ya) {
    if (!ya) return;
    var tombol = MyCertif.el('impJalankan');
    MyCertif.tombolSibuk(tombol, true, 'Mengimpor...');

    MyCertif.api('apiJalankanImpor', [ImpState.sheet, {
      hanya_user: hanyaUser,
      perbarui_nama: perbaruiNama
    }]).then(function (r) {
      MyCertif.tombolSibuk(tombol, false);
      renderHasilImpor(r);
      MyCertif.sukses('Impor selesai dalam ' + MyCertif.angka(r.durasi_ms) + ' ms.', 'Impor berhasil');
      MyCertif.perbaruiBadgeHeader();
    }).catch(function (e) {
      MyCertif.tombolSibuk(tombol, false);
      MyCertif.gagal(e.message);
    });
  });
}

function buatUserDariMasterFile() {
  var tombol = MyCertif.el('impBuatUser');
  MyCertif.konfirmasi({
    judul: 'Buat akun dari data sertifikasi',
    pesan: 'Sistem akan membaca sheet T_Sertifikasi dan membuatkan akun untuk setiap Employee Code '
      + 'yang belum terdaftar. Username diambil dari nama depan (dari sheet sumber bila ada), '
      + 'password = Employee Code penuh.',
    labelOk: 'Buat akun'
  }).then(function (ya) {
    if (!ya) return;
    MyCertif.tombolSibuk(tombol, true, 'Membuat...');
    MyCertif.api('apiBuatUserDariSertifikasi', [{
      sheet_nama: ImpState.sheet,
      perbarui_nama: true
    }]).then(function (r) {
      MyCertif.tombolSibuk(tombol, false);
      var pesan = 'Akun baru: ' + MyCertif.angka(r.dibuat)
        + ' · sudah ada: ' + MyCertif.angka(r.dilewati)
        + ' · nama dilengkapi: ' + MyCertif.angka(r.dilengkapi);
      if (r.errors && r.errors.length) {
        MyCertif.ingat(pesan + ' · ' + r.errors.length + ' error.', 'Selesai dengan catatan');
      } else {
        MyCertif.sukses(pesan, 'Pembuatan akun selesai');
      }
      renderHasilAkun(r, 'Hasil pembuatan akun dari T_Sertifikasi');
    }).catch(function (e) {
      MyCertif.tombolSibuk(tombol, false);
      MyCertif.gagal(e.message);
    });
  });
}

function rapikanAkun() {
  var tombol = MyCertif.el('impRapikan');
  MyCertif.konfirmasi({
    judul: 'Rapikan username & password',
    pesan: 'Password SELURUH akun akan disamakan dengan Employee Code masing-masing. '
      + 'Username yang bentrok akan dinomori ulang (mis. adiasa, adiasa2). '
      + 'Username yang sudah disesuaikan sendiri oleh karyawan tidak diubah.',
    labelOk: 'Rapikan sekarang'
  }).then(function (ya) {
    if (!ya) return;
    MyCertif.tombolSibuk(tombol, true, 'Merapikan...');
    MyCertif.api('apiRapikanUsernamePassword', [{}]).then(function (r) {
      MyCertif.tombolSibuk(tombol, false);
      MyCertif.sukses('Username diubah: ' + MyCertif.angka(r.username_diubah)
        + ' · password disamakan: ' + MyCertif.angka(r.password_diubah)
        + ' dari ' + MyCertif.angka(r.total) + ' akun.', 'Perapian selesai');
      renderHasilAkun(r, 'Hasil perapian username & password');
    }).catch(function (e) {
      MyCertif.tombolSibuk(tombol, false);
      MyCertif.gagal(e.message);
    });
  });
}

function renderHasilAkun(r, judul) {
  var html = '<div class="gde-card mycertif-mt-4">'
    + '<div class="mycertif-panel-head"><i class="bi bi-check-circle-fill" '
    + 'style="color:var(--gde-green-d)"></i><div><h3>' + MyCertif.esc(judul) + '</h3></div></div>'
    + '<div class="gde-card__body">'
    + '<div class="gde-grid gde-grid--3 mycertif-mb-4">'
    + miniImp('Dibuat / diubah', r.dibuat !== undefined ? r.dibuat : (r.username_diubah || 0),
      'bi-person-plus', 'green')
    + miniImp('Dilewati', r.dilewati !== undefined ? r.dilewati : (r.password_diubah || 0),
      'bi-person-check', 'blue')
    + miniImp('Nama dilengkapi', r.dilengkapi !== undefined ? r.dilengkapi : (r.total || 0),
      'bi-pencil-square', 'yellow')
    + '</div>';

  if (r.daftar && r.daftar.length) {
    html += '<div class="gde-u-eyebrow mycertif-mb-4">Akun baru</div>'
      + '<div class="mycertif-scroll-y gde-u-mono" style="font-size:.75rem">'
      + r.daftar.map(function (x) { return '<div>• ' + MyCertif.esc(x) + '</div>'; }).join('')
      + '</div>';
  }
  if (r.errors && r.errors.length) {
    html += '<div class="gde-alert gde-alert--danger mycertif-mt-3">'
      + '<i class="bi bi-x-circle gde-alert__icon"></i>'
      + '<div class="gde-alert__body"><strong>' + r.errors.length + ' error</strong>'
      + '<div class="mycertif-scroll-y mycertif-mt-3">'
      + r.errors.map(function (e) { return '<div>• ' + MyCertif.esc(e) + '</div>'; }).join('')
      + '</div></div></div>';
  }
  html += '<button class="gde-btn gde-btn--outline mycertif-mt-4" type="button" data-view="masteruser">'
    + '<i class="bi bi-people"></i> Buka Master User</button></div></div>';

  var lama = MyCertif.el('impHasilHasil');
  if (lama) lama.remove();
  MyCertif.el('impHasil').classList.remove('gde-u-hide');
  var div = document.createElement('div');
  div.id = 'impHasilHasil';
  div.innerHTML = html;
  MyCertif.el('impHasil').appendChild(div);
  MyCertif.perbaruiBadgeHeader();
}
</script>
