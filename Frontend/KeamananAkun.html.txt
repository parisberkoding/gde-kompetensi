/* ------------------------------------------------------------ perilaku UI -- */
function akPasangPerilaku() {
  var formUser = SIAGA.el('akFormUsername');
  if (formUser) formUser.addEventListener('submit', akSimpanUsername);

  var formPass = SIAGA.el('akFormPassword');
  if (formPass) formPass.addEventListener('submit', akSimpanPassword);

  var ulang = SIAGA.el('akMuatUlang');
  if (ulang) ulang.addEventListener('click', function () { akMuatSesi(); });

  /* Tombol mata untuk semua kolom password. */
  SIAGA.qsa('[data-ak-mata]').forEach(function (btn) {
    btn.addEventListener('click', function () {
      var input = SIAGA.el(btn.getAttribute('data-ak-mata'));
      if (!input) return;
      var lihat = input.type === 'password';
      input.type = lihat ? 'text' : 'password';
      btn.innerHTML = '<i class="bi ' + (lihat ? 'bi-eye-slash' : 'bi-eye') + '"></i>';
    });
  });

  /* Pratinjau slug username. */
  var baru = SIAGA.el('akUsernameBaru');
  if (baru) {
    baru.addEventListener('input', function () {
      var s = akSlug(baru.value);
      var label = SIAGA.el('akSlug');
      if (label) label.textContent = s || '—';
    });
  }

  /* Indikator kekuatan password. */
  var passBaru = SIAGA.el('akPasswordBaru');
  if (passBaru) passBaru.addEventListener('input', function () { akRenderKekuatan(passBaru.value); });
}

/* ------------------------------------------------------------- slug/username */
/** Cerminan slugUsername_() di server. */
function akSlug(s) {
  return String(s || '').toLowerCase()
    .replace(/[^a-z0-9._-]/g, '')
    .replace(/^[._-]+|[._-]+$/g, '')
    .substring(0, 40);
}

function akSimpanUsername(e) {
  e.preventDefault();
  var form = SIAGA.el('akFormUsername');
  SIAGA.bersihkanInvalid(form);

  var input = SIAGA.el('akUsernameBaru');
  var pass = SIAGA.el('akPasswordUsername');
  var slug = akSlug(input.value);

  if (!slug || slug.length < 3) {
    SIAGA.tandaiInvalid(input, 'Username minimal 3 karakter (huruf, angka, titik, garis bawah).');
    return;
  }
  if (!String(pass.value || '')) {
    SIAGA.tandaiInvalid(pass, 'Password saat ini wajib diisi.');
    return;
  }

  var tombol = SIAGA.el('akSubmitUsername');
  SIAGA.tombolSibuk(tombol, true, 'Menyimpan...');

  SIAGA.api('apiChangeUsername', [slug, pass.value]).then(function (r) {
    SIAGA.tombolSibuk(tombol, false);
    var user = (r && r.user) || SIAGA.pengguna();
    var lama = (r && r.username_lama) || '';
    var baru = (r && r.username_baru) || slug;

    SIAGA.simpanSesi(SIAGA.state.token, user);
    SIAGA.sukses('Username diubah' + (lama ? ' dari "' + lama + '"' : '') + ' menjadi "' + baru + '".', 'Berhasil');
    SIAGA.renderShell();
    SIAGA.muatView('keamanan');
  }).catch(function (err) {
    SIAGA.tombolSibuk(tombol, false);
    SIAGA.gagal(err.message);
  });
}

/* ------------------------------------------------------- kekuatan password -- */
function akKekuatan(pass) {
  var p = String(pass || '');
  if (!p) return { label: '—', kelas: 'gde-progress', persen: 0, warna: 'var(--gde-muted)' };

  var skor = 0;
  if (p.length >= 6) skor++;
  if (p.length >= 10) skor++;
  if (/[a-z]/.test(p) && /[A-Z]/.test(p)) skor++;
  if (/[0-9]/.test(p) && /[^A-Za-z0-9]/.test(p)) skor++;

  if (skor <= 1) return { label: 'Lemah', kelas: 'gde-progress gde-progress--red', persen: 33, warna: '#C4142A' };
  if (skor === 2) return { label: 'Sedang', kelas: 'gde-progress gde-progress--yellow', persen: 66, warna: '#6B5F00' };
  return { label: 'Kuat', kelas: 'gde-progress', persen: 100, warna: '#3F6B18' };
}

function akRenderKekuatan(pass) {
  var k = akKekuatan(pass);
  var wrap = SIAGA.el('akKuatWrap');
  var bar = SIAGA.el('akKuatBar');
  var label = SIAGA.el('akKuatLabel');
  if (wrap) wrap.className = k.kelas;
  if (bar) bar.style.width = k.persen + '%';
  if (label) {
    label.textContent = k.label;
    label.style.color = k.warna;
  }
}

function akSimpanPassword(e) {
  e.preventDefault();
  var form = SIAGA.el('akFormPassword');
  SIAGA.bersihkanInvalid(form);

  var v = SIAGA.nilaiForm(form);
  var lama = String(v.password_lama || '');
  var baru = String(v.password_baru || '');
  var konfirmasi = String(v.konfirmasi_password || '');

  var iLama = SIAGA.el('akPasswordLama');
  var iBaru = SIAGA.el('akPasswordBaru');
  var iKonfirmasi = SIAGA.el('akKonfirmasi');

  if (!lama) { SIAGA.tandaiInvalid(iLama, 'Password saat ini wajib diisi.'); return; }
  if (baru.length < 6) { SIAGA.tandaiInvalid(iBaru, 'Password baru minimal 6 karakter.'); return; }
  if (baru === lama) { SIAGA.tandaiInvalid(iBaru, 'Password baru tidak boleh sama dengan password lama.'); return; }
  if (baru !== konfirmasi) { SIAGA.tandaiInvalid(iKonfirmasi, 'Konfirmasi password tidak sama.'); return; }

  var tombol = SIAGA.el('akSubmitPassword');
  SIAGA.tombolSibuk(tombol, true, 'Menyimpan...');

  SIAGA.api('apiChangePassword', [lama, baru, konfirmasi]).then(function (r) {
    SIAGA.tombolSibuk(tombol, false);
    var dicabut = SIAGA.num(r && r.sesi_lain_dicabut, 0);

    SIAGA.sukses('Password diubah. ' + dicabut + ' sesi lain dicabut.', 'Berhasil');

    /* Tandai password awal sudah diganti agar peringatan tidak muncul lagi. */
    if (SIAGA.state.user) {
      SIAGA.state.user.must_change_password = false;
      SIAGA.simpanSesi(SIAGA.state.token, SIAGA.state.user);
    }

    SIAGA.resetForm(form);
    akRenderKekuatan('');
    akMuatSesi();
  }).catch(function (err) {
    SIAGA.tombolSibuk(tombol, false);
    SIAGA.gagal(err.message);
  });
}

/* ------------------------------------------------------------ keamanan sesi -- */
function akMuatSesi() {
  var box = SIAGA.el('akSesi');
  if (!box) return;
  box.innerHTML = SIAGA.pemuatHtml('Memeriksa sesi...');

  SIAGA.api('apiGetSession', []).then(function (d) {
    AkState.sesi = d;
    akRenderSesi(d);
  }).catch(function (e) {
    box.innerHTML = SIAGA.galatHtml(e.message);
  });
}

function akRenderSesi(d) {
  var box = SIAGA.el('akSesi');
  if (!box) return;

  if (!d || !d.user) {
    box.innerHTML = SIAGA.kosongHtml('bi-shield-slash', 'Informasi sesi tidak tersedia',
      'Data sesi tidak dapat dibaca. Coba segarkan halaman atau login kembali.');
    return;
  }

  var user = d.user || SIAGA.pengguna();
  var mustChange = !!d.must_change_password;

  /* Sinkronkan chip username dengan data sesi terbaru. */
  var chip = SIAGA.el('akUsernameKini');
  if (chip) chip.textContent = user.username || '—';

  var sisaHari = '';
  var valid = SIAGA.toDate(d && d.valid_until);
  if (valid) {
    var hari = Math.round((valid.getTime() - akHariIni().getTime()) / 86400000);
    sisaHari = hari >= 0 ? ('± ' + SIAGA.angka(hari) + ' hari lagi') : 'sudah lewat';
  }

  box.innerHTML = ''
    + '<div class="gde-card">'
    + '  <div class="siaga-panel-head">'
    + '    <i class="bi bi-shield-check" style="color:var(--gde-green-d)"></i>'
    + '    <div><h3>Keamanan sesi</h3>'
    + '    <div class="siaga-panel-head__sub">Informasi sesi yang sedang Anda pakai</div></div>'
    + '  </div>'
    + '  <div class="gde-card__body">'
    + (mustChange
      ? '    <div class="gde-alert gde-alert--warning" style="margin-bottom:1rem">'
        + '<i class="bi bi-exclamation-triangle-fill gde-alert__icon"></i>'
        + '<div class="gde-alert__body"><strong>Password awal masih dipakai</strong>'
        + 'Segera ubah password Anda memakai formulir Ubah Password di atas.</div></div>'
      : '')
    + '    <div class="gde-alert gde-alert--info" style="margin-bottom:1rem">'
    + '      <i class="bi bi-info-circle-fill gde-alert__icon"></i>'
    + '      <div class="gde-alert__body"><strong>Catatan keamanan</strong>'
    + 'Mengganti password akan mencabut seluruh sesi lain yang masih aktif di perangkat lain.</div>'
    + '    </div>'
    + '    <dl class="siaga-dl">'
    + akBaris('Username', user.username)
    + akBaris('Employee Code', user.employee_code)
    + akBarisHtml('Role', '<span class="gde-chip">'
      + SIAGA.esc(user.role_label || user.role || '—') + '</span>')
    + akBarisHtml('Wajib ganti password', mustChange
      ? '<span class="gde-status gde-status--yellow"><i class="bi bi-exclamation-triangle-fill"></i>Ya</span>'
      : '<span class="gde-status gde-status--green"><i class="bi bi-check-circle-fill"></i>Tidak</span>')
    + akBaris('Sesi berlaku sampai', valid ? (SIAGA.tglIndo(valid, true) + (sisaHari ? ' (' + sisaHari + ')' : '')) : '')
    + '    </dl>'
    + '  </div>'
    + '  <div class="gde-card__foot">'
    + '    <button class="gde-btn gde-btn--danger" type="button" id="akKeluar">'
    + '      <i class="bi bi-box-arrow-right"></i> Keluar</button>'
    + '    <span class="siaga-inline-note">Anda akan keluar dari sesi ini dan kembali ke halaman masuk.</span>'
    + '  </div>'
    + '</div>';

  var keluar = SIAGA.el('akKeluar');
  if (keluar) {
    keluar.addEventListener('click', function () {
      SIAGA.konfirmasi({
        judul: 'Keluar dari aplikasi',
        pesan: 'Anda akan keluar dari sesi ini. Sesi di perangkat lain tidak terpengaruh.',
        labelOk: 'Keluar',
        bahaya: true
      }).then(function (ya) { if (ya) SIAGA.logout(); });
    });
  }
}

function akBaris(k, v) {
  var nilai = (v === null || v === undefined || String(v).trim() === '') ? '—' : String(v);
  return '<dt>' + SIAGA.esc(k) + '</dt><dd>' + SIAGA.esc(nilai) + '</dd>';
}

function akBarisHtml(k, html) {
  return '<dt>' + SIAGA.esc(k) + '</dt><dd>' + html + '</dd>';
}

function akHariIni() {
  var d = new Date();
  return new Date(d.getFullYear(), d.getMonth(), d.getDate());
}
