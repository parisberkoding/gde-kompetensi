/* ==========================================================================
   KONTROLER HALAMAN LOGIN
   ========================================================================== */
function initLogin(pesanAwal) {
  'use strict';

  var form = SIAGA.el('siagaLoginForm');
  var inputUser = SIAGA.el('siagaLoginUsername');
  var inputPass = SIAGA.el('siagaLoginPassword');
  var tombol = SIAGA.el('siagaLoginSubmit');
  var toggle = SIAGA.el('siagaLoginTogglePass');
  var kotakAlert = SIAGA.el('siagaLoginAlert');
  var alertMsg = SIAGA.el('siagaLoginAlertMsg');

  var tahun = SIAGA.el('siagaLoginTahun');
  if (tahun) tahun.textContent = new Date().getFullYear();
  var versi = SIAGA.el('siagaLoginVersi');
  if (versi) versi.textContent = SIAGA.APP.version;

  function tampilkanGalat(pesan) {
    if (!kotakAlert) { SIAGA.gagal(pesan); return; }
    alertMsg.textContent = pesan;
    kotakAlert.classList.remove('gde-u-hide');
  }

  function sembunyikanGalat() {
    if (kotakAlert) kotakAlert.classList.add('gde-u-hide');
  }

  if (pesanAwal) tampilkanGalat(pesanAwal);
  setTimeout(function () { try { inputUser.focus(); } catch (e) { } }, 120);

  if (toggle) {
    toggle.addEventListener('click', function () {
      var lihat = inputPass.type === 'password';
      inputPass.type = lihat ? 'text' : 'password';
      toggle.innerHTML = '<i class="bi ' + (lihat ? 'bi-eye-slash' : 'bi-eye') + '"></i>';
      try { inputPass.focus(); } catch (e) { }
    });
  }

  if (form) {
    form.addEventListener('submit', function (e) {
      e.preventDefault();
      sembunyikanGalat();
      SIAGA.bersihkanInvalid(form);

      var username = String(inputUser.value || '').trim();
      var password = String(inputPass.value || '');

      if (!username) { SIAGA.tandaiInvalid(inputUser, 'Username wajib diisi.'); return; }
      if (!password) { SIAGA.tandaiInvalid(inputPass, 'Password wajib diisi.'); return; }

      SIAGA.tombolSibuk(tombol, true, 'Memeriksa...');
      SIAGA.login(username, password).then(function (data) {
        SIAGA.tombolSibuk(tombol, false);
        SIAGA.sukses('Selamat datang, ' + ((data.user && data.user.nama_lengkap) || username) + '.', 'Berhasil masuk');
        return SIAGA.muatMeta().then(function () {
          SIAGA.renderShell();
          SIAGA.mulaiRouter();          // mulai router + muat halaman awal
          SIAGA.perbaruiBadgeNotifikasi();
          SIAGA.perbaruiBadgeMendesak();
          if (data.user && data.user.must_change_password) {
            SIAGA.ingat('Password Anda masih password awal. Segera ganti di menu Username & Password.',
              'Perhatian keamanan');
          }
        });
      }).catch(function (err) {
        SIAGA.tombolSibuk(tombol, false);
        var msg = (err && err.message) ? err.message : 'Tidak dapat masuk. Coba lagi.';
        tampilkanGalat(msg);
        inputPass.value = '';
        try { inputPass.focus(); } catch (e2) { }
      });
    });
  }

  /* Enter di kolom username pindah ke password */
  if (inputUser) {
    inputUser.addEventListener('keydown', function (e) {
      if (e.key === 'Enter') { e.preventDefault(); try { inputPass.focus(); } catch (x) { } }
    });
  }
}
</script>
