/* ==========================================================================
   KONTROLER HALAMAN LOGIN
   ========================================================================== */
function initLogin(pesanAwal) {
  'use strict';

  var form = MyCertif.el('mycertifLoginForm');
  var inputKode = MyCertif.el('mycertifLoginKode');
  var wrapPass = MyCertif.el('mycertifLoginPassWrap');
  var inputPass = MyCertif.el('mycertifLoginPassword');
  var tombol = MyCertif.el('mycertifLoginSubmit');
  var toggle = MyCertif.el('mycertifLoginTogglePass');
  var kotakAlert = MyCertif.el('mycertifLoginAlert');
  var alertMsg = MyCertif.el('mycertifLoginAlertMsg');
  var kotakAuto = MyCertif.el('mycertifLoginAuto');
  var autoMsg = MyCertif.el('mycertifLoginAutoMsg');
  var tombolAuto = MyCertif.el('mycertifLoginCobaAuto');
  // Password selalu ditampilkan. Akun yang passwordnya belum diisi (jarang)
  // tetap bisa masuk — server mengabaikan password pada kasus itu.
  var passDiminta = true;
  var sedangProses = false;

  var tahun = MyCertif.el('mycertifLoginTahun');
  if (tahun) tahun.textContent = new Date().getFullYear();
  var versi = MyCertif.el('mycertifLoginVersi');
  if (versi) versi.textContent = MyCertif.APP.version;

  /* ---------------- pesan ---------------- */
  function tampilkanGalat(pesan) {
    if (!kotakAlert) { MyCertif.gagal(pesan); return; }
    alertMsg.textContent = pesan;
    kotakAlert.classList.remove('gde-u-hide');
  }
  function sembunyikanGalat() {
    if (kotakAlert) kotakAlert.classList.add('gde-u-hide');
  }
  function tampilkanInfoAuto(pesan) {
    if (!kotakAuto || !pesan) return;
    autoMsg.textContent = pesan;
    kotakAuto.classList.remove('gde-u-hide');
  }

  if (pesanAwal) tampilkanGalat(pesanAwal);
  if (MyCertif.state.autoLoginInfo) tampilkanInfoAuto(MyCertif.state.autoLoginInfo);

  setTimeout(function () { try { inputKode.focus(); } catch (e) { } }, 120);

  /* ---------------- tombol mata ---------------- */
  if (toggle) {
    toggle.addEventListener('click', function () {
      var lihat = inputPass.type === 'password';
      inputPass.type = lihat ? 'text' : 'password';
      toggle.innerHTML = '<i class="bi ' + (lihat ? 'bi-eye-slash' : 'bi-eye') + '"></i>';
      try { inputPass.focus(); } catch (e) { }
    });
  }

  /* ---------------- fokus & tanda ke kolom password ---------------- */
  function mintaPassword(pesan) {
    if (pesan) {
      var hint = MyCertif.el('mycertifLoginPassHint');
      if (hint) hint.textContent = pesan;
    }
    try { inputPass.focus(); } catch (e) { }
  }

  /* ---------------- proses masuk ---------------- */
  function kirim() {
    sembunyikanGalat();
    MyCertif.bersihkanInvalid(form);

    var kode = String(inputKode.value || '').trim();
    var pass = String(inputPass.value || '');

    if (!kode) { MyCertif.tandaiInvalid(inputKode, 'Username (nama depan) wajib diisi.'); return; }
    if (!pass) { MyCertif.tandaiInvalid(inputPass, 'Password wajib diisi (Employee Code Anda).'); return; }
    if (sedangProses) return;
    sedangProses = true;

    MyCertif.tombolSibuk(tombol, true, 'Memeriksa...');
    MyCertif.login(kode, pass).then(function (data) {
      // ---- akun tidak punya password (mis. dibuat manual) ----
      if (data && data.butuh_password) {
        sedangProses = false;
        MyCertif.tombolSibuk(tombol, false);
        MyCertif.ingat('Akun ' + (data.nama_lengkap || kode)
          + ' belum memiliki password. Minta Human Capital melengkapi, atau tekan Masuk sekali lagi.',
          'Password belum diisi');
        return null;
      }

      // ---- berhasil ----
      MyCertif.sukses('Selamat datang, ' + ((data.user && data.user.nama_lengkap) || kode) + '.',
        'Berhasil masuk');
      return MyCertif.masukKeAplikasi();
    }).catch(function (err) {
      sedangProses = false;
      MyCertif.tombolSibuk(tombol, false);
      var msg = (err && err.message) ? err.message : 'Tidak dapat masuk. Coba lagi.';
      tampilkanGalat(msg);
      if (/password/i.test(msg)) {
        inputPass.value = '';
        try { inputPass.focus(); } catch (e2) { }
      } else {
        try { inputKode.select(); } catch (e3) { }
      }
    });
  }

  if (form) {
    form.addEventListener('submit', function (e) {
      e.preventDefault();
      kirim();
    });
  }

  /* Enter di kolom username pindah ke password */
  if (inputKode) {
    inputKode.addEventListener('keydown', function (e) {
      if (e.key !== 'Enter') return;
      e.preventDefault();
      try { inputPass.focus(); } catch (x) { }
    });
  }

  /* ---------------- masuk otomatis ---------------- */
  if (tombolAuto) {
    tombolAuto.addEventListener('click', function () {
      sembunyikanGalat();
      MyCertif.tombolSibuk(tombolAuto, true, 'Memeriksa akun Google...');
      MyCertif.cobaLoginOtomatis().then(function (berhasil) {
        MyCertif.tombolSibuk(tombolAuto, false);
        if (!berhasil) {
          tampilkanInfoAuto(MyCertif.state.autoLoginInfo
            || 'Email akun Google Anda belum terdaftar pada Master User. Masukkan Employee Code secara manual.');
          return;
        }
        MyCertif.sukses('Berhasil masuk otomatis.', 'Masuk otomatis');
        MyCertif.masukKeAplikasi();
      }).catch(function (e) {
        MyCertif.tombolSibuk(tombolAuto, false);
        tampilkanGalat(e.message || 'Login otomatis gagal.');
      });
    });
  }
}
</script>
