/* ==========================================================================
   KONTROLER HALAMAN LOGIN
   ========================================================================== */
function initLogin(pesanAwal) {
  'use strict';

  var form = SIAGA.el('siagaLoginForm');
  var inputKode = SIAGA.el('siagaLoginKode');
  var wrapPass = SIAGA.el('siagaLoginPassWrap');
  var inputPass = SIAGA.el('siagaLoginPassword');
  var tombol = SIAGA.el('siagaLoginSubmit');
  var toggle = SIAGA.el('siagaLoginTogglePass');
  var kotakAlert = SIAGA.el('siagaLoginAlert');
  var alertMsg = SIAGA.el('siagaLoginAlertMsg');
  var kotakAuto = SIAGA.el('siagaLoginAuto');
  var autoMsg = SIAGA.el('siagaLoginAutoMsg');
  var tombolAuto = SIAGA.el('siagaLoginCobaAuto');
  var passDiminta = false;
  var sedangProses = false;

  var tahun = SIAGA.el('siagaLoginTahun');
  if (tahun) tahun.textContent = new Date().getFullYear();
  var versi = SIAGA.el('siagaLoginVersi');
  if (versi) versi.textContent = SIAGA.APP.version;

  /* ---------------- pesan ---------------- */
  function tampilkanGalat(pesan) {
    if (!kotakAlert) { SIAGA.gagal(pesan); return; }
    alertMsg.textContent = pesan;
    kotakAlert.classList.remove('gde-u-hide');
  }
  function sembunyikanGalat() {
    if (kotakAlert) kotakAlert.classList.add('gde-u-hide');
  }
  function tampilkanInfoAuto(pesan) {
    if (!kotakAuto || !pesan) return;
    autoMsg.textContent = pesan;
    kotakAuto.classList.remove('gde-u-hide');
  }

  if (pesanAwal) tampilkanGalat(pesanAwal);
  if (SIAGA.state.autoLoginInfo) tampilkanInfoAuto(SIAGA.state.autoLoginInfo);

  setTimeout(function () { try { inputKode.focus(); } catch (e) { } }, 120);

  /* ---------------- tombol mata ---------------- */
  if (toggle) {
    toggle.addEventListener('click', function () {
      var lihat = inputPass.type === 'password';
      inputPass.type = lihat ? 'text' : 'password';
      toggle.innerHTML = '<i class="bi ' + (lihat ? 'bi-eye-slash' : 'bi-eye') + '"></i>';
      try { inputPass.focus(); } catch (e) { }
    });
  }

  /* ---------------- tampilkan kolom password ---------------- */
  function mintaPassword(pesan) {
    passDiminta = true;
    wrapPass.classList.remove('gde-u-hide');
    if (pesan) {
      var hint = SIAGA.el('siagaLoginPassHint');
      if (hint) hint.textContent = pesan;
    }
    try { inputPass.focus(); } catch (e) { }
  }

  /* ---------------- proses masuk ---------------- */
  function kirim() {
    sembunyikanGalat();
    SIAGA.bersihkanInvalid(form);

    var kode = String(inputKode.value || '').trim();
    var pass = passDiminta ? String(inputPass.value || '') : '';

    if (!kode) { SIAGA.tandaiInvalid(inputKode, 'Employee Code wajib diisi.'); return; }
    if (passDiminta && !pass) { SIAGA.tandaiInvalid(inputPass, 'Password wajib diisi.'); return; }
    if (sedangProses) return;
    sedangProses = true;

    SIAGA.tombolSibuk(tombol, true, 'Memeriksa...');
    SIAGA.login(kode, pass).then(function (data) {
      // ---- server meminta password (langkah kedua) ----
      if (data && data.butuh_password) {
        sedangProses = false;
        SIAGA.tombolSibuk(tombol, false);
        mintaPassword((data.pesan || '') + ' — ' + (data.nama_lengkap || kode));
        return null;
      }

      // ---- berhasil ----
      SIAGA.sukses('Selamat datang, ' + ((data.user && data.user.nama_lengkap) || kode) + '.',
        'Berhasil masuk');
      return SIAGA.masukKeAplikasi();
    }).catch(function (err) {
      sedangProses = false;
      SIAGA.tombolSibuk(tombol, false);
      var msg = (err && err.message) ? err.message : 'Tidak dapat masuk. Coba lagi.';
      tampilkanGalat(msg);
      if (passDiminta) {
        inputPass.value = '';
        try { inputPass.focus(); } catch (e2) { }
      } else {
        try { inputKode.select(); } catch (e3) { }
      }
    });
  }

  if (form) {
    form.addEventListener('submit', function (e) {
      e.preventDefault();
      kirim();
    });
  }

  /* Enter di kolom kode */
  if (inputKode) {
    inputKode.addEventListener('keydown', function (e) {
      if (e.key !== 'Enter') return;
      e.preventDefault();
      if (passDiminta) { try { inputPass.focus(); } catch (x) { } }
      else kirim();
    });
  }

  /* ---------------- masuk otomatis ---------------- */
  if (tombolAuto) {
    tombolAuto.addEventListener('click', function () {
      sembunyikanGalat();
      SIAGA.tombolSibuk(tombolAuto, true, 'Memeriksa akun Google...');
      SIAGA.cobaLoginOtomatis().then(function (berhasil) {
        SIAGA.tombolSibuk(tombolAuto, false);
        if (!berhasil) {
          tampilkanInfoAuto(SIAGA.state.autoLoginInfo
            || 'Email akun Google Anda belum terdaftar pada Master User. Masukkan Employee Code secara manual.');
          return;
        }
        SIAGA.sukses('Berhasil masuk otomatis.', 'Masuk otomatis');
        SIAGA.masukKeAplikasi();
      }).catch(function (e) {
        SIAGA.tombolSibuk(tombolAuto, false);
        tampilkanGalat(e.message || 'Login otomatis gagal.');
      });
    });
  }
}
</script>
