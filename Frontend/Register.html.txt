/* ------------------------------------------------------------ pasang event -- */
function regPasangPerilaku() {
  var form = MyCertif.el('regForm');
  if (form) form.addEventListener('submit', regKirim);

  var reset = MyCertif.el('regReset');
  if (reset) {
    reset.addEventListener('click', function () {
      regReset();
      MyCertif.info('Formulir dibersihkan.');
    });
  }

  var batal = MyCertif.el('regBatalUbah');
  if (batal) {
    batal.addEventListener('click', function () {
      regReset();
      MyCertif.info('Mode ubah dibatalkan.');
    });
  }

  var jenis = MyCertif.el('regJenis');
  if (jenis) jenis.addEventListener('input', MyCertif.debounce(regJenisBerubah, 350));

  /* Perubahan dasar perhitungan -> expired_at dihitung ulang otomatis. */
  ['regStart', 'regEnd', 'regPerpanjang', 'regMasa'].forEach(function (id) {
    var f = MyCertif.el(id);
    if (!f) return;
    f.addEventListener('change', function () { RegState.manualExpired = false; regHitungExpired(); });
    f.addEventListener('input', function () { RegState.manualExpired = false; regHitungExpired(); });
  });

  /* Pengguna mengubah expired_at sendiri -> tandai manual. */
  var exp = MyCertif.el('regExpired');
  if (exp) {
    exp.addEventListener('change', function () { RegState.manualExpired = true; regRenderPreview(); });
    exp.addEventListener('input', function () { RegState.manualExpired = true; regRenderPreview(); });
  }

  /* ---- PEMEGANG SERTIFIKASI: pencarian-pilih (bukan dropdown panjang) ---- */
  var cari = MyCertif.el('regCariKaryawan');
  if (cari && MyCertif.isHc()) {
    RegState.cbKaryawan = MyCertif.combobox({
      input: cari,
      panel: MyCertif.el('regCbPanel'),
      placeholder: 'Ketik nama, Employee Code, atau divisi...',
      onPilih: function (item) {
        var emp = MyCertif.el('regEmployee');
        if (emp) emp.value = item.value;
        regRenderPreview();
        regMuatRiwayat();
      }
    });
  }

  var muat = MyCertif.el('regRiwayatMuatUlang');
  if (muat) muat.addEventListener('click', function () { regMuatRiwayat(); });

  var keRiwayat = MyCertif.el('regKeRiwayat');
  if (keRiwayat) {
    keRiwayat.addEventListener('click', function () {
      var t = MyCertif.el('regRiwayat');
      if (t) t.scrollIntoView({ behavior: 'smooth', block: 'start' });
    });
  }

  var keData = MyCertif.el('regKeData');
  if (keData) keData.addEventListener('click', function () { MyCertif.navigasi('data'); });

  regPasangDrop();

  var box = MyCertif.el('regRiwayat');
  if (box) {
    MyCertif.on(box, 'click', '[data-reg-ubah]', function (e, t) {
      regMuatUntukUbah(t.getAttribute('data-reg-ubah'), t);
    });
    MyCertif.on(box, 'click', '[data-reg-hapus]', function (e, t) {
      regHapus(t.getAttribute('data-reg-hapus'));
    });
  }
}

/* --------------------------------------------------------------- dropzone -- */
function regPasangDrop() {
  var drop = MyCertif.el('regDrop');
  var input = MyCertif.el('regFileInput');
  if (!drop || !input) return;

  drop.addEventListener('click', function () { input.click(); });
  input.addEventListener('click', function (e) { e.stopPropagation(); });
  input.addEventListener('change', function () {
    if (input.files && input.files.length) regPilihBerkas(input.files[0]);
  });

  ['dragenter', 'dragover'].forEach(function (ev) {
    drop.addEventListener(ev, function (e) {
      e.preventDefault();
      e.stopPropagation();
      drop.classList.add('is-over');
    });
  });
  ['dragleave', 'dragend'].forEach(function (ev) {
    drop.addEventListener(ev, function (e) {
      e.preventDefault();
      e.stopPropagation();
      drop.classList.remove('is-over');
    });
  });
  drop.addEventListener('drop', function (e) {
    e.preventDefault();
    e.stopPropagation();
    drop.classList.remove('is-over');
    var f = e.dataTransfer && e.dataTransfer.files && e.dataTransfer.files[0];
    if (f) regPilihBerkas(f);
  });

  var kartu = MyCertif.el('regFileCard');
  if (kartu) {
    kartu.addEventListener('click', function (e) {
      var t = e.target.closest('[data-reg-hapusberkas]');
      if (!t) return;
      RegState.berkas = null;
      input.value = '';
      regRenderFileCard();
      MyCertif.info('Berkas yang dipilih dibatalkan.');
    });
  }
}

function regPilihBerkas(file) {
  var salah = MyCertif.validasiBerkas(file);
  if (salah) {
    var input = MyCertif.el('regFileInput');
    if (input) input.value = '';
    MyCertif.gagal(salah, 'Berkas tidak dapat dipakai');
    return;
  }
  RegState.berkas = file;
  regRenderFileCard();
  MyCertif.sukses('Berkas "' + file.name + '" siap diunggah.', 'Berkas dipilih');
}

function regRenderFileCard() {
  var box = MyCertif.el('regFileCard');
  if (!box) return;
  var html = '';

  if (RegState.berkas) {
    html += '<div class="mycertif-filecard">'
      + '<i class="bi ' + MyCertif.attr(MyCertif.ikonBerkas(RegState.berkas.name)) + '"></i>'
      + '<div class="mycertif-filecard__meta">'
      + '<span class="mycertif-filecard__nama">' + MyCertif.esc(RegState.berkas.name) + '</span>'
      + '<span class="mycertif-filecard__sub">' + MyCertif.esc(MyCertif.ukuran(RegState.berkas.size))
      + ' &middot; siap diunggah</span>'
      + '</div>'
      + '<button class="gde-btn gde-btn--outline gde-btn--xs" type="button" data-reg-hapusberkas>'
      + '<i class="bi bi-trash"></i> Hapus</button>'
      + '</div>';
  } else if (RegState.lampiranLama) {
    var l = RegState.lampiranLama;
    html += '<div class="mycertif-filecard">'
      + '<i class="bi ' + MyCertif.attr(MyCertif.ikonBerkas(l.name)) + '"></i>'
      + '<div class="mycertif-filecard__meta">'
      + '<span class="mycertif-filecard__nama">' + MyCertif.esc(l.name) + '</span>'
      + '<span class="mycertif-filecard__sub">' + MyCertif.esc(MyCertif.ukuran(l.size))
      + ' &middot; lampiran tersimpan</span>'
      + '</div>'
      + (l.url ? '<a class="gde-btn gde-btn--outline gde-btn--xs" href="' + MyCertif.attr(l.url)
        + '" target="_blank" rel="noopener"><i class="bi bi-box-arrow-up-right"></i> Lihat</a>' : '')
      + '</div>'
      + '<div class="gde-hint">Pilih berkas baru bila ingin mengganti lampiran ini.</div>';
  }

  box.innerHTML = html;
}

/* ------------------------------------------------------ pemuatan data awal -- */
function regSiapkanAwal() {
  var masa = MyCertif.el('regMasa');
  if (masa && !masa.value) masa.value = 36;

  var hint = MyCertif.el('regEmployeeHint');
  if (hint) {
    hint.innerHTML = MyCertif.isHc()
      ? 'Pilih pemegang sertifikasi. Seluruh pekerja aktif tersedia untuk role HC/SA.'
      : 'Terisi otomatis dari akun Anda. Server juga memaksa pemegang = akun yang login.';
  }

  regRenderPreview();
}

function regMuatSaranJenis() {
  MyCertif.muatMeta().then(function (m) {
    RegState.jenis = (m && m.nama_sertifikasi_options) || [];
    regIsiPilihanJenis();
  }).catch(function (e) {
    RegState.jenis = ((MyCertif.state.meta || {}).nama_sertifikasi_options) || [];
    regIsiPilihanJenis();
    MyCertif.ingat(e.message, 'Saran nama sertifikasi tidak termuat');
  });
}

function regMuatKaryawan() {
  MyCertif.api('apiListEmployees', [{ aktif_only: true }]).then(function (d) {
    RegState.karyawan = (d && d.rows) || [];
    regIsiPilihanKaryawan('');
  }).catch(function (e) {
    MyCertif.gagal(e.message, 'Daftar pekerja gagal dimuat');
  });
}

function regIsiKaryawanSendiri() {
  var u = MyCertif.pengguna();
  var kode = String(u.employee_code || '');
  var tampil = kode + (u.nama_lengkap ? ' — ' + u.nama_lengkap : '');
  var emp = MyCertif.el('regEmployee');
  if (emp) emp.value = kode;                 // nilai yang dikirim ke server
  var cari = MyCertif.el('regCariKaryawan');
  if (cari) { cari.value = tampil; cari.disabled = true; }
}

/**
 * Isi daftar pencarian pemegang sertifikasi.
 * Pencarian & penyaringan dilakukan oleh komponen combobox, jadi di sini
 * cukup menyuplai seluruh pekerja (label + keterangan).
 */
function regIsiPilihanKaryawan(kata) {
  var cb = RegState.cbKaryawan;
  if (!cb) return;
  var items = (RegState.karyawan || []).map(function (u) {
    return {
      value: u.employee_code,
      label: u.nama_lengkap || u.employee_code,
      sub: u.employee_code
        + (u.divisi ? (' · ' + u.divisi) : '')
        + (u.jabatan ? (' · ' + u.jabatan) : '')
        + ' · ' + MyCertif.angka(u.jumlah_sertifikasi) + ' sertifikasi',
      badge: u.role_label || u.role
    };
  });
  cb.setItems(items);
  var terpilih = String((MyCertif.el('regEmployee') || {}).value || '');
  if (terpilih) cb.setNilai(terpilih);
}

function adaKaryawan(daftar, kode) {
  for (var i = 0; i < daftar.length; i++) {
    if (String(daftar[i].employee_code) === String(kode)) return true;
  }
  return false;
}

function regCariKaryawan(kode) {
  for (var i = 0; i < RegState.karyawan.length; i++) {
    if (String(RegState.karyawan[i].employee_code) === String(kode)) return RegState.karyawan[i];
  }
  return null;
}

/* ------------------------------------- isian & saran nama sertifikasi */
function regIsiPilihanJenis() {
  var list = MyCertif.el('regJenisList');
  if (!list) return;

  list.innerHTML = (RegState.jenis || []).map(function (j) {
    var nama = typeof j === 'string' ? j : String(j.nama || '');
    if (!nama) return '';
    return '<option value="' + MyCertif.attr(nama) + '"></option>';
  }).join('');
}

/** Cari saran berdasarkan nama yang diketik (untuk mengisi masa berlaku & penyelenggara). */
function regCariJenis(nama) {
  var cari = String(nama || '').trim().toLowerCase();
  if (!cari) return null;
  var daftar = RegState.jenis || [];
  for (var i = 0; i < daftar.length; i++) {
    var j = daftar[i];
    var n = typeof j === 'string' ? j : String(j.nama || '');
    if (n.trim().toLowerCase() === cari) return (typeof j === 'string') ? null : j;
  }
  return null;
}

/** Nama sertifikasi yang sedang diisi (selalu dari isian manual). */
function regNamaSertifikasi() {
  var jenis = MyCertif.el('regJenis');
  return jenis ? String(jenis.value || '').trim() : '';
}

function regJenisBerubah() {
  /* Isi otomatis masa berlaku & penyelenggara dari data yang pernah diinput,
     tanpa menimpa apa pun yang sudah diketik pengguna. */
  var j = regCariJenis(regNamaSertifikasi());
  if (j) {
    var masa = MyCertif.el('regMasa');
    if (masa && MyCertif.num(j.masa_berlaku_bulan, 0) > 0) masa.value = MyCertif.num(j.masa_berlaku_bulan, 36);

    var pen = MyCertif.el('regPenyelenggara');
    if (pen && !pen.value && j.penyelenggara) pen.value = j.penyelenggara;
  }

  RegState.manualExpired = false;
  regHitungExpired();
}

/* ------------------------------------------------ perhitungan expired_at ---- */
function regHariIni() {
  var d = new Date();
  return new Date(d.getFullYear(), d.getMonth(), d.getDate());
}

/** Samakan dengan addMonths_() di server: hari dijepit ke akhir bulan tujuan. */
function regTambahBulan(tgl, bulan) {
  var d = MyCertif.toDate(tgl);
  if (!d) return null;
  var n = MyCertif.num(bulan, 0);
  if (n <= 0) n = 36;
  var m = d.getMonth() + n;
  var y = d.getFullYear() + Math.floor(m / 12);
  m = ((m % 12) + 12) % 12;
  var akhirBulan = new Date(y, m + 1, 0).getDate();
  return new Date(y, m, Math.min(d.getDate(), akhirBulan));
}

function regMasaBulan() {
  var m = MyCertif.el('regMasa');
  var n = MyCertif.num(m ? m.value : 0, 0);
  return n > 0 ? n : 36;
}

function regDasarTanggal() {
  var p = MyCertif.el('regPerpanjang');
  if (p && p.value) return p.value;
  var e = MyCertif.el('regEnd');
  return e ? e.value : '';
}

function regHitungExpired() {
  var out = MyCertif.el('regExpired');
  if (!out) return;
  if (!RegState.manualExpired) {
    var exp = regTambahBulan(regDasarTanggal(), regMasaBulan());
    out.value = exp ? MyCertif.iso(exp) : '';
  }
  regRenderPreview();
}

function regAmbang() {
  var a = (MyCertif.state.meta && MyCertif.state.meta.ambang) || {};
  return {
    peringatan: MyCertif.num(a.peringatan, 270),
    perhatian: MyCertif.num(a.perhatian, 180),
    segera: MyCertif.num(a.segera, 90),
    mendesak: MyCertif.num(a.mendesak, 60)
  };
}

/** Cerminan statusDariSisaHari_() di server. */
function regStatusDariHari(hari) {
  var h = MyCertif.num(hari, 0);
  var a = regAmbang();
  if (h < 0) return 'KADALUARSA';
  if (h <= a.mendesak) return 'MENDEKATI_KADALUARSA';
  if (h <= a.segera) return 'SEGERA_PERPANJANG';
  if (h <= a.perhatian) return 'PERLU_PERHATIAN';
  if (h <= a.peringatan) return 'PERINGATAN';
  return 'MASIH_BERLAKU';
}

function regWarnaSisa(hari) {
  if (hari === null || hari === undefined || hari === '') return 'var(--gde-muted)';
  var h = MyCertif.num(hari);
  if (h < 0) return '#7F1D1D';
  if (h <= 60) return '#1B2430';
  if (h <= 90) return '#C4142A';
  if (h <= 180) return '#8A4B00';
  if (h <= 270) return '#6B5F00';
  return '#3F6B18';
}

function regRenderPreview() {
  var box = MyCertif.el('regPreview');
  if (!box) return;

  var input = MyCertif.el('regExpired');
  var nilai = input ? String(input.value || '') : '';
  if (!nilai) {
    box.innerHTML = '<span class="mycertif-inline-note">'
      + 'Expired_at dihitung otomatis = tanggal selesai pelaksanaan + masa berlaku.</span>';
    return;
  }

  var exp = MyCertif.toDate(nilai);
  if (!exp) {
    box.innerHTML = '<span class="mycertif-inline-note">Tanggal tidak dikenal.</span>';
    return;
  }

  var hari = Math.round((exp.getTime() - regHariIni().getTime()) / 86400000);
  var ket;
  if (RegState.manualExpired) ket = 'tanggal diisi manual';
  else if (MyCertif.el('regPerpanjang') && MyCertif.el('regPerpanjang').value) ket = 'dihitung dari tanggal perpanjangan + masa berlaku';
  else ket = 'dihitung dari tanggal selesai pelaksanaan + masa berlaku';

  box.innerHTML = ''
    + '<div class="gde-u-flex gde-u-items-center" style="gap:.5rem;flex-wrap:wrap">'
    + MyCertif.statusBadge(regStatusDariHari(hari))
    + '<b style="font-size:.8rem;color:' + regWarnaSisa(hari) + '">'
    + MyCertif.esc(MyCertif.sisaTeks(hari)) + '</b>'
    + '</div>'
    + '<div class="gde-hint">' + MyCertif.esc(MyCertif.tglIndo(exp, true)) + ' &middot; ' + MyCertif.esc(ket) + '</div>';
}

/* ------------------------------------------------------- simpan & validasi -- */
function regPayload() {
  var v = MyCertif.nilaiForm(MyCertif.el('regForm'));
  return {
    doc_id: RegState.docId || '',
    employee_id: String(v.employee_id || '').trim(),
    nama_sertifikasi: regNamaSertifikasi(),
    start_pelaksanaan: String(v.start_pelaksanaan || ''),
    end_pelaksanaan: String(v.end_pelaksanaan || ''),
    tanggal_sertifikasi: String(v.tanggal_sertifikasi || ''),
    tanggal_waktu_perpanjangan: String(v.tanggal_waktu_perpanjangan || ''),
    expired_at: String(v.expired_at || ''),
    masa_berlaku_bulan: MyCertif.num(v.masa_berlaku_bulan, 0),
    penyelenggara: String(v.penyelenggara || '').trim(),
    no_sertifikat: String(v.no_sertifikat || '').trim(),
    catatan: String(v.catatan || '').trim()
  };
}

function regValidasi() {
  var form = MyCertif.el('regForm');
  MyCertif.bersihkanInvalid(form);

  var emp = MyCertif.el('regEmployee');
  var jenis = MyCertif.el('regJenis');
  var start = MyCertif.el('regStart');
  var end = MyCertif.el('regEnd');
  var masa = MyCertif.el('regMasa');
  var exp = MyCertif.el('regExpired');

  if (!emp || !String(emp.value || '').trim()) {
    MyCertif.tandaiInvalid(emp, 'Pemegang sertifikasi wajib dipilih.');
    return false;
  }
  if (!regNamaSertifikasi()) {
    MyCertif.tandaiInvalid(jenis, 'Nama sertifikasi wajib diisi.');
    return false;
  }
  if (!start.value) { MyCertif.tandaiInvalid(start, 'Tanggal mulai pelaksanaan wajib diisi.'); return false; }
  if (!end.value) { MyCertif.tandaiInvalid(end, 'Tanggal selesai pelaksanaan wajib diisi.'); return false; }

  var dStart = MyCertif.toDate(start.value);
  var dEnd = MyCertif.toDate(end.value);
  if (!dStart) { MyCertif.tandaiInvalid(start, 'Tanggal mulai pelaksanaan tidak valid.'); return false; }
  if (!dEnd) { MyCertif.tandaiInvalid(end, 'Tanggal selesai pelaksanaan tidak valid.'); return false; }
  if (dEnd.getTime() < dStart.getTime()) {
    MyCertif.tandaiInvalid(end, 'Tanggal selesai tidak boleh lebih awal dari tanggal mulai.');
    return false;
  }
  if (!(MyCertif.num(masa.value, 0) > 0)) {
    MyCertif.tandaiInvalid(masa, 'Masa berlaku harus lebih dari 0 bulan.');
    return false;
  }
  if (!exp.value) {
    MyCertif.tandaiInvalid(exp, 'Expired_at belum bisa dihitung. Isi tanggal selesai pelaksanaan atau isi manual.');
    return false;
  }
  return true;
}

function regKirim(e) {
  e.preventDefault();
  if (!regValidasi()) return;

  var tombol = MyCertif.el('regSubmit');
  var payload = regPayload();
  var modeUbah = !!RegState.docId;
  MyCertif.tombolSibuk(tombol, true, modeUbah ? 'Menyimpan perubahan...' : 'Menyimpan...');

  var siap = RegState.berkas
    ? MyCertif.bacaBerkas(RegState.berkas).then(function (base64) {
      payload.file = {
        name: RegState.berkas.name,
        mimeType: RegState.berkas.type || '',
        base64: base64
      };
      return null;
    })
    : Promise.resolve(null);

  siap.then(function () {
    return MyCertif.api('apiSaveFile', [payload]);
  }).then(function (r) {
    MyCertif.tombolSibuk(tombol, false);
    var pesan;
    if (MyCertif.isHc()) pesan = 'Data sertifikasi tersimpan dan langsung terverifikasi.';
    else if (String((r && r.mode) || '') === 'insert') pesan = 'Data sertifikasi tersimpan dan menunggu verifikasi HC.';
    else pesan = 'Perubahan data sertifikasi tersimpan dan menunggu verifikasi HC.';
    MyCertif.sukses(pesan, 'Berhasil disimpan');
    regSetelahSimpan();
  }).catch(function (err) {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.gagal(err.message);
  });
}

function regSetelahSimpan() {
  var empId = '';
  if (MyCertif.isHc()) {
    var sel = MyCertif.el('regEmployee');
    empId = sel ? String(sel.value || '') : '';
  }
  regReset();
  if (empId) regPilihKaryawan(empId);
  regMuatRiwayat();
  var atas = MyCertif.el('regRoot');
  if (atas) atas.scrollIntoView({ behavior: 'smooth', block: 'start' });
}

/* -------------------------------------------------------------- mode ubah -- */
function regModeUbah(aktif) {
  var judul = MyCertif.el('regFormJudul');
  var sub = MyCertif.el('regFormSub');
  var tombol = MyCertif.el('regSubmit');
  var batal = MyCertif.el('regBatalUbah');

  if (judul) judul.textContent = aktif ? 'Ubah data sertifikasi' : 'Tambah data sertifikasi';
  if (sub) sub.textContent = aktif ? 'Perubahan menggantikan data yang tersimpan' : 'Isi data lalu simpan';
  if (tombol) {
    tombol.innerHTML = aktif
      ? '<i class="bi bi-save"></i> Simpan perubahan'
      : '<i class="bi bi-save"></i> Simpan data';
  }
  if (batal) batal.classList.toggle('gde-u-hide', !aktif);
}

function regIsiForm(file) {
  if (!file) return;
  var form = MyCertif.el('regForm');

  RegState.docId = String(file.doc_id || '');
  RegState.berkas = null;
  RegState.lampiranLama = file.file_id ? {
    name: file.file_name || 'Lampiran',
    size: MyCertif.num(file.ukuran_file, 0),
    url: file.view_url || file.link_file || ''
  } : null;
  RegState.manualExpired = true;

  var input = MyCertif.el('regFileInput');
  if (input) input.value = '';

  if (MyCertif.isHc()) {
    regIsiPilihanKaryawan();
    if (file.employee_id) regPilihKaryawan(file.employee_id);
  } else {
    regIsiKaryawanSendiri();
  }

  MyCertif.isiForm(form, file);
  regIsiJenis(file.nama_sertifikasi);
  regRenderFileCard();
  regRenderPreview();
  regModeUbah(true);

  try { form.scrollIntoView({ behavior: 'smooth', block: 'start' }); } catch (e) { }
}

/** Isikan nama sertifikasi ke isian manual (dipakai mode ubah). */
function regIsiJenis(nama) {
  var jenis = MyCertif.el('regJenis');
  if (!jenis) return;
  jenis.value = String(nama || '').trim();
  /* saran mungkin belum termuat saat mode ubah dibuka lebih dulu */
  if (!(RegState.jenis || []).length) regMuatSaranJenis();
}

function regMuatUntukUbah(docId, tombol) {
  if (!docId) return;
  MyCertif.tombolSibuk(tombol, true, 'Memuat...');
  MyCertif.api('apiGetFile', [docId]).then(function (d) {
    MyCertif.tombolSibuk(tombol, false);
    regIsiForm(d.file);
    MyCertif.info('Data "' + ((d.file && d.file.nama_sertifikasi) || docId) + '" dimuat ke formulir.', 'Mode ubah');
  }).catch(function (e) {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.gagal(e.message);
  });
}

function regReset() {
  var form = MyCertif.el('regForm');
  MyCertif.resetForm(form);

  RegState.docId = '';
  RegState.berkas = null;
  RegState.lampiranLama = null;
  RegState.manualExpired = false;

  var input = MyCertif.el('regFileInput');
  if (input) input.value = '';

  regIsiPilihanJenis();
  if (!MyCertif.isHc()) regIsiKaryawanSendiri();

  var masa = MyCertif.el('regMasa');
  if (masa) masa.value = 36;

  regRenderFileCard();
  regRenderPreview();
  regModeUbah(false);
}

function regPilihKaryawan(kode) {
  if (!kode) return;
  var kodeStr = String(kode);
  var emp = MyCertif.el('regEmployee');
  if (emp) emp.value = kodeStr;

  var cb = RegState.cbKaryawan;
  if (cb) {
    var ada = cb.setNilai(kodeStr);
    if (!ada) {
      // pekerja di luar daftar (mis. sudah nonaktif) — tetap tampilkan
      var u = regCariKaryawan(kodeStr);
      var cari = MyCertif.el('regCariKaryawan');
      if (cari) cari.value = kodeStr + (u && u.nama_lengkap ? ' — ' + u.nama_lengkap : '');
    }
  }
}

/* --------------------------------------------------------------- riwayat -- */
function regMuatRiwayat() {
  var box = MyCertif.el('regRiwayat');
  if (!box) return;
  box.innerHTML = MyCertif.pemuatHtml('Memuat riwayat...');

  var filter = { per_page: 50 };
  if (MyCertif.isHc()) {
    var sel = MyCertif.el('regEmployee');
    if (sel && sel.value) filter.employee_code = sel.value;
  }

  MyCertif.api('apiListFiles', [filter]).then(function (d) {
    regRenderRiwayat(d);
  }).catch(function (e) {
    box.innerHTML = MyCertif.galatHtml(e.message);
  });
}

function regRenderRiwayat(d) {
  var box = MyCertif.el('regRiwayat');
  if (!box) return;

  var rows = (d && d.rows) || [];
  var isHc = MyCertif.isHc();

  var sub = MyCertif.el('regRiwayatSub');
  if (sub) {
    if (!rows.length) sub.textContent = 'Belum ada data';
    else {
      sub.textContent = MyCertif.angka(d.total) + ' data'
        + (d.total > rows.length ? ' · menampilkan ' + MyCertif.angka(rows.length) + ' teratas' : '')
        + (isHc && MyCertif.el('regEmployee') && MyCertif.el('regEmployee').value
          ? ' · pemegang terpilih' : '');
    }
  }

  if (!rows.length) {
    box.innerHTML = MyCertif.kosongHtml('bi-inbox', 'Belum ada riwayat',
      isHc ? 'Belum ada data sertifikasi yang tercatat.' : 'Data sertifikasi yang Anda input akan tampil di sini.');
    return;
  }

  box.innerHTML = '<div class="gde-table-wrap"><table class="gde-table gde-table--compact">'
    + '<thead><tr>'
    + '<th>Sertifikasi</th>'
    + (isHc ? '<th>Pemegang</th>' : '')
    + '<th class="gde-ctr">Berlaku sampai</th>'
    + '<th class="gde-ctr">Sisa</th>'
    + '<th class="gde-ctr">Status</th>'
    + '<th class="gde-ctr">Verifikasi</th>'
    + '<th class="gde-ctr">Lampiran</th>'
    + '<th class="gde-u-text-right">Aksi</th>'
    + '</tr></thead><tbody>'
    + rows.map(function (r) { return regBarisRiwayat(r, isHc); }).join('')
    + '</tbody></table></div>';
}

function regBarisRiwayat(r, isHc) {
  var url = r.view_url || r.link_file || '';
  var lampiran = url
    ? '<a class="gde-btn gde-btn--outline gde-btn--xs" href="' + MyCertif.attr(url)
    + '" target="_blank" rel="noopener" title="' + MyCertif.attr(r.file_name || 'Lampiran') + '">'
    + '<i class="bi ' + MyCertif.attr(r.file_icon || MyCertif.ikonBerkas(r.file_name)) + '"></i></a>'
    : '<span style="color:var(--gde-muted-2)">—</span>';

  var subSert = [r.penyelenggara, r.no_sertifikat].filter(function (x) { return !!x; }).join(' · ')
    || r.doc_id || '';

  /* Bila HC meminta revisi, tampilkan catatannya langsung di baris ini supaya
     pemilik data tahu apa yang harus diperbaiki tanpa membuka notifikasi. */
  var catatan = '';
  if (String(r.status_verifikasi || '').toUpperCase() === 'REVISI' && r.catatan) {
    catatan = '<div class="mycertif-inline-note" style="color:var(--gde-red)">'
      + '<i class="bi bi-arrow-counterclockwise"></i> '
      + MyCertif.esc(r.catatan) + '</div>';
  }

  return '<tr>'
    + '<td class="mycertif-td-sert"><strong>' + MyCertif.esc(r.nama_sertifikasi || '-') + '</strong>'
    + '<span>' + MyCertif.esc(subSert) + '</span>'
    + catatan + '</td>'
    + (isHc ? '<td class="mycertif-td-nama"><strong>' + MyCertif.esc(r.nama_pemegang || r.employee_id) + '</strong>'
      + '<span>' + MyCertif.esc(r.employee_id + (r.divisi ? ' · ' + r.divisi : '')) + '</span></td>' : '')
    + '<td class="gde-ctr">' + MyCertif.esc(r.expired_at ? MyCertif.tglIndo(r.expired_at) : '—') + '</td>'
    + '<td class="gde-ctr mycertif-sisa" style="color:' + regWarnaSisa(r.sisa_hari) + '">'
    + MyCertif.esc(MyCertif.sisaTeks(r.sisa_hari)) + '</td>'
    + '<td class="gde-ctr">' + MyCertif.statusBadge(r.status, r.status_label, r.status_tone, r.status_icon) + '</td>'
    + '<td class="gde-ctr">' + regBadgeVerifikasi(r.status_verifikasi) + '</td>'
    + '<td class="gde-ctr">' + lampiran + '</td>'
    + '<td><div class="mycertif-aksi">'
    + '<button class="gde-btn gde-btn--outline gde-btn--xs" type="button" data-reg-ubah="'
    + MyCertif.attr(r.doc_id) + '"><i class="bi bi-pencil"></i> Ubah</button>'
    + (isHc ? '<button class="gde-btn gde-btn--danger gde-btn--xs" type="button" data-reg-hapus="'
      + MyCertif.attr(r.doc_id) + '"><i class="bi bi-trash"></i> Hapus</button>' : '')
    + '</div></td>'
    + '</tr>';
}

/** Badge verifikasi — definisi terpusat dari server (PENDING/VERIFIED/REVISI). */
function regBadgeVerifikasi(kode) {
  return MyCertif.verifikasiBadge(kode);
}

function regHapus(docId) {
  if (!docId) return;
  MyCertif.konfirmasi({
    judul: 'Hapus data sertifikasi',
    pesan: 'Data beserta lampirannya akan dihapus permanen dan tidak dapat dikembalikan. Lanjutkan?',
    labelOk: 'Hapus',
    bahaya: true
  }).then(function (ya) {
    if (!ya) return;
    MyCertif.api('apiDeleteFile', [docId]).then(function () {
      MyCertif.sukses('Data sertifikasi berhasil dihapus.', 'Terhapus');
      if (RegState.docId === String(docId)) regReset();
      regMuatRiwayat();
    }).catch(function (e) {
      MyCertif.gagal(e.message);
    });
  });
}

/* --------------------------------------------------------------- parameter -- */
function regBacaParam(param) {
  var s = String(param || '').trim();
  var out = { _mentah: s };
  if (!s || s.indexOf('=') === -1) return out;
  s.split('&').forEach(function (bagian) {
    var i = bagian.indexOf('=');
    if (i === -1) { out[decodeURIComponent(bagian)] = ''; return; }
    out[decodeURIComponent(bagian.substring(0, i))] = decodeURIComponent(bagian.substring(i + 1));
  });
  return out;
}

function regTerapkanParam(param) {
  var p = regBacaParam(param);
  var docId = p.doc_id || (/^DOC-/i.test(p._mentah) ? p._mentah : '');

  if (docId) { regMuatUntukUbah(docId); return; }

  if (p.employee_code && MyCertif.isHc()) {
    regPilihKaryawan(p.employee_code);
  }
}
