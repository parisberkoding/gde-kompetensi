var SetState = { config: {}, meta: null, ringkasan: null };

function initPengaturan(param) {
  'use strict';

  /* ---------- guard: hanya HC & SA ---------- */
  if (!SIAGA.isHc()) {
    SIAGA.el('setRoot').innerHTML = ''
      + '<div class="gde-card"><div class="gde-card__body">'
      + SIAGA.kosongHtml('bi-shield-lock', 'Akses ditolak',
        'Halaman pengaturan hanya dapat dibuka oleh Human Capital dan Super Admin. '
        + 'Hubungi Human Capital bila Anda memerlukan perubahan konfigurasi.')
      + '</div></div>';
    SIAGA.gagal('Anda tidak punya akses ke halaman Pengaturan.');
    return;
  }

  SetState = { config: {}, meta: null, ringkasan: null };
  renderSetKerangka();
  muatPengaturan();
}

/* -------------------------------------------------------------- kerangka -- */
function renderSetKerangka() {
  var isSa = SIAGA.isSa();

  SIAGA.el('setRoot').innerHTML = ''
    + '<div class="gde-page-head">'
    + '  <div>'
    + '    <div class="gde-u-eyebrow">Administrasi</div>'
    + '    <h1>Pengaturan Sistem</h1>'
    + '    <p>Atur ambang status masa berlaku, jadwal pengingat otomatis, tampilan aplikasi, '
    + 'serta kontak penerima notifikasi.</p>'
    + '  </div>'
    + '  <div class="gde-page-actions">'
    + '    <button class="gde-btn gde-btn--outline" type="button" id="setSegarkan">'
    + '      <i class="bi bi-arrow-clockwise"></i> Muat ulang</button>'
    + '    <button class="gde-btn gde-btn--outline-primary" type="button" id="setKeData">'
    + '      <i class="bi bi-card-checklist"></i> Hitung ulang status semua data</button>'
    + '  </div>'
    + '</div>'
    + '<div id="setIsi">' + SIAGA.pemuatHtml('Memuat konfigurasi...') + '</div>';

  var segar = SIAGA.el('setSegarkan');
  if (segar) segar.addEventListener('click', function () { muatPengaturan(true); });

  var keData = SIAGA.el('setKeData');
  if (keData) {
    keData.addEventListener('click', function () {
      SIAGA.konfirmasi({
        judul: 'Hitung ulang status semua data',
        pesan: 'Seluruh baris data sertifikasi akan dihitung ulang sisa hari, sisa bulan, dan '
          + 'statusnya memakai ambang yang tersimpan saat ini. Jalankan setelah mengubah ambang '
          + 'status atau setelah impor data massal.',
        labelOk: 'Hitung ulang sekarang'
      }).then(function (ya) {
        if (!ya) return;
        SIAGA.tombolSibuk(keData, true, 'Menghitung...');
        return SIAGA.api('apiRecalcStatus', []).then(function (r) {
          SIAGA.tombolSibuk(keData, false);
          SIAGA.sukses(SIAGA.angka(r.total) + ' baris diperiksa, ' + SIAGA.angka(r.diubah)
            + ' diperbarui. Master jenis sertifikasi: ' + SIAGA.angka(r.jenis_sertifikasi_diperbarui)
            + ' baris.', 'Hitung ulang selesai');
          muatPengaturan(true);
        }).catch(function (e) {
          SIAGA.tombolSibuk(keData, false);
          SIAGA.gagal(e.message);
        });
      });
    });
  }
}

/* ------------------------------------------------------------- pemuatan -- */
function muatPengaturan(paksa) {
  var isi = SIAGA.el('setIsi');
  if (!isi) return;
  if (paksa) isi.innerHTML = SIAGA.pemuatHtml('Memuat konfigurasi...');

  var tugas = [
    SIAGA.api('apiGetConfig', []),
    SIAGA.api('apiGetMeta', []),
    SIAGA.api('apiRingkasanReminder', []).catch(function () { return null; })
  ];

  Promise.all(tugas).then(function (hasil) {
    SetState.config = (hasil[0] && hasil[0].config) || {};
    SetState.meta = hasil[1] || null;
    SetState.ringkasan = hasil[2] || null;
    renderSetIsi();
  }).catch(function (e) {
    isi.innerHTML = SIAGA.galatHtml(e.message);
  });
}

/* --------------------------------------------------------------- render -- */
function renderSetIsi() {
  var isi = SIAGA.el('setIsi');
  if (!isi) return;
  var c = SetState.config;
  var meta = SetState.meta || {};

  isi.innerHTML = ''
    + kartuAmbang(c)
    + kartuReminder(c)
    + kartuTampilan(c, meta)
    + kartuKontak(c);

  pasangPerilakuPengaturan();
}

/* ------------------------------------------------------ potongan field -- */
function angkaField(id, nama, label, hint, nilai) {
  return '          <div>'
    + '            <label class="gde-label" for="' + id + '">' + SIAGA.esc(label) + '</label>'
    + '            <input class="gde-input" type="text" inputmode="numeric" id="' + id + '" name="' + nama + '"'
    + '                   data-set-angka="1" value="' + SIAGA.attr(formatAngka(nilai)) + '">'
    + '            <div class="gde-hint">' + SIAGA.esc(hint) + '</div>'
    + '          </div>';
}

function textField(id, nama, label, nilai, placeholder, hint) {
  return '          <div>'
    + '            <label class="gde-label" for="' + id + '">' + SIAGA.esc(label) + '</label>'
    + '            <input class="gde-input" type="text" id="' + id + '" name="' + nama + '"'
    + '                   value="' + SIAGA.attr(nilai) + '" placeholder="' + SIAGA.attr(placeholder || '') + '">'
    + '            <div class="gde-hint">' + SIAGA.esc(hint) + '</div>'
    + '          </div>';
}

function selectField(id, nama, label, opsi, nilai, hint) {
  var v = String(nilai === undefined || nilai === null ? '' : nilai);
  return '          <div>'
    + '            <label class="gde-label" for="' + id + '">' + SIAGA.esc(label) + '</label>'
    + '            <select class="gde-select" id="' + id + '" name="' + nama + '">'
    + opsi.map(function (o) {
      return '<option value="' + SIAGA.attr(o.value) + '"' + (String(o.value) === v ? ' selected' : '') + '>'
        + SIAGA.esc(o.label) + '</option>';
    }).join('')
    + '            </select>'
    + '            <div class="gde-hint">' + SIAGA.esc(hint) + '</div>'
    + '          </div>';
}

function saklar(id, nama, nilai, hint) {
  var aktif = (nilai === true || String(nilai).toUpperCase() === 'TRUE');
  return '            <label class="gde-switch" for="' + id + '">'
    + '<input type="checkbox" id="' + id + '" name="' + nama + '"' + (aktif ? ' checked' : '') + '>'
    + '<span class="gde-switch__track"></span><span>Ya</span></label>'
    + '<div class="gde-hint">' + hint + '</div>';
}

var HARI_PENGATURAN = { 0: 'Setiap hari', 1: 'Senin', 2: 'Selasa', 3: 'Rabu', 4: 'Kamis', 5: 'Jumat', 6: 'Sabtu', 7: 'Minggu' };

function opsiHari() {
  var o = [];
  for (var i = 0; i <= 7; i++) o.push({ value: String(i), label: HARI_PENGATURAN[i] });
  return o;
}

/* --------------------------------------------------------- perilaku UI -- */
function pasangPerilakuPengaturan() {
  /* tombol simpan per kartu */
  SIAGA.qsa('[data-set-simpan]').forEach(function (b) {
    b.addEventListener('click', function () {
      simpanKartu(b.getAttribute('data-set-simpan'), b);
    });
  });

  /* normalisasi isian angka + pratinjau status */
  SIAGA.qsa('[data-set-angka]').forEach(function (f) {
    f.addEventListener('input', function () {
      var pos = f.selectionStart;
      var panjangLama = f.value.length;
      f.value = formatAngka(f.value);
      var selisih = f.value.length - panjangLama;
      try { f.setSelectionRange(Math.max(pos + selisih, 0), Math.max(pos + selisih, 0)); } catch (e) { }
      perbaruiPratinjauStatus();
    });
  });
  ['setCobaHari', 'setMendesak', 'setSegera', 'setPerhatian', 'setPeringatan'].forEach(function (id) {
    var f = SIAGA.el(id);
    if (f) f.addEventListener('input', perbaruiPratinjauStatus);
    if (f) f.addEventListener('change', perbaruiPratinjauStatus);
  });
  perbaruiPratinjauStatus();

  /* tombol muat ulang di kartu C */
  var muat = SIAGA.el('setMuatUlang2');
  if (muat) {
    muat.addEventListener('click', function () {
      muatPengaturan(true);
      SIAGA.info('Pengaturan dimuat ulang dari server.');
    });
  }

  /* tombol kirim reminder sekarang di kartu B */
  var kirim = SIAGA.el('setKirimReminder');
  if (kirim) kirim.addEventListener('click', kirimReminderPengaturan);

  renderPenerimaAktual();
}

/* ------------------------------------------------------------- pratinjau -- */
function formatAngka(v) {
  var s = String(v === undefined || v === null ? '' : v).replace(/[^\d]/g, '');
  if (!s) return '';
  return s.replace(/\B(?=(\d{3})+(?!\d))/g, '.');
}

function angkaDari(v) {
  return SIAGA.num(String(v === undefined || v === null ? '' : v).replace(/\./g, ''), 0);
}

/**
 * Preview status: hitung status dari sisa hari percobaan memakai ambang
 * yang sedang diketik, tanpa memanggil server.
 */
function perbaruiPratinjauStatus() {
  var wadah = SIAGA.el('setPratinjauBadge');
  if (!wadah) return;
  var sisa = angkaDari(SIAGA.el('setCobaHari') ? SIAGA.el('setCobaHari').value : 365);
  if ((SIAGA.el('setCobaHari') && String(SIAGA.el('setCobaHari').value).trim() === '')) sisa = 365;

  var amb = {
    mendesak: angkaDari(SIAGA.el('setMendesak') ? SIAGA.el('setMendesak').value : 60),
    segera: angkaDari(SIAGA.el('setSegera') ? SIAGA.el('setSegera').value : 90),
    perhatian: angkaDari(SIAGA.el('setPerhatian') ? SIAGA.el('setPerhatian').value : 180),
    peringatan: angkaDari(SIAGA.el('setPeringatan') ? SIAGA.el('setPeringatan').value : 270)
  };

  var label, tone;
  if (sisa < 0) { label = 'Kadaluarsa'; tone = 'dead'; }
  else if (sisa <= amb.mendesak) { label = 'Mendekati Kadaluarsa'; tone = 'black'; }
  else if (sisa <= amb.segera) { label = 'Segera Perpanjang'; tone = 'red'; }
  else if (sisa <= amb.perhatian) { label = 'Perlu Perhatian'; tone = 'orange'; }
  else if (sisa <= amb.peringatan) { label = 'Peringatan'; tone = 'yellow'; }
  else { label = 'Masih Berlaku'; tone = 'green'; }

  wadah.innerHTML = '<span class="gde-status gde-status--' + tone + '">'
    + '<i class="bi bi-circle-fill" style="font-size:.5rem"></i>' + SIAGA.esc(label) + '</span>';
}

/* ---------------------------------------------------------------- simpan -- */
function simpanKartu(formId, tombol) {
  var form = SIAGA.el(formId);
  if (!form) return;

  var isi = kumpulkanKartu(formId, form);
  if (!isi) return;

  SIAGA.tombolSibuk(tombol, true, 'Menyimpan...');
  SIAGA.api('apiSetConfig', [isi.obj]).then(function (r) {
    SIAGA.tombolSibuk(tombol, false);
    var diubah = (r && r.diubah) || [];
    var dilewati = (r && r.dilewati) || [];

    /* perbarui salinan konfigurasi lokal */
    Object.keys(isi.obj).forEach(function (k) { SetState.config[k] = isi.obj[k]; });
    if (formId === 'setFormReminder') renderPenerimaAktual();

    if (!diubah.length && dilewati.length) {
      SIAGA.ingat('Tidak ada nilai yang berubah. Dilewati: ' + dilewati.join(', ') + '.', 'Tidak ada perubahan');
    } else {
      SIAGA.sukses('Diubah: ' + (diubah.length ? diubah.join(', ') : '-')
        + (dilewati.length ? ' · Dilewati: ' + dilewati.join(', ') : ''), 'Pengaturan tersimpan');
    }
    if (dilewati.length) {
      SIAGA.ingat('Field dilewati (tidak berubah atau bukan kewenangan Anda): ' + dilewati.join(', ') + '.');
    }
  }).catch(function (e) {
    SIAGA.tombolSibuk(tombol, false);
    SIAGA.gagal(e.message);
  });
}

/** Kumpulkan hanya field milik satu kartu, semua nilai sebagai string. */
function kumpulkanKartu(formId, form) {
  var out = [];

  if (formId === 'setFormAmbang') {
    var ambang = {
      status_mendesak_hari: angkaDari(SIAGA.el('setMendesak').value),
      status_segera_hari: angkaDari(SIAGA.el('setSegera').value),
      status_perhatian_hari: angkaDari(SIAGA.el('setPerhatian').value),
      status_peringatan_hari: angkaDari(SIAGA.el('setPeringatan').value),
      status_merah_hari: angkaDari(SIAGA.el('setMerah').value),
      masa_berlaku_default_bulan: angkaDari(SIAGA.el('setMasaBerlaku').value)
    };
    if (!(ambang.status_mendesak_hari <= ambang.status_segera_hari
      && ambang.status_segera_hari <= ambang.status_perhatian_hari
      && ambang.status_perhatian_hari <= ambang.status_peringatan_hari)) {
      SIAGA.gagal('Nilai ambang harus berurutan: mendesak ≤ segera ≤ perhatian ≤ peringatan.');
      return null;
    }
    if (ambang.masa_berlaku_default_bulan < 1) {
      SIAGA.gagal('Masa berlaku default minimal 1 bulan.');
      return null;
    }
    Object.keys(ambang).forEach(function (k) { out.push([k, String(ambang[k])]); });
    out.push(['masa_berlaku_5_tahun_kata_kunci', String(SIAGA.el('setKataKunci').value || '').trim()]);
    return { obj: keObjek(out) };
  }

  if (formId === 'setFormReminder') {
    out.push(['reminder_enabled', saklarNilai('setReminderAktif')]);
    out.push(['reminder_hari_kirim', String(SIAGA.el('setHariKirim').value || '0')]);
    out.push(['reminder_level_bulan', String(SIAGA.el('setLevelBulan').value || '').trim()]);
    out.push(['reminder_ke_karyawan', saklarNilai('setKeKaryawan')]);
    out.push(['reminder_ke_hc', saklarNilai('setKeHc')]);
    out.push(['reminder_ke_atasan', saklarNilai('setKeAtasan')]);
    out.push(['reminder_extra_emails', String(SIAGA.el('setExtraEmails').value || '').trim()]);
    out.push(['reminder_max_email_per_jalan', String(angkaDari(SIAGA.el('setMaxEmail').value))]);
    out.push(['reminder_ulang_kadaluarsa_hari', String(angkaDari(SIAGA.el('setUlangKadaluarsa').value))]);
    out.push(['email_pengirim', String(SIAGA.el('setEmailPengirim').value || '').trim()]);
    out.push(['email_subject_prefix', String(SIAGA.el('setSubjectPrefix').value || '').trim()]);
    out.push(['email_footer', String(SIAGA.el('setEmailFooter').value || '')]);
    return { obj: keObjek(out) };
  }

  if (formId === 'setFormTampilan') {
    out.push(['dashboard_prioritas_limit', String(angkaDari(SIAGA.el('setPrioritasLimit').value))]);
    out.push(['tabel_baris_per_halaman', String(angkaDari(SIAGA.el('setBarisHalaman').value))]);
    out.push(['export_max_baris', String(angkaDari(SIAGA.el('setExportMax').value))]);
    out.push(['unit_default', String(SIAGA.el('setUnitDefault').value || '').trim()]);
    out.push(['divisi_options', String(SIAGA.el('setDivisiOptions').value || '').trim()]);
    return { obj: keObjek(out) };
  }

  if (formId === 'setFormKontak') {
    var hc = String(SIAGA.el('setHcEmail').value || '').trim();
    if (hc && !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(hc)) {
      SIAGA.gagal('Format email Human Capital tidak valid.');
      return null;
    }
    out.push(['hc_email', hc]);
    if (SIAGA.isSa()) {
      var sa = String(SIAGA.el('setSaEmail').value || '').trim();
      if (sa && !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(sa)) {
        SIAGA.gagal('Format email Super Admin tidak valid.');
        return null;
      }
      out.push(['superadmin_email', sa]);
    }
    return { obj: keObjek(out) };
  }

  return { obj: SIAGA.nilaiForm(form) };
}

function keObjek(pasangan) {
  var o = {};
  pasangan.forEach(function (p) { o[p[0]] = p[1]; });
  return o;
}

function saklarNilai(id) {
  var f = SIAGA.el(id);
  return (f && f.checked) ? 'TRUE' : 'FALSE';
}

/* ------------------------------------------------- penerima & peringatan -- */
function renderPenerimaAktual() {
  var wadah = SIAGA.el('setPenerimaAktual');
  if (!wadah) return;

  var aktif = saklarNilai('setReminderAktif') === 'TRUE';
  var daftar = (SetState.ringkasan && SetState.ringkasan.email_hc) || [];
  var extra = String(SetState.config.reminder_extra_emails || '').split(/[,;\n]/)
    .map(function (x) { return String(x).trim(); })
    .filter(function (x) { return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(x); });

  var semua = daftar.concat(extra.filter(function (e) { return daftar.indexOf(e) === -1; }));

  var chip = semua.length
    ? semua.map(function (e) {
      return '<span class="gde-chip"><i class="bi bi-envelope"></i> ' + SIAGA.esc(e) + '</span>';
    }).join(' ')
    : '<span class="siaga-inline-note">Belum ada alamat penerima.</span>';

  wadah.innerHTML = ''
    + '<div class="siaga-inline-note" style="margin-bottom:.4rem">Penerima aktual (HC/SA + email tambahan tersimpan)</div>'
    + '<div class="gde-u-flex gde-u-flex-wrap siaga-gap-2">' + chip + '</div>'
    + (aktif && !semua.length ? '    <div class="gde-alert gde-alert--warning siaga-mt-3">'
      + '<i class="bi bi-exclamation-triangle-fill gde-alert__icon"></i>'
      + '<div class="gde-alert__body"><strong>Belum ada penerima email</strong>'
      + 'Pengingat otomatis aktif tetapi daftar penerima HC/SA kosong. Isi email pengguna HC/SA '
      + 'atau tambahkan alamat pada kolom Email tambahan.</div></div>' : '')
    + '    <div class="siaga-inline-note siaga-mt-3">Penyimpanan berkas &amp; basis data: '
    + 'Spreadsheet <b>' + (SetState.config.spreadsheet_id ? 'terkonfigurasi' : 'terkonfigurasi (bawaan)') + '</b> · '
    + 'Folder Drive <b>' + (SetState.config.drive_folder_id ? 'terkonfigurasi' : 'belum diatur') + '</b> · '
    + 'Folder ekspor <b>' + (SetState.config.drive_folder_export_id ? 'terkonfigurasi' : 'belum diatur') + '</b>. '
    + 'Nilai teknis sengaja tidak ditampilkan penuh.</div>';
}

/* ----------------------------------------------------------- kirim email -- */
function kirimReminderPengaturan() {
  SIAGA.konfirmasi({
    judul: 'Kirim reminder sekarang',
    pesan: 'Sistem akan memeriksa seluruh sertifikasi dan mengirim email pengingat ke karyawan, '
      + 'Human Capital, dan atasan sesuai tingkat status yang aktif. '
      + 'Email yang sudah pernah dikirim untuk tingkat yang sama tidak dikirim ulang.',
    labelOk: 'Kirim sekarang'
  }).then(function (ya) {
    if (!ya) return;
    var tombol = SIAGA.el('setKirimReminder');
    SIAGA.tombolSibuk(tombol, true, 'Mengirim...');
    SIAGA.api('apiRunReminderNow', []).then(function (r) {
      SIAGA.tombolSibuk(tombol, false);
      tampilkanHasilReminder(r || {});
      SIAGA.sukses('Target ' + SIAGA.angka(r.target) + ' · terkirim ' + SIAGA.angka(r.terkirim)
        + ' · gagal ' + SIAGA.angka(r.gagal) + ' · dilewati ' + SIAGA.angka(r.dilewati) + '.',
        'Reminder selesai');
      muatPengaturan(true);
    }).catch(function (e) {
      SIAGA.tombolSibuk(tombol, false);
      SIAGA.gagal(e.message);
    });
  });
}

function tampilkanHasilReminder(r) {
  var lama = SIAGA.el('setModalHasil');
  if (lama) lama.parentNode.removeChild(lama);

  var detail = r.detail || [];
  var wrap = document.createElement('div');
  wrap.className = 'gde-modal is-open';
  wrap.id = 'setModalHasil';
  wrap.innerHTML = ''
    + '<div class="gde-modal__scrim" data-tutup></div>'
    + '<div class="gde-modal__dialog gde-modal__dialog--lg" role="dialog" aria-modal="true">'
    + '  <div class="gde-modal__head">'
    + '    <i class="bi bi-send-check" style="color:var(--gde-green-d)"></i>'
    + '    <h3>Ringkasan pengiriman reminder</h3>'
    + '    <button class="gde-modal__close" type="button" data-tutup aria-label="Tutup">&times;</button>'
    + '  </div>'
    + '  <div class="gde-modal__body">'
    + '    <div class="gde-grid gde-grid--4 siaga-mb-4">'
    + kotakHasilReminder('Target', r.target, 'bi-bullseye', 'blue')
    + kotakHasilReminder('Terkirim', r.terkirim, 'bi-check2-circle', 'green')
    + kotakHasilReminder('Gagal', r.gagal, 'bi-x-circle', 'red')
    + kotakHasilReminder('Dilewati', r.dilewati, 'bi-dash-circle', 'yellow')
    + '    </div>'
    + '    <div class="gde-u-between" style="margin-bottom:.5rem">'
    + '      <strong style="font-size:.85rem">Rincian pengiriman</strong>'
    + '      <span class="siaga-inline-note">Durasi '
    + SIAGA.angka(Math.max(Math.round(SIAGA.num(r.durasi_ms) / 1000), 0)) + ' detik</span>'
    + '    </div>'
    + (detail.length
      ? '    <div class="siaga-scroll-y"><ul class="gde-list">' + detail.map(function (t) {
        return '<li><i class="bi bi-dot" style="color:var(--gde-blue)"></i>'
          + '<span style="font-size:.78rem;word-break:break-word">' + SIAGA.esc(t) + '</span></li>';
      }).join('') + '</ul></div>'
      : SIAGA.kosongHtml('bi-inbox', 'Tidak ada rincian', 'Tidak ada email yang perlu diproses pada penjalanan ini.'))
    + '    <p class="siaga-inline-note siaga-mt-4">Dijalankan pada '
    + SIAGA.esc(r.pada ? SIAGA.tglIndo(r.pada, true) : '—') + '.</p>'
    + '  </div>'
    + '  <div class="gde-modal__foot">'
    + '    <button class="gde-btn gde-btn--primary" type="button" data-tutup>Tutup</button>'
    + '  </div>'
    + '</div>';

  document.body.appendChild(wrap);
  document.body.style.overflow = 'hidden';
  SIAGA.qsa('[data-tutup]', wrap).forEach(function (b) {
    b.addEventListener('click', function () { SIAGA.modalTutup('setModalHasil'); });
  });
}

    + '<div class="siaga-mini__label">' + SIAGA.esc(label) + '</div>'
    + '<div class="siaga-mini__value">' + SIAGA.angka(nilai) + '</div>'
    + '</div></div>';
}
