var PpState = {
  kode: '',
  data: null,
  lampiranAktif: '',
  lampiranTerbuka: true,
  modeEdit: false
};

function initProfilPekerja(param) {
  'use strict';

  var kode = '';
  if (MyCertif.state.paramPekerja) { kode = MyCertif.state.paramPekerja; MyCertif.state.paramPekerja = ''; }
  if (!kode && param) {
    var q = String(param);
    if (q.indexOf('pekerja=') === 0) kode = decodeURIComponent(q.substring(8));
    else if (q.indexOf('kode=') === 0) kode = decodeURIComponent(q.substring(5));
  }
  if (!kode) kode = MyCertif.pengguna().employee_code;

  // Karyawan hanya boleh melihat dirinya sendiri
  if (!MyCertif.isHc()) kode = MyCertif.pengguna().employee_code;

  PpState.kode = kode;
  PpState.modeEdit = false;
  renderPpKerangka();
  muatProfil(kode);
}

/* -------------------------------------------------------------- kerangka -- */
function renderPpKerangka() {
  var isHc = MyCertif.isHc();
  MyCertif.isiHtml('ppRoot', ''
    + '<div class="gde-page-head">'
    + '  <div>'
    + '    <div class="gde-u-eyebrow">Profil &amp; Kompetensi</div>'
    + '    <h1>Profil Pekerja</h1>'
    + '    <p>Identitas pekerja, seluruh sertifikasi yang dimiliki, dan lampiran berkasnya.</p>'
    + '  </div>'
    + '  <div class="gde-page-actions">'
    + (isHc ? ('<div class="mycertif-toolbar__field" style="min-width:260px">'
      + '<label class="gde-label" for="ppCariPekerja">Cari pekerja</label>'
      + '<div class="mycertif-cb" id="ppCb">'
      + '<input class="gde-input" type="text" id="ppCariPekerja" autocomplete="off" '
      + 'placeholder="Ketik nama, Employee Code, atau divisi...">'
      + '<div class="mycertif-cb__panel" id="ppCbPanel"></div>'
      + '</div>'
      + '<div class="gde-hint" id="ppCbHint">Ketik untuk mencari, lalu pilih dari daftar.</div>'
      + '</div>') : '')
    + '    <button class="gde-btn gde-btn--outline" type="button" id="ppEkspor">'
    + '      <i class="bi bi-file-earmark-pdf"></i> Ekspor profil PDF</button>'
    + '    <button class="gde-btn gde-btn--primary" type="button" id="ppTambah">'
    + '      <i class="bi bi-plus-circle"></i> Tambah sertifikasi</button>'
    + '  </div>'
    + '</div>'
    + '<div id="ppIsi"></div>');

  /* ---- PENCARIAN PEKERJA (menggantikan dropdown panjang) ---- */
  var inputCari = MyCertif.el('ppCariPekerja');
  if (inputCari) {
    var hintCb = MyCertif.el('ppCbHint');
    var cbPekerja = MyCertif.combobox({
      input: inputCari,
      panel: MyCertif.el('ppCbPanel'),
      placeholder: 'Ketik nama, Employee Code, atau divisi...',
      onPilih: function (item) {
        PpState.kode = item.value;
        PpState.modeEdit = false;
        muatProfil(PpState.kode);
      }
    });
    PpState.cbPekerja = cbPekerja;

    MyCertif.api('apiListEmployees', [{}]).then(function (d) {
      var rows = d.rows || [];
      var items = rows.map(function (u) {
        var sub = u.employee_code
          + (u.divisi ? (' · ' + u.divisi) : '')
          + (u.jabatan ? (' · ' + u.jabatan) : '')
          + ' · ' + MyCertif.angka(u.jumlah_sertifikasi) + ' sertifikasi';
        return {
          value: u.employee_code,
          label: u.nama_lengkap || u.employee_code,
          sub: sub,
          badge: u.role_label || u.role
        };
      });
      cbPekerja.setItems(items);
      cbPekerja.setNilai(PpState.kode);
      if (hintCb) {
        hintCb.textContent = MyCertif.angka(items.length)
          + ' pekerja terdaftar — ketik untuk mencari, lalu pilih dari daftar.';
      }
    }).catch(function (e) {
      if (hintCb) hintCb.textContent = 'Gagal memuat daftar pekerja: ' + e.message;
    });
  }

  var tambah = MyCertif.el('ppTambah');
  if (tambah) {
    tambah.addEventListener('click', function () {
      MyCertif.state.paramPekerja = PpState.kode;
      MyCertif.navigasi('register');
    });
  }

  var ekspor = MyCertif.el('ppEkspor');
  if (ekspor) {
    ekspor.addEventListener('click', function () {
      MyCertif.tombolSibuk(ekspor, true, 'Menyiapkan...');
      MyCertif.api('apiExportProfilPdf', [PpState.kode, {}]).then(function (r) {
        MyCertif.tombolSibuk(ekspor, false);
        MyCertif.sukses('PDF profil dibuat (' + MyCertif.angka(r.jumlah) + ' sertifikasi).', 'Ekspor selesai');
        window.open(r.url, '_blank');
      }).catch(function (e) {
        MyCertif.tombolSibuk(ekspor, false);
        MyCertif.gagal(e.message);
      });
    });
  }
}

/* ------------------------------------------------------------- pemuatan -- */
function muatProfil(kode) {
  // sinkronkan kotak pencarian bila profil dibuka dari halaman lain
  if (PpState.cbPekerja && String(PpState.cbPekerja.nilai()) !== String(kode)) {
    PpState.cbPekerja.setNilai(kode);
  }
  MyCertif.isiHtml('ppIsi', MyCertif.pemuatHtml('Memuat profil...'));
  MyCertif.api('apiGetEmployeeProfile', [kode]).then(function (d) {
    PpState.data = d;
    if (!PpState.lampiranAktif || !cariLampiran(PpState.lampiranAktif)) {
      var ada = (d.files || []).filter(function (f) { return f.file_id; });
      PpState.lampiranAktif = ada.length ? ada[0].doc_id : '';
    }
    renderPpIsi(d);
  }).catch(function (e) {
    MyCertif.isiHtml('ppIsi', MyCertif.galatHtml(e.message));
  });
}

function cariLampiran(docId) {
  if (!PpState.data) return null;
  var hasil = null;
  (PpState.data.files || []).forEach(function (f) { if (f.doc_id === docId && f.file_id) hasil = f; });
  return hasil;
}

function statusPil(key, jumlah) {
  var def = null;
  (MyCertif.state.meta && MyCertif.state.meta.statuses || []).forEach(function (s) { if (s.key === key) def = s; });
  if (!def) return '';
  return '<span class="gde-status gde-status--' + def.tone + '" title="' + MyCertif.attr(def.ket || '') + '">'
    + '<i class="bi ' + def.icon + '"></i> ' + MyCertif.angka(jumlah || 0) + ' ' + MyCertif.esc(def.label)
    + '</span>';
}

function warnaSisaPp(hari) {
  if (hari === null || hari === undefined || hari === '') return 'var(--gde-muted)';
  var h = MyCertif.num(hari);
  if (h < 0) return '#7F1D1D';
  if (h <= 60) return '#1B2430';
  if (h <= 90) return '#C4142A';
  if (h <= 180) return '#8A4B00';
  if (h <= 270) return '#6B5F00';
  return '#3F6B18';
}

/* -------------------------------------------------- bagian lihat profil -- */
function renderLihatProfil(u, r) {
  var baris = [
    ['Employee Code', u.employee_code],
    ['Username', u.username],
    ['Role', u.role_label || u.role],
    ['Divisi', u.divisi || '—'],
    ['Jabatan', u.jabatan || '—'],
    ['Lokasi kerja', u.lokasi_kerja || '—'],
    ['Email perusahaan', u.email_perusahaan || '—'],
    ['Email pribadi', u.email_pribadi || '—'],
    ['Email atasan (CC reminder)', u.email_atasan || '—'],
    ['No. HP', u.no_hp || '—'],
    ['Status akun', u.status === 'ACTIVE' ? 'Aktif' : 'Nonaktif'],
    ['Login terakhir', u.login_terakhir ? MyCertif.waktuLalu(u.login_terakhir) : 'Belum pernah']
  ];
  return '<dl class="mycertif-dl">' + baris.map(function (b) {
    return '<dt>' + MyCertif.esc(b[0]) + '</dt><dd>' + MyCertif.esc(b[1] === '' ? '—' : b[1]) + '</dd>';
  }).join('') + '</dl>';
}

    + '<div><label class="gde-label" for="ppEmailAtasan">Email atasan</label>'
    + '<input class="gde-input" id="ppEmailAtasan" name="email_atasan" type="email" value="' + MyCertif.attr(u.email_atasan) + '">'
    + '<div class="gde-hint">Akan menerima salinan pengingat sertifikasi.</div></div>'
    + '<div><label class="gde-label" for="ppHp">No. HP</label>'
    + '<input class="gde-input" id="ppHp" name="no_hp" type="text" value="' + MyCertif.attr(u.no_hp) + '"></div>'

      + '<button class="gde-btn gde-btn--outline" type="button" id="ppBatal">Batal</button>'

/* ---------------------------------------------- panel viewer lampiran -- */
function renderViewerLampiran(f) {
  if (!f) {
    return '<div class="mycertif-lampiran__viewer" style="min-height:320px">'
      + '<div class="mycertif-lampiran__empty"><i class="bi bi-hand-index"></i>'
      + '<strong>Pilih salah satu berkas</strong>'
      + '<p class="mycertif-inline-note">Klik baris di bawah untuk menampilkan pratinjau.</p></div></div>';
  }
  var mime = String(f.mime_type || '');
  var ext = MyCertif.extDari(f.file_name);
  var isGambar = mime.indexOf('image/') === 0 || ['png', 'jpg', 'jpeg', 'webp', 'gif', 'bmp'].indexOf(ext) !== -1;
  var isPdf = mime.indexOf('pdf') !== -1 || ext === 'pdf';

  var isi;
  if (isGambar) {
    isi = '<img src="' + MyCertif.attr(MyCertif.pratinjauUrl(f.file_id, f.mime_type, f.file_name)) + '" alt="Lampiran sertifikat">';
  } else if (isPdf) {
    isi = '<iframe src="' + MyCertif.attr(f.preview_url) + '" title="Pratinjau lampiran" loading="lazy"></iframe>';
  } else {
    isi = '<div class="mycertif-lampiran__empty">'
      + '<i class="bi ' + MyCertif.attr(f.file_icon) + '"></i>'
      + '<strong>' + MyCertif.esc(f.file_name || 'Berkas') + '</strong>'
      + '<p class="mycertif-inline-note">Pratinjau tidak tersedia untuk jenis berkas ini.</p></div>';
  }

  return '<div>'
    + '<div class="mycertif-lampiran__toolbar">'
    + '<i class="bi ' + MyCertif.attr(f.file_icon) + '" style="color:var(--gde-blue);font-size:1.1rem"></i>'
    + '<span class="gde-u-truncate" style="flex:1;font-size:.76rem;font-weight:600">'
    + MyCertif.esc(f.nama_sertifikasi) + '</span>'
    + '<a class="gde-btn gde-btn--outline gde-btn--xs" href="' + MyCertif.attr(f.view_url) + '" target="_blank" '
    + 'title="Buka di Drive"><i class="bi bi-box-arrow-up-right"></i></a>'
    + '<a class="gde-btn gde-btn--outline gde-btn--xs" href="' + MyCertif.attr(f.download_url) + '" target="_blank" '
    + 'title="Unduh"><i class="bi bi-download"></i></a>'
    + '</div>'
    + '<div class="mycertif-lampiran__viewer">' + isi + '</div>'
    + '</div>';
}

/* --------------------------------------------------------------- event -- */
function pasangEventPp() {
  var isi = MyCertif.el('ppIsi');

  /* --- toggle panel lampiran --- */
  MyCertif.on(isi, 'click', '#ppToggleLampiran', function () {
    PpState.lampiranTerbuka = !PpState.lampiranTerbuka;
    var body = MyCertif.el('ppLampiranBody');
    if (body) body.classList.toggle('is-hidden', !PpState.lampiranTerbuka);
    var lbl = MyCertif.el('ppToggleLabel');
    if (lbl) lbl.textContent = PpState.lampiranTerbuka ? 'Lipat' : 'Buka';
    var btn = MyCertif.el('ppToggleLampiran');
    if (btn) {
      btn.innerHTML = '<i class="bi ' + (PpState.lampiranTerbuka ? 'bi-arrows-collapse' : 'bi-arrows-expand')
        + '"></i> <span id="ppToggleLabel">' + (PpState.lampiranTerbuka ? 'Lipat' : 'Buka') + '</span>';
    }
  });

  /* --- pilih lampiran --- */
  MyCertif.on(isi, 'click', '[data-lampiran]', function (e, t) {
    var docId = t.getAttribute('data-lampiran');
    PpState.lampiranAktif = docId;
    PpState.lampiranTerbuka = true;
    var f = cariLampiran(docId);
    var body = MyCertif.el('ppLampiranBody');
    if (body && f) {
      var viewer = body.querySelector('.mycertif-lampiran__viewer');
      var lama = body.querySelector('[data-viewer-wrap]');
      var htmlViewer = renderViewerLampiran(f);
      if (lama) lama.parentNode.removeChild(lama);
      var tmp = document.createElement('div');
      tmp.setAttribute('data-viewer-wrap', '1');
      tmp.innerHTML = htmlViewer;
      body.insertBefore(tmp, body.firstChild);
      if (viewer) viewer.parentNode.removeChild(viewer);
      body.classList.remove('is-hidden');
      /* segarkan penanda aktif */
      MyCertif.qsa('[data-lampiran]', body).forEach(function (n) {
        n.classList.toggle('is-active', n.getAttribute('data-lampiran') === docId);
      });
      try { body.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); } catch (x) { }
    }
  });

  /* --- ubah / hapus sertifikasi --- */
  MyCertif.on(isi, 'click', '[data-ubah]', function (e, t) {
    MyCertif.state.paramDoc = t.getAttribute('data-ubah');
    MyCertif.navigasi('register');
  });

  MyCertif.on(isi, 'click', '[data-hapus]', function (e, t) {
    var docId = t.getAttribute('data-hapus');
    var f = null;
    (PpState.data.files || []).forEach(function (x) { if (x.doc_id === docId) f = x; });
    if (!f) return;
    MyCertif.konfirmasi({
      judul: 'Hapus data sertifikasi',
      pesanHtml: 'Data <b>' + MyCertif.esc(f.nama_sertifikasi) + '</b> akan dihapus permanen'
        + (f.file_id ? ' beserta lampirannya' : '') + '.',
      labelOk: 'Hapus permanen', bahaya: true
    }).then(function (ya) {
      if (!ya) return;
      return MyCertif.api('apiDeleteFile', [docId]).then(function () {
        MyCertif.sukses('Data dihapus.', 'Berhasil');
        muatProfil(PpState.kode);
      });
    }).catch(function (e2) { MyCertif.gagal(e2.message); });
  });

  var tambah2 = MyCertif.el('ppTambah2');
  if (tambah2) tambah2.addEventListener('click', function () {
    MyCertif.state.paramPekerja = PpState.kode;
    MyCertif.navigasi('register');
  });

  /* --- mode edit profil --- */
  var editBtn = MyCertif.el('ppEditBtn');
  if (editBtn) {
    editBtn.addEventListener('click', function () {
      PpState.modeEdit = !PpState.modeEdit;
      var form = MyCertif.el('ppFormEdit');
      var lihat = MyCertif.el('ppModeLihat');
      if (form) form.classList.toggle('gde-u-hide', !PpState.modeEdit);
      if (lihat) lihat.classList.toggle('gde-u-hide', PpState.modeEdit);
      var lbl = MyCertif.el('ppEditLabel');
      if (lbl) lbl.textContent = PpState.modeEdit ? 'Batal ubah' : 'Ubah';
    });
  }

  var batal = MyCertif.el('ppBatal');
  if (batal) batal.addEventListener('click', function () {
    PpState.modeEdit = false;
    muatProfil(PpState.kode);
  });

  var form = MyCertif.el('ppFormEdit');
  if (form) {
    form.addEventListener('submit', function (e) {
      e.preventDefault();
      simpanProfil(form);
    });
  }
}

/* --------------------------------------------------------------- simpan -- */
function simpanProfil(form) {
  var u = (PpState.data && PpState.data.user) || {};
  var isHc = MyCertif.isHc();
  var diri = (u.employee_code === MyCertif.pengguna().employee_code);

  MyCertif.bersihkanInvalid(form);
  var data = MyCertif.nilaiForm(form);

  /* validasi email sederhana */
  var petaEmail = {
    email_perusahaan: 'Email perusahaan', email_pribadi: 'Email pribadi',
    email_atasan: 'Email atasan'
  };
  var invalid = false;
  Object.keys(petaEmail).forEach(function (k) {
    var v = String(data[k] || '').trim();
    if (!v) return;
    var node = form.querySelector('[name="' + k + '"]');
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) {
      MyCertif.tandaiInvalid(node, petaEmail[k] + ' tidak valid.');
      invalid = true;
    }
  });
  if (invalid) return;

  var tombol = MyCertif.el('ppSimpan');
  MyCertif.tombolSibuk(tombol, true, 'Menyimpan...');

  var aksi;
  if (isHc && !diri) {
    /* HC mengubah data pekerja lain -> apiSaveUser butuh field lengkap */
    aksi = MyCertif.api('apiSaveUser', [{
      employee_code: u.employee_code,
      nama_lengkap: data.nama_lengkap,
      username: data.username,
      role: u.role,
      email_perusahaan: data.email_perusahaan,
      email_pribadi: data.email_pribadi,
      email_atasan: data.email_atasan,
      divisi: data.divisi,
      jabatan: data.jabatan,
      lokasi_kerja: data.lokasi_kerja,
      no_hp: data.no_hp,
      status: data.status
    }]);
  } else {
    /* diri sendiri -> kolom terbatas */
    aksi = MyCertif.api('apiUpdateProfilSaya', [{
      nama_lengkap: data.nama_lengkap,
      email_perusahaan: data.email_perusahaan,
      email_pribadi: data.email_pribadi,
      email_atasan: data.email_atasan,
      divisi: data.divisi,
      jabatan: data.jabatan,
      lokasi_kerja: data.lokasi_kerja,
      no_hp: data.no_hp
    }]);
  }

  aksi.then(function () {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.sukses('Data pekerja berhasil diperbarui.', 'Tersimpan');
    PpState.modeEdit = false;
    if (diri) {
      MyCertif.api('apiGetSession', []).then(function (s) {
        MyCertif.state.user = s.user;
        MyCertif.renderShell();
        muatProfil(PpState.kode);
      }).catch(function () { muatProfil(PpState.kode); });
    } else {
      muatProfil(PpState.kode);
    }
  }).catch(function (err) {
    MyCertif.tombolSibuk(tombol, false);
    MyCertif.gagal(err.message);
  });
}
