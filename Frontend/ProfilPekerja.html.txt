var PpState = {
  kode: '',
  data: null,
  lampiranAktif: '',
  lampiranTerbuka: true,
  modeEdit: false
};

function initProfilPekerja(param) {
  'use strict';

  var kode = '';
  if (SIAGA.state.paramPekerja) { kode = SIAGA.state.paramPekerja; SIAGA.state.paramPekerja = ''; }
  if (!kode && param) {
    var q = String(param);
    if (q.indexOf('pekerja=') === 0) kode = decodeURIComponent(q.substring(8));
    else if (q.indexOf('kode=') === 0) kode = decodeURIComponent(q.substring(5));
  }
  if (!kode) kode = SIAGA.pengguna().employee_code;

  // Karyawan hanya boleh melihat dirinya sendiri
  if (!SIAGA.isHc()) kode = SIAGA.pengguna().employee_code;

  PpState.kode = kode;
  PpState.modeEdit = false;
  renderPpKerangka();
  muatProfil(kode);
}

/* -------------------------------------------------------------- kerangka -- */
function renderPpKerangka() {
  var isHc = SIAGA.isHc();
  SIAGA.el('ppRoot').innerHTML = ''
    + '<div class="gde-page-head">'
    + '  <div>'
    + '    <div class="gde-u-eyebrow">Profil &amp; Kompetensi</div>'
    + '    <h1>Profil Pekerja</h1>'
    + '    <p>Identitas pekerja, seluruh sertifikasi yang dimiliki, dan lampiran berkasnya.</p>'
    + '  </div>'
    + '  <div class="gde-page-actions">'
    + (isHc ? ('<div class="siaga-toolbar__field" style="min-width:260px">'
      + '<label class="gde-label" for="ppPilih">Pilih pekerja</label>'
      + '<select class="gde-select" id="ppPilih"></select></div>') : '')
    + '    <button class="gde-btn gde-btn--outline" type="button" id="ppEkspor">'
    + '      <i class="bi bi-file-earmark-pdf"></i> Ekspor profil PDF</button>'
    + '    <button class="gde-btn gde-btn--primary" type="button" id="ppTambah">'
    + '      <i class="bi bi-plus-circle"></i> Tambah sertifikasi</button>'
    + '  </div>'
    + '</div>'
    + '<div id="ppIsi"></div>';

  var pilih = SIAGA.el('ppPilih');
  if (pilih) {
    pilih.innerHTML = '<option value="">Memuat daftar pekerja...</option>';
    SIAGA.api('apiListEmployees', [{}]).then(function (d) {
      pilih.innerHTML = (d.rows || []).map(function (u) {
        return '<option value="' + SIAGA.attr(u.employee_code) + '"'
          + (u.employee_code === PpState.kode ? ' selected' : '') + '>'
          + SIAGA.esc(u.nama_lengkap) + ' — ' + SIAGA.esc(u.employee_code)
          + (u.divisi ? (' (' + SIAGA.esc(u.divisi) + ')') : '') + '</option>';
      }).join('');
      pilih.value = PpState.kode;
    }).catch(function () {
      pilih.innerHTML = '<option value="' + SIAGA.attr(PpState.kode) + '">' + SIAGA.esc(PpState.kode) + '</option>';
    });

    pilih.addEventListener('change', function () {
      PpState.kode = pilih.value;
      PpState.modeEdit = false;
      muatProfil(PpState.kode);
    });
  }

  var tambah = SIAGA.el('ppTambah');
  if (tambah) {
    tambah.addEventListener('click', function () {
      SIAGA.state.paramPekerja = PpState.kode;
      SIAGA.navigasi('register');
    });
  }

  var ekspor = SIAGA.el('ppEkspor');
  if (ekspor) {
    ekspor.addEventListener('click', function () {
      SIAGA.tombolSibuk(ekspor, true, 'Menyiapkan...');
      SIAGA.api('apiExportProfilPdf', [PpState.kode, {}]).then(function (r) {
        SIAGA.tombolSibuk(ekspor, false);
        SIAGA.sukses('PDF profil dibuat (' + SIAGA.angka(r.jumlah) + ' sertifikasi).', 'Ekspor selesai');
        window.open(r.url, '_blank');
      }).catch(function (e) {
        SIAGA.tombolSibuk(ekspor, false);
        SIAGA.gagal(e.message);
      });
    });
  }
}

/* ------------------------------------------------------------- pemuatan -- */
function muatProfil(kode) {
  SIAGA.el('ppIsi').innerHTML = SIAGA.pemuatHtml('Memuat profil...');
  SIAGA.api('apiGetEmployeeProfile', [kode]).then(function (d) {
    PpState.data = d;
    if (!PpState.lampiranAktif || !cariLampiran(PpState.lampiranAktif)) {
      var ada = (d.files || []).filter(function (f) { return f.file_id; });
      PpState.lampiranAktif = ada.length ? ada[0].doc_id : '';
    }
    renderPpIsi(d);
  }).catch(function (e) {
    SIAGA.el('ppIsi').innerHTML = SIAGA.galatHtml(e.message);
  });
}

function cariLampiran(docId) {
  if (!PpState.data) return null;
  var hasil = null;
  (PpState.data.files || []).forEach(function (f) { if (f.doc_id === docId && f.file_id) hasil = f; });
  return hasil;
}

function statusPil(key, jumlah) {
  var def = null;
  (SIAGA.state.meta && SIAGA.state.meta.statuses || []).forEach(function (s) { if (s.key === key) def = s; });
  if (!def) return '';
  return '<span class="gde-status gde-status--' + def.tone + '" title="' + SIAGA.attr(def.ket || '') + '">'
    + '<i class="bi ' + def.icon + '"></i> ' + SIAGA.angka(jumlah || 0) + ' ' + SIAGA.esc(def.label)
    + '</span>';
}

function warnaSisaPp(hari) {
  if (hari === null || hari === undefined || hari === '') return 'var(--gde-muted)';
  var h = SIAGA.num(hari);
  if (h < 0) return '#7F1D1D';
  if (h <= 60) return '#1B2430';
  if (h <= 90) return '#C4142A';
  if (h <= 180) return '#8A4B00';
  if (h <= 270) return '#6B5F00';
  return '#3F6B18';
}

/* -------------------------------------------------- bagian lihat profil -- */
function renderLihatProfil(u, r) {
  var baris = [
    ['Employee Code', u.employee_code],
    ['Username', u.username],
    ['Role', u.role_label || u.role],
    ['Divisi', u.divisi || '—'],
    ['Jabatan', u.jabatan || '—'],
    ['Lokasi kerja', u.lokasi_kerja || '—'],
    ['Email perusahaan', u.email_perusahaan || '—'],
    ['Email pribadi', u.email_pribadi || '—'],
    ['Email atasan (CC reminder)', u.email_atasan || '—'],
    ['No. HP', u.no_hp || '—'],
    ['Status akun', u.status === 'ACTIVE' ? 'Aktif' : 'Nonaktif'],
    ['Login terakhir', u.login_terakhir ? SIAGA.waktuLalu(u.login_terakhir) : 'Belum pernah']
  ];
  return '<dl class="siaga-dl">' + baris.map(function (b) {
    return '<dt>' + SIAGA.esc(b[0]) + '</dt><dd>' + SIAGA.esc(b[1] === '' ? '—' : b[1]) + '</dd>';
  }).join('') + '</dl>';
}

      + '<p class="siaga-inline-note">Klik baris di bawah untuk menampilkan pratinjau.</p></div></div>';
  }
  var mime = String(f.mime_type || '');
  var ext = SIAGA.extDari(f.file_name);
  var isGambar = mime.indexOf('image/') === 0 || ['png', 'jpg', 'jpeg', 'webp', 'gif', 'bmp'].indexOf(ext) !== -1;
  var isPdf = mime.indexOf('pdf') !== -1 || ext === 'pdf';

  var isi;
  if (isGambar) {
    isi = '<img src="' + SIAGA.attr(SIAGA.pratinjauUrl(f.file_id, f.mime_type, f.file_name)) + '" alt="Lampiran sertifikat">';
  } else if (isPdf) {
    isi = '<iframe src="' + SIAGA.attr(f.preview_url) + '" title="Pratinjau lampiran" loading="lazy"></iframe>';
  } else {
    isi = '<div class="siaga-lampiran__empty">'
      + '<i class="bi ' + SIAGA.attr(f.file_icon) + '"></i>'
      + '<strong>' + SIAGA.esc(f.file_name || 'Berkas') + '</strong>'
      + '<p class="siaga-inline-note">Pratinjau tidak tersedia untuk jenis berkas ini.</p></div>';
  }

  return '<div>'
    + '<div class="siaga-lampiran__toolbar">'
    + '<i class="bi ' + SIAGA.attr(f.file_icon) + '" style="color:var(--gde-blue);font-size:1.1rem"></i>'
    + '<span class="gde-u-truncate" style="flex:1;font-size:.76rem;font-weight:600">'
    + SIAGA.esc(f.nama_sertifikasi) + '</span>'
    + '<a class="gde-btn gde-btn--outline gde-btn--xs" href="' + SIAGA.attr(f.view_url) + '" target="_blank" '
    + 'title="Buka di Drive"><i class="bi bi-box-arrow-up-right"></i></a>'
    + '<a class="gde-btn gde-btn--outline gde-btn--xs" href="' + SIAGA.attr(f.download_url) + '" target="_blank" '
    + 'title="Unduh"><i class="bi bi-download"></i></a>'
    + '</div>'
    + '<div class="siaga-lampiran__viewer">' + isi + '</div>'
    + '</div>';
}

/* --------------------------------------------------------------- event -- */
function pasangEventPp() {
  var isi = SIAGA.el('ppIsi');

  /* --- toggle panel lampiran --- */
  SIAGA.on(isi, 'click', '#ppToggleLampiran', function () {
    PpState.lampiranTerbuka = !PpState.lampiranTerbuka;
    var body = SIAGA.el('ppLampiranBody');
    if (body) body.classList.toggle('is-hidden', !PpState.lampiranTerbuka);
    var lbl = SIAGA.el('ppToggleLabel');
    if (lbl) lbl.textContent = PpState.lampiranTerbuka ? 'Lipat' : 'Buka';
    var btn = SIAGA.el('ppToggleLampiran');
    if (btn) {
      btn.innerHTML = '<i class="bi ' + (PpState.lampiranTerbuka ? 'bi-arrows-collapse' : 'bi-arrows-expand')
        + '"></i> <span id="ppToggleLabel">' + (PpState.lampiranTerbuka ? 'Lipat' : 'Buka') + '</span>';
    }
  });

  /* --- pilih lampiran --- */
  SIAGA.on(isi, 'click', '[data-lampiran]', function (e, t) {
    var docId = t.getAttribute('data-lampiran');
    PpState.lampiranAktif = docId;
    PpState.lampiranTerbuka = true;
    var f = cariLampiran(docId);
    var body = SIAGA.el('ppLampiranBody');
    if (body && f) {
      var viewer = body.querySelector('.siaga-lampiran__viewer');
      var lama = body.querySelector('[data-viewer-wrap]');
      var htmlViewer = renderViewerLampiran(f);
      if (lama) lama.parentNode.removeChild(lama);
      var tmp = document.createElement('div');
      tmp.setAttribute('data-viewer-wrap', '1');
      tmp.innerHTML = htmlViewer;
      body.insertBefore(tmp, body.firstChild);
      if (viewer) viewer.parentNode.removeChild(viewer);
      body.classList.remove('is-hidden');
      /* segarkan penanda aktif */
      SIAGA.qsa('[data-lampiran]', body).forEach(function (n) {
        n.classList.toggle('is-active', n.getAttribute('data-lampiran') === docId);
      });
      try { body.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); } catch (x) { }
    }
  });

  /* --- ubah / hapus sertifikasi --- */
  SIAGA.on(isi, 'click', '[data-ubah]', function (e, t) {
    SIAGA.state.paramDoc = t.getAttribute('data-ubah');
    SIAGA.navigasi('register');
  });

  SIAGA.on(isi, 'click', '[data-hapus]', function (e, t) {
    var docId = t.getAttribute('data-hapus');
    var f = null;
    (PpState.data.files || []).forEach(function (x) { if (x.doc_id === docId) f = x; });
    if (!f) return;
    SIAGA.konfirmasi({
      judul: 'Hapus data sertifikasi',
      pesanHtml: 'Data <b>' + SIAGA.esc(f.nama_sertifikasi) + '</b> akan dihapus permanen'
        + (f.file_id ? ' beserta lampirannya' : '') + '.',
      labelOk: 'Hapus permanen', bahaya: true
    }).then(function (ya) {
      if (!ya) return;
      return SIAGA.api('apiDeleteFile', [docId]).then(function () {
        SIAGA.sukses('Data dihapus.', 'Berhasil');
        muatProfil(PpState.kode);
      });
    }).catch(function (e2) { SIAGA.gagal(e2.message); });
  });

  var tambah2 = SIAGA.el('ppTambah2');
  if (tambah2) tambah2.addEventListener('click', function () {
    SIAGA.state.paramPekerja = PpState.kode;
    SIAGA.navigasi('register');
  });

  /* --- mode edit profil --- */
  var editBtn = SIAGA.el('ppEditBtn');
  if (editBtn) {
    editBtn.addEventListener('click', function () {
      PpState.modeEdit = !PpState.modeEdit;
      var form = SIAGA.el('ppFormEdit');
      var lihat = SIAGA.el('ppModeLihat');
      if (form) form.classList.toggle('gde-u-hide', !PpState.modeEdit);
      if (lihat) lihat.classList.toggle('gde-u-hide', PpState.modeEdit);
      var lbl = SIAGA.el('ppEditLabel');
      if (lbl) lbl.textContent = PpState.modeEdit ? 'Batal ubah' : 'Ubah';
    });
  }

  var batal = SIAGA.el('ppBatal');
  if (batal) batal.addEventListener('click', function () {
    PpState.modeEdit = false;
    muatProfil(PpState.kode);
  });

  var form = SIAGA.el('ppFormEdit');
  if (form) {
    form.addEventListener('submit', function (e) {
      e.preventDefault();
      simpanProfil(form);
    });
  }
}

/* --------------------------------------------------------------- simpan -- */
function simpanProfil(form) {
  var u = (PpState.data && PpState.data.user) || {};
  var isHc = SIAGA.isHc();
  var diri = (u.employee_code === SIAGA.pengguna().employee_code);

  SIAGA.bersihkanInvalid(form);
  var data = SIAGA.nilaiForm(form);

  /* validasi email sederhana */
  var petaEmail = {
    email_perusahaan: 'Email perusahaan', email_pribadi: 'Email pribadi',
    email_atasan: 'Email atasan'
  };
  var invalid = false;
  Object.keys(petaEmail).forEach(function (k) {
    var v = String(data[k] || '').trim();
    if (!v) return;
    var node = form.querySelector('[name="' + k + '"]');
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) {
      SIAGA.tandaiInvalid(node, petaEmail[k] + ' tidak valid.');
      invalid = true;
    }
  });
  if (invalid) return;

  var tombol = SIAGA.el('ppSimpan');
  SIAGA.tombolSibuk(tombol, true, 'Menyimpan...');

  var aksi;
  if (isHc && !diri) {
    /* HC mengubah data pekerja lain -> apiSaveUser butuh field lengkap */
    aksi = SIAGA.api('apiSaveUser', [{
      employee_code: u.employee_code,
      nama_lengkap: data.nama_lengkap,
      username: data.username,
      role: u.role,
      email_perusahaan: data.email_perusahaan,
      email_pribadi: data.email_pribadi,
      email_atasan: data.email_atasan,
      divisi: data.divisi,
      jabatan: data.jabatan,
      lokasi_kerja: data.lokasi_kerja,
      no_hp: data.no_hp,
      status: data.status
    }]);
  } else {
    /* diri sendiri -> kolom terbatas */
    aksi = SIAGA.api('apiUpdateProfilSaya', [{
      nama_lengkap: data.nama_lengkap,
      email_perusahaan: data.email_perusahaan,
      email_pribadi: data.email_pribadi,
      email_atasan: data.email_atasan,
      divisi: data.divisi,
      jabatan: data.jabatan,
      lokasi_kerja: data.lokasi_kerja,
      no_hp: data.no_hp
    }]);
  }

  aksi.then(function () {
    SIAGA.tombolSibuk(tombol, false);
    SIAGA.sukses('Data pekerja berhasil diperbarui.', 'Tersimpan');
    PpState.modeEdit = false;
    if (diri) {
      SIAGA.api('apiGetSession', []).then(function (s) {
        SIAGA.state.user = s.user;
        SIAGA.renderShell();
        muatProfil(PpState.kode);
      }).catch(function () { muatProfil(PpState.kode); });
    } else {
      muatProfil(PpState.kode);
    }
  }).catch(function (err) {
    SIAGA.tombolSibuk(tombol, false);
    SIAGA.gagal(err.message);
  });
}
