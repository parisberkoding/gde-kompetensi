/* --------------------------------------------------------- pasang event -- */
function pasangPerilakuMu() {
  var akar = MyCertif.el('muRoot');
  var form = MyCertif.el('muForm');
  var cari = MyCertif.el('muCari');

  /* --- header --- */
  var tambah = MyCertif.el('muTambahBtn');
  if (tambah) tambah.addEventListener('click', function () { bukaMuForm('baru', ''); });
  var impor = MyCertif.el('muImporBtn');
  if (impor) impor.addEventListener('click', bukaMuImpor);
  var sync = MyCertif.el('muSyncBtn');
  if (sync) sync.addEventListener('click', muSinkron);

  /* --- toolbar (klien) --- */
  if (cari) {
    cari.addEventListener('input', MyCertif.debounce(function () {
      MUState.filter.q = cari.value;
      MUState.halaman = 1;
      renderMuTabel();
    }, 320));
  }
  var fRole = MyCertif.el('muRoleFilter');
  if (fRole) {
    fRole.addEventListener('change', function () {
      MUState.filter.role = fRole.value;
      MUState.halaman = 1;
      renderMuTabel();
    });
  }
  var fDivisi = MyCertif.el('muDivisiFilter');
  if (fDivisi) {
    fDivisi.addEventListener('change', function () {
      MUState.filter.divisi = fDivisi.value;
      MUState.halaman = 1;
      renderMuTabel();
    });
  }
  var fStatus = MyCertif.el('muStatusFilter');
  if (fStatus) {
    fStatus.addEventListener('change', function () {
      MUState.filter.status = fStatus.value;
      MUState.halaman = 1;
      renderMuTabel();
    });
  }
  var reset = MyCertif.el('muReset');
  if (reset) {
    reset.addEventListener('click', function () {
      MUState.filter = { q: '', role: '', divisi: '', status: '' };
      MUState.halaman = 1;
      if (cari) cari.value = '';
      if (fRole) fRole.value = '';
      if (fDivisi) fDivisi.value = '';
      if (fStatus) fStatus.value = '';
      renderMuTabel();
    });
  }

  /* --- form --- */
  if (form) {
    form.addEventListener('submit', function (e) { e.preventDefault(); simpanMu(); });
  }
  var simpan = MyCertif.el('muSimpan');
  if (simpan) simpan.addEventListener('click', simpanMu);

  /* --- impor --- */
  var imporJalan = MyCertif.el('muImporJalan');
  if (imporJalan) imporJalan.addEventListener('click', jalanMuImpor);

  /* --- delegasi umum --- */
  MyCertif.on(akar, 'click', '[data-mu-tutup]', function (e, t) {
    MyCertif.modalTutup(t.getAttribute('data-mu-tutup'));
  });

  MyCertif.on(akar, 'click', '[data-mu-hal]', function (e, t) {
    e.preventDefault();
    MUState.halaman = MyCertif.num(t.getAttribute('data-mu-hal'), 1);
    renderMuTabel();
  });

  MyCertif.on(akar, 'click', '[data-mu-pekerja]', function (e, t) {
    e.preventDefault();
    MyCertif.state.paramPekerja = t.getAttribute('data-mu-pekerja');
    MyCertif.navigasi('pekerja');
  });

  MyCertif.on(akar, 'click', '[data-mu-aksi]', function (e, t) {
    var aksi = t.getAttribute('data-mu-aksi');
    var kode = t.getAttribute('data-mu-kode') || '';
    if (aksi === 'tambah') bukaMuForm('baru', '');
    else if (aksi === 'ubah') bukaMuForm('ubah', kode);
    else if (aksi === 'reset') muResetPassword(kode);
    else if (aksi === 'toggle') muToggleStatus(kode);
    else if (aksi === 'hapus') muHapus(kode);
  });
}

/* --------------------------------------------------------------- muatan -- */
function muatUser() {
  var isi = MyCertif.el('muIsi');
  if (isi) isi.innerHTML = MyCertif.pemuatHtml('Memuat data user...');

  var janjiMeta = MyCertif.api('apiGetMeta', []).then(function (m) {
    return m;
  }).catch(function () { return null; });

  var janjiUser = MyCertif.api('apiListUsers', [{}]);

  Promise.all([janjiMeta, janjiUser]).then(function (hasil) {
    MUState.meta = hasil[0] || MUState.meta;
    var d = hasil[1] || {};
    MUState.baris = d.rows || [];
    muIsiPilihanDivisi();
    renderMuKpi();
    renderMuTabel();
  }).catch(function (e) {
    var wadah = MyCertif.el('muIsi');
    if (wadah) wadah.innerHTML = MyCertif.galatHtml(e.message);
    var kpi = MyCertif.el('muKpi');
    if (kpi) kpi.innerHTML = '';
  });
}

function muIsiPilihanDivisi() {
  var daftar = ((MUState.meta && MUState.meta.divisi_options) || []).slice();
  MUState.baris.forEach(function (u) {
    var v = String(u.divisi || '').trim();
    if (v && daftar.indexOf(v) === -1) daftar.push(v);
  });
  daftar.sort();

  var sel = MyCertif.el('muDivisiFilter');
  if (sel) {
    var nilai = MUState.filter.divisi;
    sel.innerHTML = '<option value="">Semua divisi</option>'
      + daftar.map(function (o) {
        return '<option value="' + MyCertif.attr(o) + '">' + MyCertif.esc(o) + '</option>';
      }).join('');
    sel.value = nilai;
  }

  var dl = MyCertif.el('muDaftarDivisi');
  if (dl) {
    dl.innerHTML = daftar.map(function (o) {
      return '<option value="' + MyCertif.attr(o) + '"></option>';
    }).join('');
  }
}

/* ------------------------------------------------------------------ KPI -- */
function renderMuKpi() {
  var total = MUState.baris.length;
  var hcsa = 0, emp = 0, nonaktif = 0;
  MUState.baris.forEach(function (u) {
    var r = String(u.role || '').toUpperCase();
    if (r === 'HC' || r === 'SA') hcsa++;
    if (r === 'EMP') emp++;
    if (String(u.status || '').toUpperCase() === 'INACTIVE') nonaktif++;
  });

  var wadah = MyCertif.el('muKpi');
  if (!wadah) return;
  wadah.innerHTML = ''
    + muMini('Total user', total, 'bi-people', 'blue')
    + muMini('HC / SA', hcsa, 'bi-shield-lock', 'green')
    + muMini('Employee (EMP)', emp, 'bi-person-badge', 'yellow')
    + muMini('Akun nonaktif', nonaktif, 'bi-person-slash', 'black');
}

function muMini(label, nilai, ikon, tone) {
  return '<div class="mycertif-mini mycertif-mini--' + tone + '">'
    + '<div class="mycertif-mini__icon"><i class="bi ' + ikon + '"></i></div>'
    + '<div style="min-width:0">'
    + '<div class="mycertif-mini__label">' + MyCertif.esc(label) + '</div>'
    + '<div class="mycertif-mini__value">' + MyCertif.angka(nilai) + '</div>'
    + '</div></div>';
}

/* ---------------------------------------------------------------- tabel -- */
function muBarisTersaring() {
  var f = MUState.filter;
  var q = String(f.q || '').trim().toLowerCase();
  var role = String(f.role || '').toUpperCase();
  var status = String(f.status || '').toUpperCase();

  return MUState.baris.filter(function (u) {
    if (role && String(u.role || '').toUpperCase() !== role) return false;
    if (f.divisi && String(u.divisi || '') !== String(f.divisi)) return false;
    if (status && String(u.status || '').toUpperCase() !== status) return false;
    if (q) {
      var gab = [u.employee_code, u.nama_lengkap, u.username, u.email_perusahaan,
        u.email_pribadi, u.jabatan, u.divisi].join(' ').toLowerCase();
      if (gab.indexOf(q) === -1) return false;
    }
    return true;
  });
}

function renderMuTabel() {
  var wadah = MyCertif.el('muIsi');
  if (!wadah) return;

  var semua = muBarisTersaring();
  var per = MUState.perHalaman;
  var maxHal = Math.max(1, Math.ceil(semua.length / per));
  if (MUState.halaman > maxHal) MUState.halaman = maxHal;
  if (MUState.halaman < 1) MUState.halaman = 1;
  var mulai = (MUState.halaman - 1) * per;
  var halamanIni = semua.slice(mulai, mulai + per);

  /* keadaan loading sudah ditangani pemanggil; di sini kosong / terisi */
  if (!MUState.baris.length) {
    wadah.innerHTML = MyCertif.kosongHtml('bi-people', 'Belum ada user',
      'Tambahkan akun satu per satu, impor massal, atau sinkronkan dari data sertifikasi yang sudah ada.',

        + '<i class="bi bi-person-plus"></i> Tambah user</button>'
        + '<button class="gde-btn gde-btn--outline-primary" type="button" id="muKosongSync">'
        + '<i class="bi bi-arrow-repeat"></i> Sinkron dari data sertifikasi</button>'
      + '</div>');
    var kosongSync = MyCertif.el('muKosongSync');
    if (kosongSync) kosongSync.addEventListener('click', muSinkron);
    return;
  }

  if (!semua.length) {
    wadah.innerHTML = MyCertif.kosongHtml('bi-search', 'Tidak ada user yang cocok',
      'Ubah kata kunci atau reset filter untuk melihat seluruh user.');
    return;
  }

  var html = '<div class="mycertif-tablewrap">'
    + '<table class="gde-table gde-table--compact">'
    + '<thead><tr>'
    + '<th class="gde-ctr" style="width:52px">No</th>'
    + '<th>Pekerja</th>'
    + '<th>Username</th>'
    + '<th class="gde-ctr">Role</th>'
    + '<th>Divisi / Jabatan</th>'
    + '<th>Email</th>'
    + '<th class="gde-ctr">Sertifikasi</th>'
    + '<th class="gde-ctr">Status</th>'
    + '<th class="gde-u-text-right gde-u-nowrap">Aksi</th>'
    + '</tr></thead><tbody>';

  html += halamanIni.map(function (u, i) {
    return muBarisHtml(u, mulai + i + 1);
  }).join('');

  html += '</tbody></table></div>';
  html += muPagerHtml(semua.length);

  wadah.innerHTML = html;
}

function muPagerHtml(total) {
  var per = MUState.perHalaman;
  var maxHal = Math.max(1, Math.ceil(total / per));
  var hal = MUState.halaman;
  var dari = total ? ((hal - 1) * per + 1) : 0;
  var sampai = Math.min(hal * per, total);

  var awal = Math.max(1, hal - 2);
  var akhir = Math.min(maxHal, awal + 4);
  awal = Math.max(1, akhir - 4);

  var nomor = '';
  for (var i = awal; i <= akhir; i++) {
    if (i === hal) nomor += '<span class="is-active">' + MyCertif.angka(i) + '</span>';
    else nomor += '<a href="#" data-mu-hal="' + i + '">' + MyCertif.angka(i) + '</a>';
  }

  return '<div class="mycertif-pager">'
    + '<div class="mycertif-pager__info">Menampilkan <b>' + MyCertif.angka(dari) + '–' + MyCertif.angka(sampai)
    + '</b> dari <b>' + MyCertif.angka(total) + '</b> user · halaman ' + MyCertif.angka(hal) + ' / ' + MyCertif.angka(maxHal)
    + '</div>'
    + '<div class="gde-pagination">'
    + '<a href="#" data-mu-hal="' + (hal - 1) + '"' + (hal <= 1 ? ' class="is-disabled"' : '') + '>Sebelumnya</a>'
    + nomor
    + '<a href="#" data-mu-hal="' + (hal + 1) + '"' + (hal >= maxHal ? ' class="is-disabled"' : '') + '>Berikutnya</a>'
    + '</div></div>';
}

/* -------------------------------------------------------------- bantuan -- */
function muCari(kode) {
  for (var i = 0; i < MUState.baris.length; i++) {
    if (String(MUState.baris[i].employee_code) === String(kode)) return MUState.baris[i];
  }
  return null;
}

function muBersihkan(obj) {
  var out = {};
  Object.keys(obj || {}).forEach(function (k) {
    var v = obj[k];
    out[k] = (typeof v === 'string') ? v.trim() : v;
  });
  return out;
}

function muEmailValid(v) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(String(v || '').trim());
}

/* Data user dari baris tabel — dipakai untuk simpan ulang (mis. ubah status).
   Data ini disegarkan lewat apiGetUser sebelum dikirim, lihat muToggleStatus(). */
function muDataDariBaris(u, statusBaru) {
  return {
    employee_code: u.employee_code,
    nama_lengkap: u.nama_lengkap,
    username: u.username,
    role: u.role,
    email_perusahaan: u.email_perusahaan,
    email_pribadi: u.email_pribadi,
    email_atasan: u.email_atasan,
    divisi: u.divisi,
    jabatan: u.jabatan,
    lokasi_kerja: u.lokasi_kerja,
    no_hp: u.no_hp,
    status: statusBaru || u.status
  };
}

/* ------------------------------------------------------------ form user -- */
function bukaMuForm(mode, kode) {
  var form = MyCertif.el('muForm');
  if (!form) return;

  MyCertif.resetForm(form);
  MyCertif.bersihkanInvalid(form);
  MUState.mode = mode;
  MUState.kode = kode || '';

  var judul = MyCertif.el('muModalJudul');
  var inKode = MyCertif.el('muEmployeeCode');
  var hintKode = MyCertif.el('muHintCode');
  var passWrap = MyCertif.el('muPassWrap');

  if (mode === 'ubah') {
    var u = muCari(kode);
    if (!u) { MyCertif.gagal('Data user tidak ditemukan pada tabel. Muat ulang halaman.'); return; }
    if (judul) judul.textContent = 'Ubah user — ' + (u.nama_lengkap || u.employee_code);
    if (inKode) inKode.disabled = true;
    if (hintKode) {
      hintKode.innerHTML = 'Employee Code tidak dapat diubah karena menjadi kunci data: <b>'
        + MyCertif.esc(u.employee_code) + '</b>.';
    }
    if (passWrap) passWrap.classList.remove('gde-u-hide');
    MyCertif.isiForm(form, u);
    // Segarkan dari server: tabel bisa saja sudah usang bila HC lain mengubah data.
    MyCertif.api('apiGetUser', [kode]).then(function (r) {
      if (r && r.user && MUState.mode === 'ubah' && MUState.kode === kode) {
        MyCertif.isiForm(form, r.user);
      }
    }).catch(function () { /* biarkan data tabel sebagai cadangan */ });
  } else {
    if (judul) judul.textContent = 'Tambah user';
    if (inKode) inKode.disabled = false;
    if (hintKode) hintKode.innerHTML = 'Menjadi kunci data sertifikasi sekaligus <b>password awal</b> akun.';
    if (passWrap) passWrap.classList.add('gde-u-hide');
    MyCertif.isiForm(form, { role: 'EMP', status: 'ACTIVE' });
  }

  MyCertif.modalBuka('muModal');
  setTimeout(function () {
    try { if (inKode && !inKode.disabled) inKode.focus(); } catch (e) { }
  }, 120);
}

function simpanMu() {
  var form = MyCertif.el('muForm');
  var btn = MyCertif.el('muSimpan');
  if (!form) return;

  MyCertif.bersihkanInvalid(form);
  var data = muBersihkan(MyCertif.nilaiForm(form));
  if (MUState.mode === 'ubah') data.employee_code = MUState.kode;

  if (!data.employee_code) {
    MyCertif.tandaiInvalid(MyCertif.el('muEmployeeCode'), 'Employee Code wajib diisi.'); return;
  }
  if (!/^[A-Za-z0-9\-_/.]+$/.test(data.employee_code)) {
    MyCertif.tandaiInvalid(MyCertif.el('muEmployeeCode'),
      'Employee Code hanya boleh berisi huruf, angka, tanda hubung, garis miring, dan titik.'); return;
  }
  if (!data.nama_lengkap) {
    MyCertif.tandaiInvalid(MyCertif.el('muNama'), 'Nama lengkap wajib diisi.'); return;
  }
  if (data.email_perusahaan && !muEmailValid(data.email_perusahaan)) {
    MyCertif.tandaiInvalid(MyCertif.el('muEmailPerusahaan'), 'Format email perusahaan tidak valid (contoh: nama@geodipa.co.id).'); return;
  }
  if (data.email_pribadi && !muEmailValid(data.email_pribadi)) {
    MyCertif.tandaiInvalid(MyCertif.el('muEmailPribadi'), 'Format email pribadi tidak valid (contoh: nama@gmail.com).'); return;
  }
  if (data.email_atasan && !muEmailValid(data.email_atasan)) {
    MyCertif.tandaiInvalid(MyCertif.el('muEmailAtasan'), 'Format email atasan tidak valid (contoh: atasan@geodipa.co.id).'); return;
  }
  if (data.status !== 'INACTIVE' && String(data.status).toUpperCase() !== 'ACTIVE') {
    data.status = 'ACTIVE';
  }
  if (!data.password) delete data.password;

  MyCertif.tombolSibuk(btn, true, 'Menyimpan...');
  MyCertif.api('apiSaveUser', [data]).then(function (r) {
    MyCertif.tombolSibuk(btn, false);
    MyCertif.modalTutup('muModal');
    var hasil = r || {};
    if (hasil.mode === 'update') {
      MyCertif.sukses('Data user ' + hasil.employee_code + ' (' + (hasil.username || '') + ') berhasil diperbarui.',
        'Perubahan disimpan');
    } else {
      MyCertif.sukses('User ' + hasil.employee_code + ' dibuat dengan username "' + (hasil.username || '')
        + '". Username: ' + (hasil.username || '-') + ' · Password: ' + (hasil.password_awal || hasil.employee_code)
        + '. Sampaikan username & password tersebut ke karyawan.', 'User ditambahkan');
    }
    muatUser();
  }).catch(function (e) {
    MyCertif.tombolSibuk(btn, false);
    MyCertif.gagal(e.message);
  });
}

function muResetPassword(kode) {
  var u = muCari(kode);
  var nama = u ? (u.nama_lengkap || kode) : kode;
  MyCertif.konfirmasi({
    judul: 'Reset akun pengguna',
    pesan: 'Password akun ' + nama + ' (' + kode + ') akan dikembalikan ke Employee Code: '
      + kode + '. Passwordnya dikembalikan ke Employee Code dan seluruh sesi aktifnya dicabut.',
    labelOk: 'Reset akun',
    bahaya: true
  }).then(function (ya) {
    if (!ya) return;
    return MyCertif.api('apiResetPassword', [kode]).then(function (r) {
      var d = r || {};
      MyCertif.sukses('Akun ' + kode + ' direset: password kembali ke Employee Code ('
        + (d.password_awal || kode) + ') dan sesi lama dicabut.', 'Akun direset');
    });
  }).catch(function (e) {
    MyCertif.gagal(e.message);
  });
}

function muToggleStatus(kode) {
  var u = muCari(kode);
  if (!u) { MyCertif.gagal('Data user tidak ditemukan pada tabel. Muat ulang halaman.'); return; }
  var aktif = String(u.status || 'ACTIVE').toUpperCase() === 'ACTIVE';
  var baru = aktif ? 'INACTIVE' : 'ACTIVE';

  MyCertif.konfirmasi({
    judul: aktif ? 'Nonaktifkan akun' : 'Aktifkan akun',
    pesan: aktif
      ? 'Akun ' + (u.nama_lengkap || kode) + ' (' + kode + ') akan dinonaktifkan sehingga tidak dapat login. '
        + 'Data sertifikasinya tetap tersimpan.'
      : 'Akun ' + (u.nama_lengkap || kode) + ' (' + kode + ') akan diaktifkan kembali dan dapat login.',
    labelOk: aktif ? 'Nonaktifkan' : 'Aktifkan',
    bahaya: aktif
  }).then(function (ya) {
    if (!ya) return;
    // Ambil data terbaru dari server supaya perubahan status tidak menimpa
    // perubahan yang dilakukan HC lain sejak tabel dimuat.
    return MyCertif.api('apiGetUser', [kode]).then(function (r) {
      var segar = (r && r.user) ? r.user : u;
      return MyCertif.api('apiSaveUser', [muDataDariBaris(segar, baru)]);
    }).then(function () {
      MyCertif.sukses('Akun ' + kode + ' kini berstatus ' + baru + '.', 'Status diperbarui');
      muatUser();
    });
  }).catch(function (e) {
    MyCertif.gagal(e.message);
  });
}

function muHapus(kode) {

  var nama = u ? (u.nama_lengkap || kode) : kode;

      muatUser();
    });
  }).catch(function (e) {
    MyCertif.gagal(e.message);
  });
}

/* ------------------------------------------------------------- sinkron -- */
function muSinkron() {
  var btn = MyCertif.el('muSyncBtn');
  MyCertif.konfirmasi({
    judul: 'Sinkron user dari data sertifikasi',
    pesan: 'Sistem akan membuat akun role EMP untuk setiap Employee Code yang muncul di data '
      + 'sertifikasi tetapi belum punya akun di Master User. Akun yang sudah ada akan dilewati. '
      + 'Nama akun baru sementara memakai Employee Code dan perlu dilengkapi di halaman ini.',
    labelOk: 'Jalankan sinkron'
  }).then(function (ya) {
    if (!ya) return;
    MyCertif.tombolSibuk(btn, true, 'Menyinkronkan...');
    return MyCertif.api('apiSyncUsersFromFiles', []).then(function (r) {
      MyCertif.tombolSibuk(btn, false);
      muTampilkanHasilSinkron(r);
      muatUser();
    }).catch(function (e) {
      MyCertif.tombolSibuk(btn, false);
      MyCertif.gagal(e.message);
    });
  });
}

function muTampilkanHasilSinkron(r) {
  var d = r || {};
  var dibuat = MyCertif.num(d.dibuat);
  var dilewati = MyCertif.num(d.dilewati);
  var errors = d.errors || [];
  var daftar = d.daftar || [];

  var html = '<div class="gde-alert gde-alert--' + (errors.length ? 'warning' : 'success') + '">'
    + '<i class="bi ' + (errors.length ? 'bi-exclamation-triangle-fill' : 'bi-check-circle-fill')
    + ' gde-alert__icon"></i>'
    + '<div class="gde-alert__body"><strong>Sinkron selesai</strong>'
    + 'Akun dibuat: <b>' + MyCertif.angka(dibuat) + '</b> · dilewati (sudah ada): <b>' + MyCertif.angka(dilewati)
    + '</b> · gagal: <b>' + MyCertif.angka(errors.length) + '</b>.'
    + '</div></div>';

  if (!dibuat && !errors.length) {
    html += '<p class="mycertif-inline-note mycertif-mt-3">Semua Employee Code pada data sertifikasi sudah '
      + 'memiliki akun. Tidak ada yang perlu ditambahkan.</p>';
  }
  if (daftar.length) html += muKotakDaftar('Akun baru yang dibuat', daftar, 'info');
  if (errors.length) html += muKotakDaftar('Akun yang gagal dibuat', errors, 'danger');

  MyCertif.isiHtml('muHasilIsi', html);
  MyCertif.modalBuka('muHasil');

  if (dibuat) {
    MyCertif.sukses(MyCertif.angka(dibuat) + ' akun baru dibuat, ' + MyCertif.angka(dilewati)
      + ' dilewati. Lengkapi nama dan data pekerjaannya di tabel.', 'Sinkron selesai');
  } else {
    MyCertif.info('Tidak ada akun baru. ' + MyCertif.angka(dilewati) + ' akun sudah ada sebelumnya.', 'Sinkron selesai');
  }
}

/* --------------------------------------------------------------- impor -- */
function bukaMuImpor() {
  var teks = MyCertif.el('muImporTeks');
  var ganti = MyCertif.el('muImporGanti');
  var hasil = MyCertif.el('muImporHasil');
  if (teks) teks.value = '';
  if (ganti) ganti.checked = false;
  if (hasil) hasil.innerHTML = '';
  MyCertif.modalBuka('muImpor');
  setTimeout(function () { try { if (teks) teks.focus(); } catch (e) { } }, 120);
}

function jalanMuImpor() {
  var teks = String((MyCertif.el('muImporTeks') || {}).value || '');
  var ganti = MyCertif.el('muImporGanti');
  var btn = MyCertif.el('muImporJalan');
  var hasil = MyCertif.el('muImporHasil');

  if (!teks.trim()) {
    MyCertif.ingat('Tempelkan data user terlebih dahulu sebelum menjalankan impor.', 'Data masih kosong');
    return;
  }

  var opsi = { ganti_nama: !!(ganti && ganti.checked) };
  MyCertif.tombolSibuk(btn, true, 'Mengimpor...');
  MyCertif.api('apiImportUsers', [teks, opsi]).then(function (r) {
    MyCertif.tombolSibuk(btn, false);
    muTampilkanHasilImpor(r);
    muatUser();
  }).catch(function (e) {
    MyCertif.tombolSibuk(btn, false);
    if (hasil) {
      hasil.innerHTML = '<div class="gde-alert gde-alert--danger mycertif-mt-3">'
        + '<i class="bi bi-x-octagon-fill gde-alert__icon"></i>'
        + '<div class="gde-alert__body"><strong>Impor gagal</strong>' + MyCertif.esc(e.message) + '</div></div>';
    }
    MyCertif.gagal(e.message);
  });
}

function muTampilkanHasilImpor(r) {
  var d = r || {};
  var gagal = MyCertif.num(d.gagal);
  var pesan = d.pesan || [];

  var html = '<div class="gde-alert gde-alert--' + (gagal ? 'warning' : 'success') + ' mycertif-mt-3">'
    + '<i class="bi ' + (gagal ? 'bi-exclamation-triangle-fill' : 'bi-check-circle-fill')
    + ' gde-alert__icon"></i>'
    + '<div class="gde-alert__body"><strong>Hasil impor</strong>'
    + 'Dibuat: <b>' + MyCertif.angka(d.dibuat) + '</b> · diperbarui: <b>' + MyCertif.angka(d.diperbarui)
    + '</b> · gagal: <b>' + MyCertif.angka(gagal) + '</b>.'
    + '</div></div>';

  if (pesan.length) html += muKotakDaftar('Catatan baris', pesan, gagal ? 'danger' : 'info');

  MyCertif.isiHtml('muImporHasil', html);

  if (gagal) {
    MyCertif.ingat('Impor selesai dengan ' + MyCertif.angka(gagal)
      + ' baris gagal. Periksa catatan di dalam jendela impor.', 'Impor sebagian berhasil');
  } else {
    MyCertif.sukses('Impor selesai: ' + MyCertif.angka(d.dibuat) + ' dibuat, '
      + MyCertif.angka(d.diperbarui) + ' diperbarui.', 'Impor berhasil');
  }
}

function muKotakDaftar(judul, arr, tone) {
  var warna = tone === 'danger' ? 'var(--gde-red)' : 'var(--gde-blue)';
  var isi = (arr && arr.length)
    ? '<ul style="margin:0;padding-left:1.05rem">' + arr.map(function (x) {
      return '<li style="font-size:.78rem;color:var(--gde-ink-2);margin-bottom:2px">' + MyCertif.esc(x) + '</li>';
    }).join('') + '</ul>'
    : '<span class="mycertif-inline-note">—</span>';

  return '<div class="mycertif-mt-3">'
    + '<div class="gde-u-between" style="margin-bottom:4px">'
    + '<span class="mycertif-inline-note">' + MyCertif.esc(judul) + '</span>'
    + '<b style="font-size:.78rem;color:' + warna + '">' + MyCertif.angka((arr || []).length) + '</b>'
    + '</div>'
    + '<div class="mycertif-scroll-y" style="border:1px solid var(--gde-line);border-radius:var(--gde-r-sm);'
    + 'padding:.6rem .85rem;background:#fff">' + isi + '</div>'
    + '</div>';
}
