  function qs(sel, root) { return (root || document).querySelector(sel); }
  function qsa(sel, root) { return Array.prototype.slice.call((root || document).querySelectorAll(sel)); }
  function el(id) { return document.getElementById(id); }

  function esc(s) {
    return String(s === null || s === undefined ? '' : s)
      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
  }

  function attr(s) { return esc(s); }

  function on(root, event, selector, handler) {
    (root || document).addEventListener(event, function (e) {
      var t = e.target.closest(selector);
      if (t && (root || document).contains(t)) handler.call(t, e, t);
    });
  }

  function debounce(fn, wait) {
    var t;
    return function () {
      var ctx = this, args = arguments;
      clearTimeout(t);
      t = setTimeout(function () { fn.apply(ctx, args); }, wait || 320);
    };
  }

  function num(v, def) {
    var n = Number(v);
    return isNaN(n) ? (def === undefined ? 0 : def) : n;
  }

  /* ========================================================================

  var BULAN = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli',
    'Agustus', 'September', 'Oktober', 'November', 'Desember'];
  var BULAN_SINGKAT = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];
  var HARI = ['Minggu', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];

  function toDate(v) {
    if (!v) return null;
    if (v instanceof Date) return isNaN(v.getTime()) ? null : v;
    var s = String(v).trim();
    var m = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
    if (m) return new Date(+m[1], +m[2] - 1, +m[3]);
    var d = new Date(s);
    return isNaN(d.getTime()) ? null : d;
  }

  /**
   * Normalisasi menjadi Date DENGAN jam-menit-detik.
   *
   * BERBEDA dari toDate(), yang sengaja memotong ke tengah malam karena dipakai
   * untuk kolom tanggal (Expired_at, Start_pelaksanaan). Kolom waktu seperti
   * Updated_At / Login_Terakhir / Created_At butuh jamnya — memakai toDate()
   * membuat semua waktu tampak "00:00" dan "x jam lalu" jadi salah hitung.
   */
  function toDateTime(v) {
    if (v === null || v === undefined || v === '') return null;
    if (v instanceof Date) return isNaN(v.getTime()) ? null : v;
    var s = String(v).trim();
    if (!s) return null;
    var m = s.match(/^(\d{4})-(\d{2})-(\d{2})[ T](\d{1,2}):(\d{2})(?::(\d{2}))?/);
    if (m) return new Date(+m[1], +m[2] - 1, +m[3], +m[4], +m[5], +(m[6] || 0));
    // string Date mentah (mis. "Thu Oct 08 2026 15:20:32 GMT+0700")
    var d = new Date(s);
    return isNaN(d.getTime()) ? null : d;
  }

  function tglIndo(v, denganHari) {
    var d = toDate(v);
    if (!d) return '—';
    var teks = d.getDate() + ' ' + BULAN[d.getMonth()] + ' ' + d.getFullYear();
    return denganHari ? (HARI[d.getDay()] + ', ' + teks) : teks;
  }

  function tglSingkat(v) {
    var d = toDate(v);
    if (!d) return '—';
    return d.getDate() + ' ' + BULAN_SINGKAT[d.getMonth()] + ' ' + String(d.getFullYear()).substring(2);
  }

  /**
   * Tanggal + jam, mis. "8 Oktober 2026, 15:20".
   * Dipakai untuk kolom waktu (Terakhir diperbarui, Login terakhir, dll.).
   *
   * Backend mengirim 'yyyy-MM-dd HH:mm:ss'. Bila nilainya ternyata string Date
   * mentah (mis. "Thu Oct 08 2026 15:20:32 GMT+0700"), toDate() tetap bisa
   * membacanya — sehingga tampilan tidak pernah menampilkan format mentah itu.
   */
  function waktuIndo(v) {
    var d = toDateTime(v);
    if (!d) return '—';
    return d.getDate() + ' ' + BULAN[d.getMonth()] + ' ' + d.getFullYear()
      + ', ' + pad(d.getHours()) + ':' + pad(d.getMinutes());
  }

  function iso(v) {
    var d = toDate(v);
    if (!d) return '';
    return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate());
  }

  function pad(n) { return (n < 10 ? '0' : '') + n; }

  function waktuLalu(v) {
    var d = toDateTime(v);
    if (!d) return '—';
    var detik = Math.floor((Date.now() - d.getTime()) / 1000);
    if (detik < 60) return 'baru saja';
    if (detik < 3600) return Math.floor(detik / 60) + ' menit lalu';
    if (detik < 86400) return Math.floor(detik / 3600) + ' jam lalu';
    if (detik < 604800) return Math.floor(detik / 86400) + ' hari lalu';
    return tglSingkat(d);
  }

  function angka(n) {
    var v = num(n, 0);
    return v.toString().replace(/\B(?=(\d{3})+(?!\d))/g, '.');
  }

  function ukuran(bytes) {
    var b = num(bytes, 0);
    if (b < 1024) return b + ' B';
    if (b < 1048576) return (b / 1024).toFixed(1) + ' KB';
    return (b / 1048576).toFixed(2) + ' MB';
  }

  function inisial(nama) {
    var b = String(nama || '').trim().split(/\s+/).filter(function (x) { return x; });
    if (!b.length) return '?';
    if (b.length === 1) return b[0].substring(0, 2).toUpperCase();
    return (b[0].charAt(0) + b[1].charAt(0)).toUpperCase();
  }

  function statusBadge(key, label, tone, icon) {
    var t = tone || statusTone(key);
    var l = label || statusLabel(key);
    var i = icon || statusIcon(key);
    return '<span class="gde-status gde-status--' + attr(t) + '">'

  function statusTone(key) {
    var def = cariStatus(key);
    return def ? def.tone : 'gray';

  function statusLabel(key) {
    var def = cariStatus(key);

  function statusIcon(key) {

  function cariStatus(key) {
    var list = (state.meta && state.meta.statuses) || [];
    for (var i = 0; i < list.length; i++) { if (list[i].key === key) return list[i]; }
    return null;
  }

  /**
   * Definisi verifikasi dari meta server (PENDING / VERIFIED / REVISI).

  function verifikasiDef(kode) {
    var list = (state.meta && state.meta.verifikasi_defs) || [];
    var s = String(kode || 'PENDING').trim().toUpperCase();
    // istilah lama 'REJECTED' tetap dikenali supaya data lama tidak tampil aneh
    if (s === 'REJECTED' || s === 'DITOLAK') s = 'REVISI';
    for (var i = 0; i < list.length; i++) { if (list[i].key === s) return list[i]; }
    return { key: s, label: s || 'Menunggu Verifikasi', short: s || 'Menunggu', tone: 'gray', icon: 'bi-dash-circle', ket: '' };
  }

  function verifikasiLabel(kode) { return verifikasiDef(kode).label; }

  }

  function sisaTeks(hari, bulan) {
    if (hari === null || hari === undefined || hari === '') return '—';
    var h = num(hari);
    if (h < 0) return 'Lewat ' + angka(Math.abs(h)) + ' hari';
    return angka(h) + ' hari';
  }

  function sisaBulanTeks(bulan) {
    if (bulan === null || bulan === undefined || bulan === '') return '';
    return num(bulan).toLocaleString('id-ID', { maximumFractionDigits: 1 }) + ' bln';
  }

  /**
   * Sisa masa berlaku berjenjang tahun/bulan/hari, mis. "3 Tahun 2 Bulan 4 Hari".
   *
   * Sumber utamanya `r.sisa_teks` dari backend. Bila kosong, nilainya dihitung
   * ulang dari `r.expired_at` memakai algoritma kalender yang SAMA dengan
   * backend (durasiKalender), sehingga tampilan tidak pernah berbeda tergantung
   * jalur mana yang dipakai. `r.sisa_hari` hanya dipakai sebagai upaya terakhir.
   */
  function masaBerlakuTeks(r) {
    if (!r) return '—';
    if (r.sisa_teks) return r.sisa_teks;
    if (r.expired_at) {
      var kal = durasiKalender(r.expired_at);
      if (kal) return kal;
    }
    if (r.sisa_hari !== null && r.sisa_hari !== undefined && r.sisa_hari !== '') {
      return durasiDariHari(r.sisa_hari);
    }
    return '—';
  }

  /**
   * Selisih dua tanggal sebagai "x Tahun y Bulan z Hari".
   * Meniru durasiTeks_() di Backend/subprocess.gs supaya hasil backend dan
   * frontend tidak pernah berbeda. Tanpa `ref`, acuannya hari ini.
   */
  function durasiKalender(tanggalAkhir, ref) {
    var akhir = toDate(tanggalAkhir);
    if (!akhir) return '';
    var acuan = ref ? toDate(ref) : (function () {
      var t = new Date();
      return new Date(t.getFullYear(), t.getMonth(), t.getDate());
    })();
    if (!acuan) return '';

    var lewat = akhir.getTime() < acuan.getTime();
    var a = lewat ? akhir : acuan;
    var b = lewat ? acuan : akhir;

    var tahun = b.getFullYear() - a.getFullYear();
    var bulan = b.getMonth() - a.getMonth();
    var hari = b.getDate() - a.getDate();

    if (hari < 0) {
      bulan -= 1;
      hari += new Date(b.getFullYear(), b.getMonth(), 0).getDate();
    }
    if (bulan < 0) {
      tahun -= 1;
      bulan += 12;
    }

    var bagian = [];
    if (tahun > 0) bagian.push(tahun + ' Tahun');
    if (bulan > 0) bagian.push(bulan + ' Bulan');
    if (hari > 0 || !bagian.length) bagian.push(hari + ' Hari');
    return (lewat ? 'Lewat ' : '') + bagian.join(' ');
  }

  /**
   * Perkiraan kasar dari jumlah hari -> "x Tahun y Bulan z Hari".
   * Hanya dipakai bila `expired_at` tidak ada sama sekali; hasilnya bisa
   * berbeda 1–5 hari dari durasiKalender karena bulan punya jumlah hari berbeda.
   */
  function durasiDariHari(hari) {
    var h = num(hari);
    var lewat = h < 0;
    var sisa = Math.abs(h);
    var tahun = Math.floor(sisa / 365);
    sisa -= tahun * 365;
    var bulan = Math.floor(sisa / 30);
    sisa -= bulan * 30;
    var bagian = [];
    if (tahun) bagian.push(tahun + ' Tahun');
    if (bulan) bagian.push(bulan + ' Bulan');
    if (sisa || !bagian.length) bagian.push(sisa + ' Hari');
    return (lewat ? 'Lewat ' : '') + bagian.join(' ');
  }

  function toast(msg, tipe, judul) {
    var t = tipe || 'info';
    var ikon = { success: 'bi-check-circle-fill', danger: 'bi-x-circle-fill', warning: 'bi-exclamation-triangle-fill', info: 'bi-info-circle-fill' }[t] || 'bi-info-circle-fill';
    var div = document.createElement('div');
    div.className = 'mycertif-toast mycertif-toast--' + t;
    div.innerHTML = '<i class="bi ' + ikon + ' mycertif-toast__icon"></i>'
      + '<div class="mycertif-toast__body">'
      + (judul ? '<strong class="mycertif-toast__title">' + esc(judul) + '</strong>' : '')
      + '<div class="mycertif-toast__msg">' + esc(msg) + '</div></div>'
      + '<button class="mycertif-toast__close" type="button" aria-label="Tutup">&times;</button>';
    toastContainer().appendChild(div);
    var tutup = function () { if (div.parentNode) div.parentNode.removeChild(div); };
    div.querySelector('.mycertif-toast__close').addEventListener('click', tutup);
    setTimeout(tutup, t === 'danger' ? 9000 : 5200);
  }

  var sukses = function (m, j) { toast(m, 'success', j); };
  var gagal = function (m, j) { toast(m, 'danger', j); };
  var info = function (m, j) { toast(m, 'info', j); };
  var ingat = function (m, j) { toast(m, 'warning', j); };

  function modalBuka(id) {

  function modalTutupSemua() {
    qsa('.gde-modal.is-open').forEach(function (m) { m.classList.remove('is-open'); });
    document.body.style.overflow = '';
  }

  function konfirmasi(opsi) {
    return new Promise(function (resolve) {
      var o = opsi || {};
      var id = 'mycertifConfirm';
      var lama = el(id);
      if (lama && lama.parentNode) lama.parentNode.removeChild(lama);

      var wrap = document.createElement('div');
      wrap.className = 'gde-modal is-open';
      wrap.id = id;
      wrap.innerHTML = ''
        + '<div class="gde-modal__scrim" data-tutup></div>'
        + '<div class="gde-modal__dialog" style="max-width:480px">'
        + '  <div class="gde-modal__head"><h3>' + esc(o.judul || 'Konfirmasi') + '</h3></div>'
        + '  <div class="gde-modal__body"><p style="margin:0;color:var(--gde-ink-2)">' + (o.pesanHtml || esc(o.pesan || '')) + '</p></div>'
        + '  <div class="gde-modal__foot">'
        + '    <button class="gde-btn gde-btn--outline" data-batal type="button">' + esc(o.labelBatal || 'Batal') + '</button>'
        + '    <button class="gde-btn ' + (o.bahaya ? 'gde-btn--danger' : 'gde-btn--primary') + '" data-ok type="button">'

        + '  </div>'
        + '</div>';
      document.body.appendChild(wrap);
      document.body.style.overflow = 'hidden';

      var selesai = function (hasil) {
        if (wrap.parentNode) wrap.parentNode.removeChild(wrap);
        document.body.style.overflow = '';
        resolve(hasil);
      };
      wrap.querySelector('[data-ok]').addEventListener('click', function () { selesai(true); });
      wrap.querySelector('[data-batal]').addEventListener('click', function () { selesai(false); });
      wrap.querySelector('[data-tutup]').addEventListener('click', function () { selesai(false); });
    });
  }

  /**
   * Dialog dengan isian (bukan sekadar ya/tidak).
   *
   * @param {Object} opsi {
   *   judul, pesanHtml,
   *   fields: [{ name, label, tipe:'text'|'textarea'|'select'|'number',
   *              nilai, placeholder, hint, wajib, opsi:[{value,label}] }],
   *   labelOk, labelBatal, bahaya,
   *   validasi: function(nilai) -> string pesan error ('' bila valid)
   * }
   * @return {Promise<Object|null>} nilai field bila disetujui, null bila dibatalkan.
   */
  function dialogForm(opsi) {
    return new Promise(function (resolve) {
      var o = opsi || {};
      var id = 'mycertifFormDialog';
      var lama = el(id);
      if (lama && lama.parentNode) lama.parentNode.removeChild(lama);

      var isiField = (o.fields || []).map(function (f) {
        var fid = id + '_' + f.name;
        var kontrol;
        if (f.tipe === 'textarea') {
          kontrol = '<textarea class="gde-textarea" id="' + fid + '" rows="' + (f.rows || 3) + '" '
            + 'placeholder="' + attr(f.placeholder || '') + '">' + esc(f.nilai || '') + '</textarea>';
        } else if (f.tipe === 'select') {
          kontrol = '<select class="gde-select" id="' + fid + '">'
            + (f.opsi || []).map(function (op) {
              return '<option value="' + attr(op.value) + '"'
                + (String(op.value) === String(f.nilai) ? ' selected' : '') + '>'
                + esc(op.label) + '</option>';
            }).join('') + '</select>';
        } else {
          kontrol = '<input class="gde-input" id="' + fid + '" type="' + attr(f.tipe || 'text') + '" '
            + 'value="' + attr(f.nilai === undefined || f.nilai === null ? '' : f.nilai) + '" '
            + 'placeholder="' + attr(f.placeholder || '') + '">';
        }
        return '<div class="mycertif-mb-3">'
          + '<label class="gde-label" for="' + fid + '">' + esc(f.label || f.name)
          + (f.wajib ? ' <span class="gde-req">*</span>' : '') + '</label>'
          + kontrol
          + (f.hint ? '<div class="gde-hint">' + esc(f.hint) + '</div>' : '')
          + '</div>';
      }).join('');

      var wrap = document.createElement('div');
      wrap.className = 'gde-modal is-open';
      wrap.id = id;
      wrap.innerHTML = ''
        + '<div class="gde-modal__scrim" data-tutup></div>'
        + '<div class="gde-modal__dialog" style="max-width:520px">'
        + '  <div class="gde-modal__head"><h3>' + esc(o.judul || 'Formulir') + '</h3></div>'
        + '  <div class="gde-modal__body">'
        + (o.pesanHtml ? ('<div style="margin-bottom:14px;color:var(--gde-ink-2)">' + o.pesanHtml + '</div>') : '')
        + isiField
        + '    <div class="gde-hint gde-u-hide" data-galat style="color:var(--gde-red)"></div>'
        + '  </div>'
        + '  <div class="gde-modal__foot">'
        + '    <button class="gde-btn gde-btn--outline" data-batal type="button">'
        + esc(o.labelBatal || 'Batal') + '</button>'
        + '    <button class="gde-btn ' + (o.bahaya ? 'gde-btn--danger' : 'gde-btn--primary')
        + '" data-ok type="button">' + esc(o.labelOk || 'Simpan') + '</button>'
        + '  </div>'
        + '</div>';
      document.body.appendChild(wrap);
      document.body.style.overflow = 'hidden';

      var selesai = function (hasil) {
        if (wrap.parentNode) wrap.parentNode.removeChild(wrap);
        document.body.style.overflow = '';
        resolve(hasil);
      };

      var nilai = function () {
        var out = {};
        (o.fields || []).forEach(function (f) {
          var node = el(id + '_' + f.name);
          out[f.name] = node ? String(node.value || '').trim() : '';
        });
        return out;
      };

      var galat = wrap.querySelector('[data-galat]');
      var tampilGalat = function (pesan) {
        if (!galat) return;
        galat.textContent = pesan || '';
        galat.classList.toggle('gde-u-hide', !pesan);
      };

      wrap.querySelector('[data-ok]').addEventListener('click', function () {
        var v = nilai();
        var kosong = (o.fields || []).filter(function (f) { return f.wajib && !v[f.name]; });
        if (kosong.length) {
          tampilGalat('Wajib diisi: ' + kosong.map(function (f) { return f.label || f.name; }).join(', ') + '.');
          return;
        }
        if (typeof o.validasi === 'function') {
          var pesan = o.validasi(v);
          if (pesan) { tampilGalat(pesan); return; }
        }
        selesai(v);
      });
      wrap.querySelector('[data-batal]').addEventListener('click', function () { selesai(null); });
      wrap.querySelector('[data-tutup]').addEventListener('click', function () { selesai(null); });

      var fokus = wrap.querySelector('.gde-input, .gde-textarea, .gde-select');
      if (fokus) { try { fokus.focus(); } catch (e) { } }
    });
  }

  /* ========================================================================
     5. PEMANGGIL API
     ======================================================================== */
  function gsr(fn) {
    var args = Array.prototype.slice.call(arguments, 1);
    return new Promise(function (resolve, reject) {
      var runner = google.script.run
        .withSuccessHandler(function (res) { resolve(res); })
        .withFailureHandler(function (err) { reject(err); });
      runner[fn].apply(runner, args);
    });
  }

  /**
   * Panggil endpoint apiXxx dan buka bungkusan { ok, data } / { ok, error }.
   * Argumen token disisipkan otomatis sebagai parameter pertama bila
   * `opts.token !== false`.
   */
  function api(fn, args, opts) {
    var o = opts || {};
    var a = (args || []).slice();
    if (o.token !== false) a.unshift(state.token);
    var seq = state.viewSeq;

    return gsr.apply(null, [fn].concat(a)).then(function (res) {
      /*
       * PENTING — mencegah error "Cannot set properties of null".
       * Bila pengguna berpindah halaman sebelum permintaan server selesai,
       * seluruh isi halaman lama sudah dibuang. Melanjutkan rantai .then()
       * akan menulis ke elemen yang sudah tidak ada. Karena itu hasilnya
       * dibuang begitu saja (promise tidak diselesaikan) sehingga .then()
       * milik halaman lama tidak pernah dijalankan.
       *
       * Konsekuensi yang disengaja: bila pengguna pindah halaman di tengah
       * proses simpan/hapus, prosesnya TETAP berjalan di server (audit trail
       * mencatatnya) hanya saja notifikasinya tidak ditampilkan.
       * Opt-out: panggil dengan { tetap: true } bila memang ingin lanjut.
       */
      if (o.tetap !== true && seq !== state.viewSeq) {
        return new Promise(function () { /* sengaja tidak diselesaikan */ });
      }

      if (!res || typeof res !== 'object') throw new Error('Respons server tidak dikenal.');
      if (res.ok === false) {
        var e = new Error(res.error || 'Terjadi kesalahan pada server.');
        e.server = true;
        throw e;
      }
      return res.data === undefined ? null : res.data;
    }).catch(function (err) {
      /*
       * Penjaga yang sama untuk jalur GAGAL. Tanpa ini, respons error milik
       * halaman lama tetap muncul sebagai toast di halaman baru — dan
       * penanganan sesi bisa memicu tampilan login di halaman yang salah.
       */
      if (o.tetap !== true && seq !== state.viewSeq) {
        return new Promise(function () { /* sengaja tidak diselesaikan */ });
      }

      var msg = (err && err.message) ? err.message : String(err);
      if (/Sesi|login|AUTH/i.test(msg) && o.silent !== true) {
        tanganiSesiHabis_(msg);
      }
      state.lastError = msg;
      throw new Error(msg);
    });
  }

  var perluLoginUlang = false;
  /**
   * Sesi berakhir. TIDAK memuat ulang halaman (dulu location.reload() membuat
   * tampilan "loncat"); cukup tampilkan kembali halaman masuk di tempat.
   */
  function tanganiSesiHabis_(msg) {
    if (perluLoginUlang) return;
    perluLoginUlang = true;
    gagal(msg || 'Sesi berakhir. Silakan masuk kembali.', 'Sesi berakhir');
    hapusSesi();
    tampilkanLogin(msg || '');
  }

  /* ========================================================================
     6. SESI  (penyimpanan BERLAPIS agar tidak mudah hilang)
     ------------------------------------------------------------------------
     Aplikasi Apps Script tampil di dalam iframe pada domain googleusercontent,
     sehingga localStorage dapat diblokir browser dan parameter URL hilang saat
     halaman di-refresh. Karena itu token disimpan berlapis:
       1. login otomatis lewat email akun Google (paling andal, tanpa storage)
       2. localStorage
       3. parameter pada hash URL   (#/view!t=TOKEN)
       4. memori saja (selama tab terbuka)
     ======================================================================== */
  function simpanSesi(token, user) {
    state.token = token;
    state.user = user;
    try {
      localStorage.setItem(TOKEN_KEY, token);
      localStorage.setItem(USER_KEY, JSON.stringify(user || {}));
    } catch (e) { /* diblokir: lapis lain tetap jalan */ }
    perbaruiTokenDiHash();
  }

  function hapusSesi() {
    state.token = '';
    state.user = null;
    try {
      localStorage.removeItem(TOKEN_KEY); localStorage.removeItem(USER_KEY);
      localStorage.removeItem(TOKEN_KEY_LAMA);
    } catch (e) { }
    perbaruiTokenDiHash();
  }

  function tokenTersimpan() {
    // lapis 2: localStorage
    try {
      var t = localStorage.getItem(TOKEN_KEY) || '';
      if (t) return t;
      // kompatibilitas: sesi yang tersimpan sebelum aplikasi berganti nama
      var lama = localStorage.getItem(TOKEN_KEY_LAMA) || '';
      if (lama) {
        try { localStorage.setItem(TOKEN_KEY, lama); localStorage.removeItem(TOKEN_KEY_LAMA); } catch (e2) { }
        return lama;
      }
    } catch (e) { }
    // lapis 3: hash URL
    return tokenDariHash();
  }

  /** Token yang tertulis pada hash URL: #/dashboard!t=XXXX */
  function tokenDariHash() {
    try {
      var m = String(location.hash || '').match(/!t=([A-Za-z0-9_-]+)/);
      return m ? m[1] : '';
    } catch (e) { return ''; }
  }

  /** Tulis ulang hash tanpa mengubah halaman yang sedang dibuka. */
  function perbaruiTokenDiHash() {
    try {
      var hash = String(location.hash || '');
      hash = hash.replace(/!t=[A-Za-z0-9_-]+/g, '');
      if (state.token) {
        hash = (hash || ('#/' + (state.view || 'dashboard'))) + '!t=' + state.token;
      }
      if (hash !== location.hash) location.hash = hash;
    } catch (e) { }
  }

  function pengguna() { return state.user || {}; }
  function role() { return (state.user && state.user.role) || ''; }
  function isHc() { return role() === 'HC' || role() === 'SA'; }
  function isSa() { return role() === 'SA'; }
  function namaSaya() { return (state.user && state.user.nama_lengkap) || ''; }

  /**
   * Masuk dengan Employee Code.
   * Bila server menjawab `butuh_password`, hasilnya diteruskan apa adanya
   * supaya halaman login bisa menampilkan kolom password.
   */
  function login(identitas, password) {
    return api('apiLogin', [identitas, password], { token: false }).then(function (data) {
      if (data && data.butuh_password) return data;   // perlu satu langkah lagi
      simpanSesi(data.token, data.user);
      perluLoginUlang = false;
      return data;
    });
  }

  function logout() {
    var t = state.token;
    var lanjut = function () {
      hapusSesi();
      perluLoginUlang = false;
      tampilkanLogin('');
    };
    if (!t) { lanjut(); return Promise.resolve(); }
    // logoutToken: pakai token eksplisit walau state sudah dibersihkan
    return gsr('apiLogout', t).then(lanjut, lanjut);
  }

  /**
   * Muat sesi saat aplikasi dibuka. Urutan percobaan:
   *   token tersimpan -> validasi -> kalau gagal, coba login otomatis via email.
   * @return {Promise<boolean>} true bila berhasil masuk
   */
  function bootSesi() {
    var t = tokenTersimpan();
    if (!t) return bootSesiOtomatis();

    state.token = t;
    return api('apiGetSession', []).then(function (data) {
      state.user = data.user;
      try { localStorage.setItem(USER_KEY, JSON.stringify(data.user)); } catch (e) { }
      perbaruiTokenDiHash();
      return true;
    }).catch(function () {
      state.token = '';
      try { localStorage.removeItem(TOKEN_KEY); localStorage.removeItem(TOKEN_KEY_LAMA); } catch (e) { }
      return bootSesiOtomatis();
    });
  }

  /** Lapis 1: login otomatis lewat email akun Google. */
  function bootSesiOtomatis() {
    return api('apiGetSessionAuto', [], { token: false, silent: true }).then(function (d) {
      if (!d || !d.auto) {
        state.autoLoginInfo = (d && d.alasan) ? d.alasan : '';
        return false;
      }
      simpanSesi(d.token, d.user);
      perluLoginUlang = false;
      state.autoLoginInfo = 'Masuk otomatis sebagai ' + (d.email || '') + '.';
      return true;
    }).catch(function () {
      return false;
    });
  }

  /** Coba lagi login otomatis (dipakai halaman login). */
  function cobaLoginOtomatis() { return bootSesiOtomatis(); }

    if (state.meta && !paksa) return Promise.resolve(state.meta);

      state.meta = m;

    });

  function navigasi(namaView, opsi) {
    var v = VIEWS[namaView];
    if (!v) { gagal('Halaman "' + namaView + '" tidak dikenal.'); return; }
    if (!bolehAkses(namaView)) { gagal('Anda tidak punya akses ke halaman ini.'); return; }
    var o = opsi || {};
    state.view = namaView;
    state.param = o.param || '';
    if (o.hash !== false) {
      // token tetap dibawa pada hash supaya tidak hilang saat berpindah halaman
      try {
        var h = '#/' + namaView + (o.param ? '?' + o.param : '');
        if (state.token) h += '!t=' + state.token;
        if (location.hash !== h) location.hash = h;
      } catch (e) { }
    }
    muatView(namaView);
  }

  function muatView(namaView) {
    var v = VIEWS[namaView];
    if (!v) return;
    var konten = el('mycertifContent');
    if (!konten) return;

    // tandai menu aktif
    qsa('[data-view]').forEach(function (a) {
      a.classList.toggle('is-active', a.getAttribute('data-view') === namaView);
    });

    // judul halaman di topbar
    var jt = el('mycertifPageTitle');
    if (jt) jt.textContent = v.judul;
    var js = el('mycertifPageSub');
    if (js) js.textContent = v.sub || '';

    // nomor urut: supaya jawaban permintaan LAMA tidak menimpa halaman baru
    var seq = ++state.viewSeq;

    // tampilkan isi segera; kalau HTML halaman sudah pernah diambil, pakai
    // cache supaya perpindahan halaman terasa mulus (tanpa kedipan).
    terapkanView_(konten, namaView, v, state.viewCache[namaView] || '');

    if (state.viewCache[namaView]) return;

    konten.innerHTML = pemuatHtml('Memuat ' + v.judul + '...');
    gsr('getView', v.file).then(function (html) {
      if (seq !== state.viewSeq) return;     // sudah pindah halaman lagi
      if (!html) throw new Error('Halaman ' + v.file + ' tidak ditemukan.');
      state.viewCache[namaView] = html;      // cache: view bersifat statis
      terapkanView_(konten, namaView, v, html);
    }).catch(function (err) {
      if (seq !== state.viewSeq) return;
      konten.innerHTML = galatHtml(err.message || String(err));
    });
  }

  /** Pasang HTML view + jalankan initXxx() milik view tersebut. */
  function terapkanView_(konten, namaView, v, html) {
    if (!html) return;
    konten.innerHTML = html;
    jalankanSkrip(konten);          // aktifkan <script> milik view
    var fn = window[v.init];
    if (typeof fn === 'function') {
      try {
        fn(state.param);
      } catch (e) {
        konten.innerHTML = galatHtml(e.message || String(e));
      }
    } else {
      konten.innerHTML = galatHtml('Fungsi ' + v.init + '() belum tersedia.');
    }
    perbaruiBadgeHeader();
  }

  /**
   * Tulis HTML ke elemen dengan AMAN.
   * Kalau elemennya sudah tidak ada (mis. pengguna sudah pindah halaman),
   * penulisan dilewati tanpa melempar error.
   * @return {boolean} true bila berhasil ditulis
   */
  function isiHtml(id, html) {
    var n = el(id);
    if (!n) {
      // Biasanya karena halaman sudah berganti -> wajar, cukup dicatat.
      // Kalau muncul saat halaman TIDAK berpindah, berarti ada salah ketik id.
      try {
        console.warn('[myCertif] #' + id + ' tidak ada — penulisan dilewati '
          + '(wajar bila halaman sudah berganti).');
      } catch (e) { }
      return false;
    }
    n.innerHTML = html;
    return true;
  }

  /** Sama seperti isiHtml(), untuk textContent. */
  function isiTeks(id, teks) {
    var n = el(id);
    if (!n) {
      try {
        console.warn('[myCertif] #' + id + ' tidak ada — penulisan dilewati '
          + '(wajar bila halaman sudah berganti).');
      } catch (e) { }
      return false;
    }
    n.textContent = teks === null || teks === undefined ? '' : teks;
    return true;
  }

  /**
   * PENTING: skrip yang disuntikkan lewat innerHTML TIDAK dijalankan browser.
   * Fungsi ini membuat ulang setiap <script> di dalam `root` supaya
   * initXxx() milik view benar-benar terdefinisi.
   */
  function jalankanSkrip(root) {
    if (!root) return;
    qsa('script', root).forEach(function (lama) {
      var baru = document.createElement('script');
      baru.type = 'text/javascript';
      if (lama.src) { baru.src = lama.src; }
      else { baru.textContent = lama.textContent; }
      lama.parentNode.replaceChild(baru, lama);
    });
  }

  function combobox(opsi) {
    var o = opsi || {};
    var input = o.input;
    var panel = o.panel;
    if (!input || !panel) return null;

    pastikanCssCombobox_();

    // Pembungkus wajib ada agar panel bisa diposisikan relatif terhadap input.
    // Bila markup tidak menyediakannya, dibuat otomatis (lebih tahan salah tempel).
    var wrap = input.closest('.mycertif-cb') || input.parentNode;
    if (!wrap || wrap === document.body) {
      wrap = document.createElement('div');
      wrap.className = 'mycertif-cb';
      if (input.parentNode) input.parentNode.insertBefore(wrap, input);
      wrap.appendChild(input);
    }

    var items = o.items || [];
    var tersaring = items.slice();
    var aktif = -1;
    var terpilih = null;
    var onPilih = o.onPilih || function () { };

    if (panel.parentNode !== wrap) {
      try { wrap.appendChild(panel); } catch (ePanel) { }
    }

    input.setAttribute('autocomplete', 'off');
    if (o.placeholder) input.setAttribute('placeholder', o.placeholder);
    try {
      var idIkon = (panel.id || 'mycertifCb') + '_ikon';
      if (!document.getElementById(idIkon)) {
        var ikon = document.createElement('i');
        ikon.className = 'bi bi-search mycertif-cb__icon';
        ikon.id = idIkon;
        wrap.appendChild(ikon);
      }
    } catch (eIkon) { /* ikon hanya hiasan: jangan sampai menghentikan komponen */ }

    // PENGAMAN: panel dikendalikan juga lewat inline style, sehingga tetap
    // benar walau CSS (Stylesheet.html) belum ikut diperbarui.
    panel.style.display = 'none';

    function buka() {
      if (wrap) wrap.classList.add('is-open');
      panel.style.display = 'block';
    }
    function tutup() {
      if (wrap) wrap.classList.remove('is-open');
      panel.style.display = 'none';
      aktif = -1;
    }

    function labelItem(it) {
      var tone = it.tone ? (' gde-status--' + it.tone) : '';
      // Badge hanya ditampilkan bila memang informatif (mis. HC / Super Admin).
      var badge = String(it.badge || '');
      if (/^(employee|emp)$/i.test(badge)) badge = '';
      return '<div class="mycertif-cb__item' + (terpilih && terpilih.value === it.value ? ' is-dipilih' : '')
        + '" data-cb-val="' + esc(it.value) + '" '
        + 'style="display:flex;align-items:center;gap:.75rem;padding:.5rem .65rem;cursor:pointer">'
        + '<div class="mycertif-cb__meta" style="min-width:0;flex:1">'
        + '<span class="mycertif-cb__nama" style="display:block;font-weight:600">'
        + esc(it.label || it.value) + '</span>'
        + (it.sub ? '<span class="mycertif-cb__sub" style="display:block;font-size:.7rem;opacity:.75">'
          + esc(it.sub) + '</span>' : '')
        + '</div>'
        + (badge ? '<span class="gde-status' + tone + '">' + esc(badge) + '</span>' : '')
        + '</div>';
    }

    function render() {
      var q = String(input.value || '').trim().toLowerCase();
      tersaring = items.filter(function (it) {
        if (!q) return true;
        var gab = [it.label, it.sub, it.value].join(' ').toLowerCase();
        return gab.indexOf(q) !== -1;
      });
      if (!tersaring.length) {
        panel.innerHTML = '<div class="mycertif-cb__info">Tidak ada yang cocok dengan "' + esc(q) + '".</div>';
        return;
      }
      var potong = tersaring.slice(0, 60);
      var kepala = q ? ('<div class="mycertif-cb__hitung">' + potong.length + ' dari ' + tersaring.length + ' hasil</div>') : '';
      panel.innerHTML = kepala + potong.map(labelItem).join('')
        + (tersaring.length > potong.length
          ? '<div class="mycertif-cb__info">Persempit pencarian untuk melihat sisanya.</div>' : '');
    }

    function sorot(idx) {
      var nodes = qsa('.mycertif-cb__item', panel);
      if (!nodes.length) return;
      if (idx < 0) idx = 0;
      if (idx >= nodes.length) idx = nodes.length - 1;
      nodes.forEach(function (n) { n.classList.remove('is-active'); });
      nodes[idx].classList.add('is-active');
      aktif = idx;
      try { nodes[idx].scrollIntoView({ block: 'nearest' }); } catch (e) { }
    }

    function pilih(it) {
      if (!it) return;
      terpilih = it;
      input.value = it.label || it.value;
      tutup();
      try { onPilih(it); } catch (e) { }
    }

    /**
     * Pilih item berdasarkan kodenya.
     * PENTING: bila kode tidak ditemukan, pilihan LAMA dibuang (bukan
     * dipertahankan) supaya tidak ada risiko mengirim kode yang salah.
     * @return {Object|null} item yang ditemukan
     */
    function pilihDariKode(v) {
      var target = String(v || '');
      if (!target) { terpilih = null; input.value = ''; return null; }
      for (var i = 0; i < items.length; i++) {
        if (String(items[i].value) === target) {
          terpilih = items[i];
          input.value = items[i].label || items[i].value;
          return items[i];
        }
      }
      // tidak ditemukan: kosongkan supaya pemanggil memutuskan sendiri
      terpilih = null;
      return null;
    }

    input.addEventListener('focus', function () { render(); buka(); });
    input.addEventListener('input', function () {
      terpilih = null;
      render();
      buka();
    });
    input.addEventListener('keydown', function (e) {
      if (e.key === 'ArrowDown') { e.preventDefault(); if (!wrap.classList.contains('is-open')) { render(); buka(); } sorot(aktif + 1); return; }
      if (e.key === 'ArrowUp') { e.preventDefault(); sorot(aktif - 1); return; }
      if (e.key === 'Enter') {
        var nodes = qsa('.mycertif-cb__item', panel);
        if (wrap.classList.contains('is-open') && nodes.length) {
          e.preventDefault();
          e.stopPropagation();
          var idx = aktif >= 0 ? aktif : 0;
          var val = nodes[idx] ? nodes[idx].getAttribute('data-cb-val') : '';
          var it = null;
          for (var i = 0; i < items.length; i++) { if (String(items[i].value) === String(val)) { it = items[i]; break; } }
          pilih(it);
        }
        return;
      }
      if (e.key === 'Escape') { tutup(); input.blur(); }
    });

    // dasar tata letak panel (cadangan bila CSS belum ada)
    try {
      panel.style.position = 'absolute';
      panel.style.left = '0';
      panel.style.right = '0';
      panel.style.top = 'calc(100% + 4px)';
      panel.style.zIndex = '1040';
      panel.style.maxHeight = '320px';
      panel.style.overflowY = 'auto';
      panel.style.background = '#fff';
      panel.style.border = '1px solid #E4E9F0';
      panel.style.borderRadius = '14px';
      panel.style.boxShadow = '0 12px 32px rgba(16,32,46,.12)';
      panel.style.padding = '.25rem';
    } catch (e) { }

    panel.addEventListener('mousedown', function (e) {
      var t = e.target.closest('[data-cb-val]');
      if (!t) return;
      e.preventDefault();
      var val = t.getAttribute('data-cb-val');
      var it = null;
      for (var i = 0; i < items.length; i++) { if (String(items[i].value) === String(val)) { it = items[i]; break; } }
      pilih(it);
    });

    document.addEventListener('click', function (e) {
      if (!wrap.contains(e.target)) tutup();
    });

    if (o.nilai) pilihDariKode(o.nilai);

    return {
      setItems: function (arr) {
        items = arr || [];
        render();
        return items.length;
      },
      setNilai: function (v) { return pilihDariKode(v); },
      nilai: function () { return terpilih ? String(terpilih.value) : ''; },
      item: function () { return terpilih; },
      fokus: function () { try { input.focus(); } catch (e) { } },
      bersihkan: function () { terpilih = null; input.value = ''; },
      tutup: tutup
    };
  }

  function nilaiForm(form) {
    var out = {};
    qsa('input, select, textarea', form).forEach(function (f) {
      if (!f.name) return;
      if (f.type === 'checkbox') { out[f.name] = f.checked; return; }
      if (f.type === 'radio') { if (f.checked) out[f.name] = f.value; return; }
      out[f.name] = f.value;
    });
    return out;
  }

  function isiForm(form, data) {
    qsa('input, select, textarea', form).forEach(function (f) {
      if (!f.name) return;
      var v = data[f.name];
      if (v === undefined || v === null) v = '';
      if (f.type === 'checkbox') { f.checked = (v === true || String(v).toUpperCase() === 'TRUE'); return; }
      if (f.type === 'radio') { f.checked = (String(f.value) === String(v)); return; }
      f.value = v;
    });
  }

  function resetForm(form) {
    if (!form) return;
    form.reset();
    qsa('.is-invalid', form).forEach(function (f) { f.classList.remove('is-invalid'); });
    qsa('.gde-error', form).forEach(function (f) { f.remove(); });
  }

  function tandaiInvalid(input, pesan) {
    if (!input) return;
    input.classList.add('is-invalid');
    var lama = input.parentNode.querySelector('.gde-error');
    if (lama) lama.remove();
    var d = document.createElement('div');
    d.className = 'gde-error';
    d.textContent = pesan;
    input.parentNode.appendChild(d);
    try { input.focus(); } catch (e) { }
  }

  function bersihkanInvalid(form) {
    if (!form) return;
    qsa('.is-invalid', form).forEach(function (f) { f.classList.remove('is-invalid'); });
    qsa('.gde-error', form).forEach(function (f) { f.remove(); });
  }

  function tombolSibuk(btn, sibuk, teksSibuk) {
    if (!btn) return;
    if (sibuk) {
      btn.dataset.teksAsli = btn.innerHTML;
      btn.disabled = true;
      btn.innerHTML = '<span class="gde-spinner" style="border-color:rgba(255,255,255,.4);border-top-color:#fff"></span> '
        + esc(teksSibuk || 'Memproses...');
    } else {
      btn.disabled = false;
      if (btn.dataset.teksAsli) { btn.innerHTML = btn.dataset.teksAsli; delete btn.dataset.teksAsli; }
    }
  }

  var ATURAN_BERKAS = {
    ext: ['pdf', 'xls', 'xlsx', 'csv', 'doc', 'docx', 'png', 'jpg', 'jpeg', 'webp', 'gif', 'bmp', 'heic', 'tif', 'tiff'],
    mime: ['application/pdf', 'application/vnd.ms-excel',
      'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
      'application/msword',
      'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
      'image/png', 'image/jpeg', 'image/webp', 'image/gif', 'image/bmp', 'image/heic', 'image/tiff'],
    maksMb: 12
  };

  function extDari(nama) {
    var s = String(nama || '');
    var i = s.lastIndexOf('.');
    return i === -1 ? '' : s.substring(i + 1).toLowerCase();
  }

  function ikonBerkas(nama) {
    var e = extDari(nama);
    if (e === 'pdf') return 'bi-file-earmark-pdf';
    if (e === 'xls' || e === 'xlsx' || e === 'csv') return 'bi-file-earmark-excel';
    if (e === 'doc' || e === 'docx') return 'bi-file-earmark-word';
    if (['png', 'jpg', 'jpeg', 'webp', 'gif', 'bmp', 'heic', 'tif', 'tiff'].indexOf(e) !== -1) return 'bi-file-earmark-image';
    return 'bi-file-earmark';
  }

  function validasiBerkas(file) {
    if (!file) return 'Berkas tidak terbaca.';
    var e = extDari(file.name);
    if (ATURAN_BERKAS.ext.indexOf(e) === -1) {
      return 'Jenis berkas .' + e + ' belum didukung. Gunakan PDF, Excel, Word, atau gambar.';
    }
    if (file.size > ATURAN_BERKAS.maksMb * 1024 * 1024) {
      return 'Ukuran berkas ' + ukuran(file.size) + ' melebihi batas ' + ATURAN_BERKAS.maksMb + ' MB.';
    }
    return '';
  }

  function bacaBerkas(file) {
    return new Promise(function (resolve, reject) {
      var r = new FileReader();
      r.onload = function () {
        var s = String(r.result || '');
        var koma = s.indexOf(',');
        resolve(koma === -1 ? s : s.substring(koma + 1));
      };
      r.onerror = function () { reject(new Error('Gagal membaca berkas.')); };
      r.readAsDataURL(file);
    });
  }

  function pratinjauUrl(fileId, mime, nama) {
    if (!fileId) return '';
    var m = String(mime || '');
    if (m.indexOf('image/') === 0 || ['png', 'jpg', 'jpeg', 'webp', 'gif', 'bmp'].indexOf(extDari(nama)) !== -1) {
      return 'https://drive.google.com/thumbnail?id=' + encodeURIComponent(fileId) + '&sz=w1000';
    }
    return 'https://drive.google.com/file/d/' + encodeURIComponent(fileId) + '/preview';
  }

  function pasangPerilakuShell() {
    var sidebar = el('mycertifSidebar');
    var scrim = el('mycertifScrim');
    var toggle = el('mycertifSidebarToggle');

    if (toggle) {
      toggle.addEventListener('click', function () {
        var terbuka = sidebar.classList.toggle('is-open');
        if (scrim) scrim.classList.toggle('is-open', terbuka);
      });
    }
    if (scrim) {
      scrim.addEventListener('click', function () {
        sidebar.classList.remove('is-open');
        scrim.classList.remove('is-open');
      });
    }

    /* Listener tingkat dokumen hanya dipasang SEKALI, walau shell
       di-render ulang (mis. setelah profil diperbarui). */
    if (state._shellGlobalBound) return;
    state._shellGlobalBound = true;

    document.addEventListener('click', function (e) {
      var pemicu = e.target.closest('[data-gde-dropdown]');
      qsa('.gde-dropdown.is-open').forEach(function (d) {
        if (!pemicu || d !== pemicu.closest('.gde-dropdown')) d.classList.remove('is-open');
      });
      if (pemicu) {
        e.preventDefault();
        pemicu.closest('.gde-dropdown').classList.toggle('is-open');
      }
    });

    document.addEventListener('click', function (e) {
      var link = e.target.closest('[data-view]');
      if (!link) return;
      var key = link.getAttribute('data-view');
      if (!VIEWS[key]) return;
      e.preventDefault();
      var drawer = el('mycertifSidebar');
      var sc = el('mycertifScrim');
      if (drawer) drawer.classList.remove('is-open');
      if (sc) sc.classList.remove('is-open');
      navigasi(key);
    });

    /* tombol keluar dipasang ulang tiap render, jadi pakai delegasi */
    document.addEventListener('click', function (e) {
      var keluar = e.target.closest('#mycertifLogout');
      if (!keluar) return;
      e.preventDefault();
      konfirmasi({
        judul: 'Keluar dari aplikasi',
        pesan: 'Anda akan keluar dari sesi ini.',
        labelOk: 'Keluar',
        bahaya: true
      }).then(function (ya) { if (ya) logout(); });
    });
  }

  function viewAwal() {
    // buang penanda token (!t=...) sebelum membaca view & parameter
    var hash = String(location.hash || '').replace(/!t=[A-Za-z0-9_-]+/g, '').replace(/^#\/?/, '');
    var nama = hash.split('?')[0];
    if (nama && VIEWS[nama] && bolehAkses(nama)) return { view: nama, param: hash.split('?')[1] || '' };
    if (bolehAkses('dashboard')) return { view: 'dashboard', param: '' };
    if (bolehAkses('data')) return { view: 'data', param: '' };
    return { view: 'profil', param: '' };
  }

  function mulaiRouter() {
    if (!state._routerBound) {
      state._routerBound = true;
      window.addEventListener('hashchange', function () {
        var t = viewAwal();
        if (t.view !== state.view) navigasi(t.view, { hash: false, param: t.param });
      });
    }
    var awal = viewAwal();
    navigasi(awal.view, { hash: false, param: awal.param });
  }

      APP.version = bootstrap.version || APP.version;
      state.bootstrap = bootstrap;
    }

      if (!masuk) { tampilkanLogin(); return; }
      return masukKeAplikasi_();
    }).catch(function (err) {
      tampilkanLogin(err.message || '');
    });
  }

  /** Dipakai setelah login berhasil (dari halaman Login maupun login otomatis). */

    return muatMeta().then(function () {
      perluLoginUlang = false;

      mulaiRouter();
      perbaruiBadgeHeader();

      if (u.must_change_password) {
        ingat('Akun Anda masih memakai pengaturan awal. Periksa menu Username & Password.',
          'Perhatian keamanan');
      }
      if (state.autoLoginInfo && !state._autoInfoShown) {
        state._autoInfoShown = true;
        info(state.autoLoginInfo, 'Masuk otomatis');
      }
      return true;
    });
  }
