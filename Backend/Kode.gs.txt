 * Saat dipasang ke editor Apps Script, ekstensi .txt DIHAPUS, nama dipertahankan.
 *   Backend/Kode.gs.txt        -> Kode.gs
 *   Backend/subprocess.gs.txt  -> subprocess.gs
 *   Frontend/Index.html.txt    -> Index.html
 *   ... dst.
 *
 * ARSITEKTUR
 * ---------------------------------------------------------------------------
 *   Kode.gs        (berkas ini)  -> konstanta, skema, entry point, kickstarter,
 *                                   helper DB/Drive/umum, menu, trigger.
 *   subprocess.gs                -> seluruh logika bisnis: autentikasi, CRUD,
 *                                   mesin status, dashboard, profil, export,
 *                                   reminder email.
 *
 * Model komunikasi frontend<->backend: google.script.run memanggil fungsi
 * `apiXxx` yang SELALU mengembalikan { ok:true, data:... } atau { ok:false, error:... }.
 *
 * ============================================================================
 * KICKSTARTER (SETUP OTOMATIS)
 * ============================================================================
 * Saat pertama kali halaman dibuka (doGet), aplikasi otomatis:
 *   1. mencari ID spreadsheet (konstanta -> Script Property -> spreadsheet aktif)
 *   2. membuat sheet yang belum ada + merapikan header yang berubah
 *   3. memastikan semua key Config default ada
 *   4. membuat folder Drive untuk lampiran & hasil export
 *   5. mengisi Master_Sertifikasi (69 jenis hasil normalisasi data lama)
 *   6. membuat akun awal: superadmin + HC
 *   7. menandai setup selesai (Script Property)
 *
 * Jalankan ulang paksa: buka URL web app + "?bootstrap=1"
 * atau jalankan kickstartFromMenu_() dari editor.
 */

function doGet(e) {
  e = e || {};
  var params = e.parameter || {};
  var force = String(params.bootstrap || '') === '1';
  var diagnose = String(params.diagnose || '') === '1';

  // ---------- 1. KICKSTARTER ----------
  var report = null;
  try {
    report = kickstart_(force);
  } catch (err) {
    return renderSetupPage_('Setup belum bisa dijalankan',
      (err && err.message) ? err.message : String(err), null);
  }

  if (diagnose) return renderSetupPage_('Status aplikasi', 'Diagnosa kickstarter', report);

  // ---------- 2. RENDER APLIKASI ----------
  try {
    var tpl = HtmlService.createTemplateFromFile('Index');
    tpl.bootstrap = JSON.stringify({
      appName: APP_NAME,
      appSub: APP_SUB,
      appOwner: APP_OWNER,
      version: APP_VERSION,
      execUrl: getExecUrl_(),
      tz: getConfig_('zona_waktu') || DEFAULT_TZ,
      statuses: STATUS_DEFS,
      roles: ROLES_,
      bootstrap: report
    });

    return tpl.evaluate()
      .setTitle(APP_NAME + ' — Unit Patuha')
      .setSandboxMode(HtmlService.SandboxMode.IFRAME)
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
  } catch (err2) {
    return renderSetupPage_('Gagal memuat halaman aplikasi',
      (err2 && err2.message) ? err2.message : String(err2), report);
  }
}

function doPost(e) { return doGet(e); }

function isBootstrapDone_() {
  try {
    return PropertiesService.getScriptProperties().getProperty(BOOTSTRAP_PROP_) === BOOTSTRAP_VERSION_;
  } catch (e) { return false; }
}

/**
 * Jalankan seluruh persiapan aplikasi.
 * @param {boolean} force true = jalankan ulang walau sudah pernah selesai.
 * @return {Object} laporan langkah.
 */
function kickstart_(force) {
  var t0 = new Date();
  var report = {
    ok: true, version: BOOTSTRAP_VERSION_, app_version: APP_VERSION,
    sudah_sebelumnya: false, spreadsheet_id: '', drive_folder_id: '',
    steps: [], errors: [], durasi_ms: 0
  };

  if (!force && isBootstrapDone_()) {
    report.sudah_sebelumnya = true;
    try { report.spreadsheet_id = resolveSpreadsheetId_(); } catch (e) { }
    report.steps.push('bootstrap sudah pernah selesai — dilewati');
    return report;
  }

  // ---------- LANGKAH 1: spreadsheet ----------
  var id = resolveSpreadsheetId_();
  if (!id) {
    throw new Error('Spreadsheet database belum ditentukan. Isi konstanta SPREADSHEET_ID di Kode.gs '
      + '(hanya ID-nya, bukan URL), atau bind script ini ke spreadsheet, lalu buka halaman ini lagi.');
  }
  try { PropertiesService.getScriptProperties().setProperty('SPREADSHEET_ID', id); } catch (e) { }
  report.spreadsheet_id = id;
  report.steps.push('spreadsheet ditemukan: ' + id);

  // ---------- LANGKAH 2 & 3: sheet + config ----------
  var initRes = initSheets_();
  report.steps.push('sheet diperiksa: ' + (initRes.created || []).length + ' dibuat, '
    + (initRes.repaired || []).length + ' header dirapikan');
  report.steps.push('config default dipastikan lengkap');

  // ---------- LANGKAH 4: folder Drive ----------
  try {
    report.drive_folder_id = ensureDriveFolder_();
    report.steps.push('folder lampiran siap: ' + report.drive_folder_id);
  } catch (eDrive) {
    report.errors.push('Folder Drive: ' + errMsg_(eDrive));
    report.steps.push('folder Drive BELUM siap (unggah lampiran bisa gagal)');
  }

  // ---------- LANGKAH 5: seed jenis sertifikasi ----------
  try {
    var seedCert = seedMasterSertifikasi_();
    report.steps.push(seedCert.created > 0
      ? ('master jenis sertifikasi diisi ' + seedCert.created + ' baris')
      : 'master jenis sertifikasi sudah terisi');
  } catch (eCert) {
    report.errors.push('Seed sertifikasi: ' + errMsg_(eCert));
  }

  // ---------- LANGKAH 6: akun awal ----------
  if (String(getConfig_('bootstrap_auto_seed') || 'TRUE').toUpperCase() !== 'FALSE') {
    try {
      var seeded = seedAwal_();
      report.seed = seeded;
      report.steps.push('akun awal disiapkan: ' + (seeded.akun || []).join(', '));
    } catch (eSeed) {
      report.errors.push('Seeder: ' + errMsg_(eSeed));
    }
  }

  // ---------- LANGKAH 7: tandai selesai ----------
  try {
    setConfig_('bootstrap_version', BOOTSTRAP_VERSION_);
    setConfig_('bootstrap_done_at', nowStr_());
  } catch (e) { }
  try {
    PropertiesService.getScriptProperties().setProperty(BOOTSTRAP_PROP_, BOOTSTRAP_VERSION_);
  } catch (e) {
    report.errors.push('Tidak bisa menyimpan penanda bootstrap: ' + errMsg_(e));
  }

  report.durasi_ms = (new Date()).getTime() - t0.getTime();
  report.steps.push('bootstrap selesai dalam ' + report.durasi_ms + ' ms');
  try { logAudit_('SYSTEM', 'system', 'SA', 'BOOTSTRAP', 'APP', BOOTSTRAP_VERSION_, report.steps.join(' | ')); } catch (e) { }

  report.ok = report.errors.length === 0;
  return report;
}

function initSheets_() {
  var db = getDb_();
  var created = [];
  var repaired = [];

  Object.keys(SCHEMAS).forEach(function (name) {
    try {
      var sh = db.getSheetByName(name);
      var header = SCHEMAS[name];

      if (!sh) {
        sh = db.insertSheet(name);
        styleHeader_(sh, header);
        created.push(name);
      } else {
        var lastCol = Math.max(sh.getLastColumn(), header.length);
        var cur = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(function (s) { return String(s).trim(); });
        cur = cur.slice(0, header.length);
        var need = cur.length < header.length;
        for (var i = 0; i < header.length && !need; i++) { if (cur[i] !== header[i]) need = true; }
        if (need) {
          styleHeader_(sh, header);
          repaired.push(name);
        }
      }
      sh.setFrozenRows(1);
    } catch (e) {
      Logger.log('initSheets ' + name + ' gagal: ' + errMsg_(e));
    }
  });

  seedConfigSheet_();

  // Kolom tanggal & kode dipaksa teks agar Sheets tidak mengubah format
  // (mis. "4122196-C" aman, tetapi tanggal tetap ditulis sebagai Date).
  try { forceTextColumns_('Master_User', ['Employee_code', 'No_HP']); } catch (e) { }
  try { forceTextColumns_('Master_File', ['Doc_id', 'Employee_id', 'No_sertifikat']); } catch (e) { }
  try { forceTextColumns_(null, null); } catch (e) { }

  try { ensureDriveFolder_(); } catch (e) { Logger.log('drive folder: ' + errMsg_(e)); }
  try { ensureExportFolder_(); } catch (e) { Logger.log('export folder: ' + errMsg_(e)); }

  return { ok: true, created: created, repaired: repaired };
}

/**
 * Paksa kolom tertentu berformat TEKS.
 * @param {string} sheetName nama sheet (null = tidak melakukan apa-apa)
 * @param {string[]} cols daftar nama kolom
 */
function forceTextColumns_(sheetName, cols) {
  if (!sheetName || !cols || !cols.length) return false;
  var sh = getSheet_(sheetName);
  if (!sh) return false;
  var header = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(function (h) { return String(h).trim(); });
  cols.forEach(function (c) {
    var idx = header.indexOf(c);
    if (idx === -1) return;
    try { sh.getRange(2, idx + 1, Math.max(sh.getMaxRows() - 1, 1), 1).setNumberFormat('@'); } catch (e) { }
  });
  return true;
}

function styleHeader_(sh, header) {
  var r = sh.getRange(1, 1, 1, header.length);
  r.setValues([header])
    .setFontWeight('bold')
    .setBackground('#093D7D')
    .setFontColor('#FFFFFF')
    .setVerticalAlignment('middle');
  try { sh.setColumnWidths(1, header.length, 140); } catch (e) { }
  try { sh.setRowHeight(1, 30); } catch (e) { }
  try { if (!sh.getFilter()) r.createFilter(); } catch (e) { }
}

}

function getAllConfig_() {
  var out = {};
  Object.keys(CONFIG_DEFAULTS).forEach(function (k) { out[k] = CONFIG_DEFAULTS[k]; });
  try {
    var sh = getSheet_('Config');
    if (!sh) return out;
    var vals = sh.getDataRange().getValues();
    for (var i = 1; i < vals.length; i++) {
      var k = String(vals[i][0]).trim();
      if (!k) continue;
      var v = vals[i][1];

      }
    }
  } catch (e) { Logger.log('getAllConfig_: ' + errMsg_(e)); }
  return out;

function getConfig_(key) {
  var sh;
  try { sh = getSheet_('Config'); } catch (e) { return CONFIG_DEFAULTS[key] || ''; }
  if (!sh) return CONFIG_DEFAULTS[key] || '';
  var vals = sh.getDataRange().getValues();

      if (v !== '') return v;
      return CONFIG_DEFAULTS[key] || '';
    }
  }
  return CONFIG_DEFAULTS[key] || '';
}

function getConfigNum_(key, def) {

  return isNaN(v) ? (def === undefined ? 0 : def) : v;
}

function getConfigBool_(key, def) {
  var v = String(getConfig_(key)).trim().toUpperCase();
  if (v === '') return !!def;
  return v === 'TRUE' || v === '1' || v === 'YA' || v === 'YES';

// ---------- helper baris generik ----------

/** Susun array nilai sesuai urutan kolom SCHEMAS. */
function rowArrBySchema_(sheetName, obj) {
  var header = SCHEMAS[sheetName];
  if (!header) throw new Error('Skema tidak dikenal: ' + sheetName);
  return header.map(function (col) {
    var v = obj[col];
    return (v === undefined || v === null) ? '' : v;
  });
}

/** Ambil semua baris sheet sebagai array objek (header -> nilai). */
function sheetToObjects_(sheetName) {
  var sh = getSheet_(sheetName);
  if (!sh || sh.getLastRow() < 2) return [];
  var values = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getValues();
  var header = values[0].map(function (h) { return String(h).trim(); });
  var out = [];
  for (var r = 1; r < values.length; r++) {
    var row = values[r];
    var kosong = true;
    for (var c = 0; c < header.length; c++) {
      if (row[c] !== '' && row[c] !== null && row[c] !== undefined) { kosong = false; break; }
    }
    if (kosong) continue;
    var obj = { __row: r + 1 };
    for (var i = 0; i < header.length; i++) {
      if (!header[i]) continue;
      obj[header[i]] = row[i] === null || row[i] === undefined ? '' : row[i];
    }
    out.push(obj);
  }
  return out;
}

/** Nilai satu baris sebagai objek. */
function rowToObject_(sheetName, rowIndex) {
  var sh = getSheet_(sheetName);
  if (!sh) return null;
  var header = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
    .map(function (h) { return String(h).trim(); });
  var row = sh.getRange(rowIndex, 1, 1, header.length).getValues()[0];
  var obj = { __row: rowIndex };
  for (var i = 0; i < header.length; i++) {
    if (!header[i]) continue;
    obj[header[i]] = row[i] === null || row[i] === undefined ? '' : row[i];
  }
  return obj;
}

/** Cari nomor baris pertama yang kolom `col` nilainya sama dengan `val`. */
function findRowByCol_(sheetName, col, val) {
  var sh = getSheet_(sheetName);
  if (!sh || sh.getLastRow() < 2) return -1;
  var header = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
    .map(function (h) { return String(h).trim(); });
  var idx = header.indexOf(col);
  if (idx === -1) throw new Error('Kolom "' + col + '" tidak ada di sheet ' + sheetName);
  var values = sh.getRange(2, idx + 1, sh.getLastRow() - 1, 1).getValues();
  var target = String(val).trim().toLowerCase();
  for (var i = 0; i < values.length; i++) {
    if (String(values[i][0]).trim().toLowerCase() === target) return i + 2;
  }
  return -1;
}

/** Tulis objek ke baris yang sudah ada (hanya kolom yang ada di objek). */
function updateRowByCol_(sheetName, rowIndex, obj) {
  var sh = getSheet_(sheetName);
  if (!sh) return false;
  var header = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
    .map(function (h) { return String(h).trim(); });
  var changed = false;
  Object.keys(obj).forEach(function (k) {
    var idx = header.indexOf(k);
    if (idx === -1) return;
    sh.getRange(rowIndex, idx + 1).setValue(obj[k]);
    changed = true;
  });
  return changed;
}

function deleteRowByCol_(sheetName, col, val) {
  var row = findRowByCol_(sheetName, col, val);
  if (row === -1) return false;
  getSheet_(sheetName).deleteRow(row);
  return true;
}

function appendObject_(sheetName, obj) {
  var sh = getSheet_(sheetName);
  if (!sh) throw new Error('Sheet tidak ditemukan: ' + sheetName);
  sh.appendRow(rowArrBySchema_(sheetName, obj));
  return sh.getLastRow();
}

/** Pastikan kolom ada; kalau belum, tambahkan di kanan. */
function ensureColumns_(sheetName, cols) {
  var sh = getSheet_(sheetName);
  if (!sh) return [];
  var lastCol = Math.max(sh.getLastColumn(), 1);
  var header = sh.getRange(1, 1, 1, lastCol).getValues()[0]
    .map(function (h) { return String(h).trim(); });
  var added = [];
  cols.forEach(function (c) {
    if (header.indexOf(c) !== -1) return;
    lastCol = sh.getLastColumn() + 1;
    sh.getRange(1, lastCol).setValue(c).setFontWeight('bold')
      .setBackground('#093D7D').setFontColor('#FFFFFF');
    header.push(c);
    added.push(c);
  });
  if (added.length && SCHEMAS[sheetName]) {
    added.forEach(function (c) { if (SCHEMAS[sheetName].indexOf(c) === -1) SCHEMAS[sheetName].push(c); });
  }
  return added;
}

}

/**
 * Unggah lampiran ke folder Drive.
 * Nama berkas: <uuid:6char>_<employeecode>_<Nama Sertifikasi>.<ext>
 * @param {Object} file { name, mimeType, base64 }
 * @param {string} employeeCode
 * @param {string} certName
 * @return {Object} { fileId, url, name, mime, size, icon }
 */
function uploadAttachment_(file, employeeCode, certName) {
  if (!file || !file.base64) throw new Error('Berkas lampiran kosong.');

  var mime = String(file.mimeType || '').trim();
  var sizeBytes = Math.round(String(file.base64).length * 3 / 4);
  var maxBytes = MAX_UPLOAD_MB_ * 1024 * 1024;
  if (sizeBytes > maxBytes) {
    throw new Error('Ukuran berkas ' + humanSize_(sizeBytes) + ' melebihi batas ' + MAX_UPLOAD_MB_ + ' MB.');
  }

  var ext = extFromName_(file.name);
  var okMime = !!ALLOWED_MIME_[mime];
  var okExt = ext && ALLOWED_EXT_.indexOf(ext) !== -1;
  if (!okMime && !okExt) {
    throw new Error('Jenis berkas tidak didukung (' + (mime || ext || 'tidak dikenal')
      + '). Format yang diterima: PDF, Excel, Word, dan gambar.');
  }
  if (!mime || !ALLOWED_MIME_[mime]) mime = mimeFromExt_(ext);

  var fileName = buildFileName_(employeeCode, certName, ext);
  var blob = Utilities.newBlob(Utilities.base64Decode(file.base64), mime, fileName);
  var folder = DriveApp.getFolderById(ensureDriveFolder_());
  var created = folder.createFile(blob);
  try { created.setDescription('Lampiran sertifikasi ' + certName + ' — ' + employeeCode); } catch (e) { }

  return {
    fileId: created.getId(),
    url: created.getUrl(),
    name: fileName,
    mime: mime,
    size: sizeBytes,
    ext: ext,
    icon: FILE_ICON_[ext] || 'bi-file-earmark'
  };
}

/**
 * Nama berkas tersimpan: <uuid:6char>_<employeecode>_<Nama Sertifikasi>.<ext>
 * Karakter terlarang dibersihkan, panjang dibatasi 140 karakter.
 */
function buildFileName_(employeeCode, certName, ext) {
  var kode = String(employeeCode || 'TANPA-KODE').trim();
  var nama = String(certName || 'Sertifikasi').trim();
  nama = nama.replace(/[\\\/:*?"<>|#%]/g, ' ').replace(/\s+/g, ' ').trim();
  if (!nama) nama = 'Sertifikasi';
  var base = uuid6_() + '_' + kode + '_' + nama;
  if (base.length > 140) base = base.substring(0, 140).trim();
  return base + (ext ? ('.' + ext) : '');
}

  var chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

// ---------- tanggal ----------

/** Normalisasi nilai apa pun menjadi Date (00:00) atau null. */
function toDate_(v) {
  if (v === null || v === undefined || v === '') return null;
  if (v instanceof Date) {
    if (isNaN(v.getTime())) return null;
    return new Date(v.getFullYear(), v.getMonth(), v.getDate());
  }
  var s = String(v).trim();
  if (!s) return null;

  // ISO yyyy-mm-dd
  var iso = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (iso) return new Date(+iso[1], +iso[2] - 1, +iso[3]);

  // dd/mm/yyyy atau dd-mm-yyyy
  var dmy = s.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{4})/);
  if (dmy) {
    var d1 = +dmy[1], m1 = +dmy[2], y1 = +dmy[3];
    if (d1 > 12 && m1 <= 12) return new Date(y1, m1 - 1, d1);
    return new Date(y1, m1 - 1, d1);
  }
  var t = new Date(s);
  if (!isNaN(t.getTime())) return new Date(t.getFullYear(), t.getMonth(), t.getDate());
  return null;
}

function fmtDate_(v, pola) {
  var d = toDate_(v);
  if (!d) return '';
  return Utilities.formatDate(d, tz_(), pola || 'yyyy-MM-dd');
}

/**
 * Normalisasi menjadi Date DENGAN jam-menit-detik (untuk kolom seperti
 * Valid_Until / Terkunci_Sampai). Menerima "yyyy-MM-dd HH:mm:ss".
 */
function toDateTime_(v) {
  if (v === null || v === undefined || v === '') return null;
  if (v instanceof Date) return isNaN(v.getTime()) ? null : v;
  var s = String(v).trim();
  var m = s.match(/^(\d{4})-(\d{2})-(\d{2})[ T](\d{1,2}):(\d{2})(?::(\d{2}))?/);
  if (m) return new Date(+m[1], +m[2] - 1, +m[3], +m[4], +m[5], +(m[6] || 0));
  var d = toDate_(v);
  return d;
}

function fmtTanggalIndo_(v) {
  var d = toDate_(v);
  if (!d) return '';
  return d.getDate() + ' ' + BULAN_ID_[d.getMonth()] + ' ' + d.getFullYear();
}

function addMonths_(date, months) {
  var d = toDate_(date);
  if (!d) return null;
  var m = d.getMonth() + Number(months || 0);
  var y = d.getFullYear() + Math.floor(m / 12);
  m = ((m % 12) + 12) % 12;
  var hariTerakhir = new Date(y, m + 1, 0).getDate();
  var dd = Math.min(d.getDate(), hariTerakhir);
  return new Date(y, m, dd);
}

function diffDays_(a, b) {
  var d1 = toDate_(a), d2 = toDate_(b);
  if (!d1 || !d2) return null;
  return Math.round((d1.getTime() - d2.getTime()) / 86400000);
}

// ---------- password ----------

/**
 * Hash password: "sha256$<salt>$<hex>".
 * Nilai lama yang masih plaintext (mis. Employee_code) tetap diterima
 * oleh verifyPassword_ lalu otomatis di-upgrade.
 */
function hashPassword_(plain, salt) {
  var s = salt || uuid6_() + uuid6_();
  var raw = s + '::' + String(plain);
  var digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, raw, Utilities.Charset.UTF_8);
  return 'sha256$' + s + '$' + digest.map(function (b) {
    var v = (b < 0 ? b + 256 : b).toString(16);
    return v.length === 1 ? '0' + v : v;
  }).join('');
}

function isHashed_(stored) {
  return String(stored || '').indexOf('sha256$') === 0;
}

function verifyPassword_(plain, stored) {
  var s = String(stored || '');
  if (!s) return false;
  if (isHashed_(s)) {
    var bagian = s.split('$');
    if (bagian.length < 3) return false;
    return hashPassword_(plain, bagian[1]) === s;
  }
  // mode lama: plaintext
  return String(plain) === s;
}

function clone_(o) { return JSON.parse(JSON.stringify(o)); }

function pick_(obj, keys) {
  var out = {};
  (keys || []).forEach(function (k) { if (obj[k] !== undefined) out[k] = obj[k]; });
  return out;
}

function uniq_(arr) {
  var seen = {}, out = [];
  (arr || []).forEach(function (v) {
    var k = String(v);
    if (!seen[k]) { seen[k] = 1; out.push(v); }
  });
  return out;
}
