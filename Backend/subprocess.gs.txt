/** Ambang batas status dalam HARI (diambil dari sheet Config). */
function ambangHari_() {
  return {
    peringatan: getConfigNum_('status_peringatan_hari', 270),  // 9 bulan
    perhatian: getConfigNum_('status_perhatian_hari', 180),    // 6 bulan
    segera: getConfigNum_('status_segera_hari', 90),           // 3 bulan
    mendesak: getConfigNum_('status_mendesak_hari', 60),       // 2 bulan
    merah: getConfigNum_('status_merah_hari', 30)              // 1 bulan
  };
}

/**
 * Tentukan definisi status dari sisa hari.
 * @param {number} sisaHari
 * @param {Object=} ambang ambang yang sudah dihitung (opsional, untuk loop besar)
 * @return {Object} salah satu STATUS_DEFS
 */
function statusDariSisaHari_(sisaHari, ambang) {
  var h = Number(sisaHari);
  if (isNaN(h)) return null;
  var a = ambang || ambangHari_();
  if (h < 0) return STATUS_MAP_['KADALUARSA'];
  if (h <= a.mendesak) return STATUS_MAP_['MENDEKATI_KADALUARSA'];
  if (h <= a.segera) return STATUS_MAP_['SEGERA_PERPANJANG'];
  if (h <= a.perhatian) return STATUS_MAP_['PERLU_PERHATIAN'];
  if (h <= a.peringatan) return STATUS_MAP_['PERINGATAN'];
  return STATUS_MAP_['MASIH_BERLAKU'];
}

/**
 * Tanggal rekomendasi perpanjangan = Expired_at dikurangi 3 bulan.
 * Dipakai untuk memberi tanda "mulai siapkan perpanjangan" sebelum dokumen habis.
 * @param {*} expiredAt
 * @return {string} 'yyyy-MM-dd' atau '' bila tanggal tidak valid
 */
function recomUpdate_(expiredAt) {
  var exp = toDate_(expiredAt);
  if (!exp) return '';
  return fmtDate_(addMonths_(exp, -3), 'yyyy-MM-dd');
}

/**
 * Ubah selisih hari menjadi teks berjenjang tahun/bulan/hari.
 *
 * Contoh keluaran: "3 Tahun 2 Bulan 4 Hari", "2 Bulan", "17 Hari",
 * "Lewat 1 Bulan 5 Hari" (bila sudah kedaluwarsa).
 *
 * Perhitungan memakai kalender sungguhan (bukan pembagian 30 hari) supaya
 * hasilnya cocok dengan yang dibaca manusia pada kalender, termasuk untuk
 * bulan yang jumlah harinya berbeda.
 *
 * @param {*} tanggalAkhir tanggal kedaluwarsa
 * @param {*} refDate tanggal acuan (default hari ini)
 */
function durasiTeks_(tanggalAkhir, refDate) {
  var akhir = toDate_(tanggalAkhir);
  if (!akhir) return '';
  var ref = refDate ? toDate_(refDate) : todayDate_();
  var lewat = akhir.getTime() < ref.getTime();
  var a = lewat ? akhir : ref;      // titik awal (lebih lampau)
  var b = lewat ? ref : akhir;      // titik akhir (lebih baru)

  var tahun = b.getFullYear() - a.getFullYear();
  var bulan = b.getMonth() - a.getMonth();
  var hari = b.getDate() - a.getDate();

  if (hari < 0) {
    bulan -= 1;
    // jumlah hari pada bulan sebelum tanggal akhir
    hari += new Date(b.getFullYear(), b.getMonth(), 0).getDate();
  }
  if (bulan < 0) {
    tahun -= 1;
    bulan += 12;
  }

  var bagian = [];
  if (tahun > 0) bagian.push(tahun + ' Tahun');
  if (bulan > 0) bagian.push(bulan + ' Bulan');
  if (hari > 0 || !bagian.length) bagian.push(hari + ' Hari');

  return (lewat ? 'Lewat ' : '') + bagian.join(' ');
}

/**
 * Hitung sisa hari + sisa bulan + status untuk sebuah tanggal kedaluwarsa.
 * @param {*} expiredAt tanggal kedaluwarsa
 * @param {*} refDate tanggal acuan (default hari ini)
 * @param {Object=} ambang ambang status yang sudah dihitung (opsional)
 */
function hitungMasaBerlaku_(expiredAt, refDate, ambang) {
  var exp = toDate_(expiredAt);
  if (!exp) {
    return {
      sisa_hari: '', sisa_bulan: '', sisa_teks: '', status: '',
      status_label: 'Belum ada tanggal', ada_tanggal: false, recom_update: ''
    };
  }
  var ref = refDate ? toDate_(refDate) : todayDate_();
  var sisaHari = Math.round((exp.getTime() - ref.getTime()) / 86400000);
  var sisaBulan = Math.round((sisaHari / 30) * 10) / 10;
  var def = statusDariSisaHari_(sisaHari, ambang);
  return {
    sisa_hari: sisaHari,
    sisa_bulan: sisaBulan,
    sisa_teks: durasiTeks_(exp, ref),
    status: def ? def.key : '',
    status_label: def ? def.label : '',
    status_tone: def ? def.tone : 'gray',
    ada_tanggal: true,
    expired_at: fmtDate_(exp, 'yyyy-MM-dd'),
    recom_update: recomUpdate_(exp)
  };
}

/**
 * Hitung expired_at otomatis = End_Pelaksanaan + masa berlaku.
 * @param {*} endPelaksanaan
 * @param {number} masaBulan
 */
function deriveExpiredAt_(endPelaksanaan, masaBulan) {
  var dasar = toDate_(endPelaksanaan);
  if (!dasar) return null;
  var bulan = Number(masaBulan);
  if (isNaN(bulan) || bulan <= 0) bulan = getConfigNum_('masa_berlaku_default_bulan', 36);
  return addMonths_(dasar, bulan);
}

/**
 * Tebak masa berlaku (bulan) dari nama sertifikasi.
 *
 * Sejak jenjang jenis sertifikasi dihapus, satu-satunya sumber otomatis adalah
 * daftar kata kunci pada Config `masa_berlaku_5_tahun_kata_kunci` (mis. CRMP,
 * Infrared Thermography) yang berlaku 60 bulan; sisanya memakai default 36 bulan.
 * Nilai `masa_berlaku_bulan` yang diisi HC di formulir selalu menang — lihat
 * apiSaveFile.
 */
function masaBerlakuDariNama_(namaSertifikasi) {
  var def = getConfigNum_('masa_berlaku_default_bulan', 36);
  var kunci = String(getConfig_('masa_berlaku_5_tahun_kata_kunci') || '').split(',');
  var nama = String(namaSertifikasi || '').toLowerCase();
  for (var i = 0; i < kunci.length; i++) {
    var k = String(kunci[i]).trim().toLowerCase();
    if (k && nama.indexOf(k) !== -1) return 60;
  }
  return def;
}

/** Susun ulang kolom turunan (sisa hari/bulan/status) untuk sebuah objek file. */
function lengkapiKolomHitung_(obj, refDate) {
  var hit = hitungMasaBerlaku_(obj.Expired_at, refDate);
  obj.Sisa_masa_berlaku_day = hit.ada_tanggal ? hit.sisa_hari : '';
  obj.Sisa_masa_berlaku_by_month = hit.ada_tanggal ? hit.sisa_bulan : '';
  obj.Status = hit.status;
  return obj;
}

/** Hitung ulang SELURUH baris T_Sertifikasi (baca & tulis sekali, hasil di-batch). */
function recalcAllStatus_() {
  var semua = sheetValues_('T_Sertifikasi');
  if (semua.length < 2) return { ok: true, total: 0, diubah: 0 };

  var header = sheetHeader_('T_Sertifikasi');
  var iExp = header.indexOf('Expired_at');
  var iDay = header.indexOf('Sisa_masa_berlaku_day');
  var iMon = header.indexOf('Sisa_masa_berlaku_by_month');
  var iSta = header.indexOf('Status');
  var iRec = header.indexOf('Recom_Update');
  if (iExp === -1 || iDay === -1 || iSta === -1) {
    throw new Error('Kolom T_Sertifikasi tidak lengkap (butuh Expired_at, Sisa_masa_berlaku_day, Status).');
  }

  var vals = semua.slice(1);
  var out = [];
  var diubah = 0;
  var batas = ambangHari_();          // diambil SEKALI, bukan per baris
  var ref = todayDate_();
  for (var r = 0; r < vals.length; r++) {
    var exp = vals[r][iExp];
    var hit = hitungMasaBerlaku_(exp, ref, batas);
    var baru = vals[r].slice();
    var nDay = hit.ada_tanggal ? hit.sisa_hari : '';
    var nMon = hit.ada_tanggal ? hit.sisa_bulan : '';
    var nSta = hit.status;
    if (String(vals[r][iDay]) !== String(nDay) || String(vals[r][iMon]) !== String(nMon)
      || String(vals[r][iSta]) !== String(nSta)) diubah++;
    baru[iDay] = nDay;
    if (iMon !== -1) baru[iMon] = nMon;
    baru[iSta] = nSta;
    // Recom_Update selalu Expired_at - 3 bulan, jadi ikut dirapikan di sini
    if (iRec !== -1) baru[iRec] = hit.recom_update || '';
    out.push(baru);
  }
  if (out.length) tulisBanyak_('T_Sertifikasi', 2, out);
  return { ok: true, total: out.length, diubah: diubah, pada: nowStr_() };
}

/** Metadata untuk frontend: status, ambang, opsi dropdown, aturan berkas. */
function apiGetMeta(token) {
  return apiGuard_(function () {
    var auth = token ? requireAuth_(token) : null;

    /*
     * Sejak jenjang jenis/kategori sertifikasi dihapus, nama sertifikasi diisi
     * manual. Supaya penulisan tetap konsisten (tidak muncul varian typo seperti
     * "POP" vs "Pengawas Operasional Pertama"), daftar SARAN diambil dari nama
     * yang sudah pernah dipakai pada T_Sertifikasi. Ini hanya autocomplete —
     * bukan master dan tidak membatasi input — tetapi sekaligus membawa masa
     * berlaku & penyelenggara terakhir supaya formulir bisa terisi otomatis.
     */
    var saran = {};
    try {
      sheetToObjects_('T_Sertifikasi').forEach(function (f) {
        var n = String(f.Nama_Sertifikasi || '').trim();
        if (!n) return;
        var k = n.toLowerCase();
        if (!saran[k]) {
          saran[k] = {
            nama: n,
            masa_berlaku_bulan: Number(f.Masa_berlaku_bulan) || 0,
            penyelenggara: String(f.Penyelenggara || ''),
            jumlah: 0
          };
        }
        saran[k].jumlah++;
      });
    } catch (e) { }
    var namaOptions = Object.keys(saran).map(function (k) { return saran[k]; })
      .sort(function (a, b) { return a.nama.localeCompare(b.nama); });

    return {
      app_name: APP_NAME,
      app_sub: APP_SUB,
      app_owner: APP_OWNER,
      app_version: APP_VERSION,
      zona_waktu: tz_(),
      statuses: STATUS_DEFS,
      verifikasi_defs: VERIFIKASI_DEFS_,
      roles: ROLES_,
      role_labels: ROLE_LABEL_,
      ambang: ambangHari_(),
      divisi_options: opsiDivisi_(),
      nama_sertifikasi_options: namaOptions,
      allowed_ext: ALLOWED_EXT_,
      allowed_mime: Object.keys(ALLOWED_MIME_),
      max_upload_mb: MAX_UPLOAD_MB_,
      file_icons: FILE_ICON_,
      reminder: {
        enabled: getConfigBool_('reminder_enabled', true),
        hari_kirim: getConfigNum_('reminder_hari_kirim', 1),
        level: String(getConfig_('reminder_level_bulan') || ''),
        ke_karyawan: getConfigBool_('reminder_ke_karyawan', true),
        ke_hc: getConfigBool_('reminder_ke_hc', true),
        ke_atasan: getConfigBool_('reminder_ke_atasan', true),
        extra_emails: String(getConfig_('reminder_extra_emails') || '')
      },
      boleh: auth ? {
        kelola_user: (auth.role === 'HC' || auth.role === 'SA'),
        jalankan_reminder: (auth.role === 'HC' || auth.role === 'SA'),
        lihat_semua: (auth.role === 'HC' || auth.role === 'SA')
      } : {}
    };
  });
}

function apiGetConfig(token) {
  return apiGuard_(function () {
    requireHc_(token);
    return { config: getAllConfig_() };
  });
}

function apiSetConfig(token, obj) {
  return apiGuard_(function () {
    var auth = requireHc_(token);
    if (!obj) throw new Error('Tidak ada perubahan.');
    var dilindungi = { 'spreadsheet_id': 1, 'superadmin_email': 1 };
    var diubah = {}, dilewati = {};
    Object.keys(obj).forEach(function (k) {
      if (dilindungi[k] && auth.role !== 'SA') { dilewati[k] = true; return; }
      if (CONFIG_DEFAULTS[k] === undefined) { dilewati[k] = true; return; }
      setConfig_(k, obj[k]);
      diubah[k] = obj[k];
    });
    logAudit_(auth.employee_code, auth.username, auth.role, 'UBAH_CONFIG', 'CONFIG', '-', diubah);
    return { diubah: Object.keys(diubah), dilewati: Object.keys(dilewati) };
  });
}

/**
 * Hitung ulang kolom turunan (sisa hari, sisa bulan, status) untuk SELURUH
 * baris T_Sertifikasi. Dipakai setelah ambang status diubah di halaman
 * Pengaturan atau setelah impor data massal.
 */
function apiRecalcStatus(token) {
  return apiGuard_(function () {
    var auth = requireHc_(token);
    var r = recalcAllStatus_();
    logAudit_(auth.employee_code, auth.username, auth.role, 'HITUNG_ULANG_STATUS', 'SYSTEM', '-',
      'total ' + r.total + ', diubah ' + r.diubah);
    return {
      total: r.total,
      diubah: r.diubah,
      pada: r.pada,
      ambang: ambangHari_()
    };
  });
}

/**
 * Satu panggilan untuk SEMUA angka pada header/sidebar.
 * Digabung supaya perpindahan halaman hanya memerlukan satu perjalanan ke
 * server (sebelumnya 2-3 panggilan terpisah sehingga terasa "loncat-loncat").
 * @return {Object} { notif_belum_dibaca, kadaluarsa, mendesak, perlu_verifikasi,
 *                    total_sertifikasi, badge_mendesak }
 */
function apiGetRingkasanHeader(token) {
  return apiGuard_(function () {
    var auth = requireAuth_(token);
    var peta = petaUser_();
    var out = {
      notif_belum_dibaca: 0,
      kadaluarsa: 0,
      mendesak: 0,
      perlu_verifikasi: 0,
      perlu_revisi: 0,
      total_sertifikasi: 0,
      badge_mendesak: 0,
      badge_verifikasi: 0
    };

    // --- notifikasi belum dibaca ---
    try {
      sheetToObjects_('T_Notifikasi').forEach(function (n) {
        if (String(n.Is_Read).toUpperCase() === 'TRUE') return;
        if (auth.role === 'EMP' && String(n.Employee_code) !== auth.employee_code) return;
        out.notif_belum_dibaca++;
      });
    } catch (e) { }

    // --- hitungan sertifikasi ---
    var rows = sheetToObjects_('T_Sertifikasi').map(function (x) { return fileDto_(x, peta); });
    if (auth.role === 'EMP') {
      rows = rows.filter(function (r) { return r.employee_id === auth.employee_code; });
    }
    out.total_sertifikasi = rows.length;
    rows.forEach(function (r) {
      if (r.status === 'KADALUARSA') out.kadaluarsa++;
      if (r.status === 'KADALUARSA' || r.status === 'MENDEKATI_KADALUARSA'
        || r.status === 'SEGERA_PERPANJANG') out.mendesak++;
      if (r.status_verifikasi === 'PENDING') out.perlu_verifikasi++;
      if (r.status_verifikasi === 'REVISI') out.perlu_revisi++;
    });
    // Dua hal yang BERBEDA, jadi badge-nya juga dipisah:
    //   badge_mendesak   = masa berlaku kritis (perlu diperpanjang)
    //   badge_verifikasi = antrean pemeriksaan HC (menunggu + perlu revisi)
    out.badge_mendesak = out.mendesak;
    out.badge_verifikasi = out.perlu_verifikasi + out.perlu_revisi;
    return out;
  });
}


/**
 * PEMULIHAN DARURAT — jalankan dari editor Apps Script, bukan dari web.
 * Dipakai bila HC & Super Admin sama-sama tidak bisa masuk.
 *
 *   repairPasswords_()                  -> SEMUA password = Employee Code
 *                                          masing-masing
 *   repairPasswords_('Patuha2026#')      -> semua password disamakan ke nilai itu
 *
 * Tidak ada hashing: nilai disimpan apa adanya (plaintext).
 * @param {string=} passwordBaru
 */
function repairPasswords_(passwordBaru) {
  var sh = getSheet_('Master_User');
  if (!sh) return { ok: false, msg: 'Sheet Master_User tidak ditemukan' };

  var rows = sheetToObjects_('Master_User');
  var diubah = 0;
  rows.forEach(function (u) {
    var sandi = (passwordBaru === undefined || passwordBaru === null || passwordBaru === '')
      ? String(u.Employee_code || '') : String(passwordBaru);
    try {
      updateRowByCol_('Master_User', u.__row, {
        Password: sandi,                  // default = Employee Code penuh
        Must_Change_Password: 'FALSE',
        Gagal_Login: 0,
        Terkunci_Sampai: '',
        Updated_At: nowStr_()
      });
      diubah++;
    } catch (e) { }
  });

  cabutSemuaSesi_('', null);
  try {
    PropertiesService.getScriptProperties().setProperty('MYCERTIF_REPAIR_AT', nowStr_());
  } catch (e) { }

  var hasil = {
    ok: true,
    diubah: diubah,
    mode: passwordBaru ? 'password seragam (plaintext)' : 'password = Employee Code masing-masing',
    catatan: 'Semua sesi lama dicabut.'
  };
  alertOrLog_('repairPasswords_ selesai.\nUser diubah: ' + diubah
    + '\nPassword: ' + hasil.mode + '\n\n' + hasil.catatan);
  return hasil;
}

/**
 * Utilitas: tandai ulang semua akun agar wajib ganti password
 * tanpa mengubah passwordnya. Jalankan dari editor.
 */
function wajibkanGantiPasswordSemua_() {
  var rows = sheetToObjects_('Master_User');
  var n = 0;
  rows.forEach(function (u) {
    try { updateRowByCol_('Master_User', u.__row, { Must_Change_Password: 'TRUE' }); n++; } catch (e) { }
  });
  alertOrLog_(n + ' akun ditandai wajib ganti password.');
  return { ok: true, diubah: n };
}
